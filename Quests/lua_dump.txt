-- ============================================================================
-- NSQC4 / Quests / Field / Bootstrap
-- Одна логика: порядок инициализации экземпляра поля + слэш /ns_field.
-- Грузится ПОСЛЕДНИМ в блоке Field/*.
--
-- БЕЗ таймеров и OnUpdate. Всё синхронно.
-- ОТЛАДКА: [BOOT]. Отключить — BOOT_DEBUG = false.
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local BOOT_DEBUG = false
    local function dbg(...)
        if BOOT_DEBUG then print("|cffffaa00[BOOT]|r", ...) end
    end

    local field = NSQC4.Field
    if not field then
        print("|cffff0000[NSQC4]|r Field/Bootstrap: NSQC4.Field не создан.")
        return
    end

    dbg("Старт. field=", tostring(field))

    -- ========================================================================
    -- Безопасный вызов метода: если метода нет — пишем в чат и идём дальше.
    -- ========================================================================
    local function call(method, ...)
        if type(field[method]) == "function" then
            dbg("call:", method)
            local ok, err = pcall(field[method], field, ...)
            if not ok then
                print("|cffff0000[NSQC4]|r Bootstrap:", method, "УПАЛ:", tostring(err))
            end
            return ok
        else
            print("|cffff0000[NSQC4]|r Bootstrap: метод", method, "НЕ ОПРЕДЕЛЁН (файл не загружен?)")
            return false
        end
    end

    -- ========================================================================
    -- Инициализация поля
    -- ========================================================================
    call("Load")
    call("CreateFrame")
    call("CreateCells")
    call("ReflowCells")
    call("InitPanels")

    -- 1) Свитчеры панелей
    call("CreateSwitchers")

    -- 2) Отрисовка API-кнопок
    call("RefreshAllButtons")

    -- 3) Раскладка свитчеров
    call("RefreshSwitchers")

    -- ========================================================================
    -- Ввод
    -- ========================================================================
    call("SetupMouse")
    call("SetupFrameClick")

    -- ========================================================================
    -- Оформление
    -- ========================================================================
    call("CreateBackdrop")
    call("SetupFrameDrag")
    call("CreateCloseButton")
    call("CreateResizeHandle")

    -- ========================================================================
    -- Слэш /ns_field
    -- ========================================================================
    SLASH_NSFIELD1 = "/ns_field"
    SlashCmdList["NSFIELD"] = function()
        local f = NSQC4.Field
        if f and f.Toggle then f:Toggle() end
    end
    dbg("Слэш /ns_field зарегистрирован")

    -- ========================================================================
    -- SlashPanel
    -- ========================================================================
    if NSQC4.SlashPanel_RegisterSectionButton then
        dbg("SlashPanel_RegisterSectionButton есть — регистрирую кнопку «Квесты»")
        NSQC4.SlashPanel_RegisterSectionButton(
            "Квесты", "П", "/ns_field",
            {
                { text = "Поле квестов 10×10", r = 1.00, g = 0.82, b = 0.00 },
            }
        )
    else
        print("|cffff0000[NSQC4]|r Bootstrap: NSQC4.SlashPanel_RegisterSectionButton == nil — кнопка на слэш-панели НЕ зарегистрирована")
    end

    -- ========================================================================
    -- Итоговая диагностика: что в итоге создано
    -- ========================================================================
    dbg("=== ИТОГ ===")
    dbg("frame:", tostring(field.frame))
    dbg("_switchers:", field._switchers and #field._switchers or "nil")
    dbg("_panelButtons:", field._panelButtons and "есть" or "nil")
    dbg("_mouseHandlers:", field._mouseHandlers and #field._mouseHandlers or "nil")

    if field._switchers then
        for i, rec in ipairs(field._switchers) do
            dbg(string.format("  switcher[%d] side=%s shown=%s",
                i, tostring(rec.side),
                tostring(rec.texture and rec.texture:IsShown())))
        end
    end

    print("|cff00ffff[FIELD]|r Блок 4 загружен (панели). /ns_field")
end, 630)

-- ============================================================================
-- NSQC4 / Quests / Field / Border / Drag
-- Одна логика: драг frame + кламп по экрану + Save.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local GRID_PADDING = CONST.GRID_PADDING

    local function ClampToCells(frame)
        local left   = frame:GetLeft()   or 0
        local right  = frame:GetRight()  or 0
        local top    = frame:GetTop()    or 0
        local bottom = frame:GetBottom() or 0

        local pad = GRID_PADDING
        local uiW = UIParent:GetWidth()
        local uiH = UIParent:GetHeight()

        local dx, dy = 0, 0
        if left + pad < 0 then dx = -(left + pad) end
        if right - pad > uiW then dx = uiW - (right - pad) end
        if bottom + pad < 0 then dy = -(bottom + pad) end
        if top - pad > uiH then dy = uiH - (top - pad) end

        if dx ~= 0 or dy ~= 0 then
            local x = frame:GetLeft()   or 0
            local y = frame:GetBottom() or 0
            frame:ClearAllPoints()
            frame:SetPoint("BOTTOMLEFT", UIParent, "BOTTOMLEFT", x + dx, y + dy)
        end
    end

    function QuestsField:StartDrag()
        local f = self.frame
        if not f then return end
        self._dragging = true
        f:StartMoving()
    end

    function QuestsField:StopDrag()
        local f = self.frame
        if not f then return end
        if not self._dragging then return end
        self._dragging = false
        f:StopMovingOrSizing()
        ClampToCells(f)
        self:Save()
    end

    -- навесить драг на frame
    function QuestsField:SetupFrameDrag()
        local f = self.frame
        if not f then return end

        f:RegisterForDrag("LeftButton")
        f:SetScript("OnDragStart", function()
            self:StartDrag()
        end)
        f:SetScript("OnDragStop", function()
            self:StopDrag()
        end)
    end

end, 370)

--
--   SetBorderVisible(v)     -- v = true → alpha 1 + title Show
--                             -- v = false → alpha 0.01 + title Hide
--   ToggleBorderVisible()   -- SetBorderVisible(not IsBorderVisible())
--   IsBorderVisible()       -- true | false
--
-- Подложка — текстура (Border/Backdrop.lua), self.backdrop.
-- Заголовок — FontString, self.backdropTitle.
--
-- Сохранение — в nsDbc4.quests.field.borderVisible (boolean).
-- Загрузка — при Load() (Core/Storage.lua) в self.borderVisible.
--
-- После смены — пересчёт кнопок панелей (opts.visible 2/3) и свитчеров.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    function QuestsField:IsBorderVisible()
        return self.borderVisible and true or false
    end

    function QuestsField:SetBorderVisible(v)
        local b = self.backdrop
        if not b then return end

        self.borderVisible = v and true or false

        if self.borderVisible then
            b:SetAlpha(1)
            if self.backdropTitle then self.backdropTitle:Show() end
        else
            b:SetAlpha(0.01)
            if self.backdropTitle then self.backdropTitle:Hide() end
        end

        self:Save()

        if self.RefreshAllButtons then self:RefreshAllButtons() end
        if self.RefreshSwitchers   then self:RefreshSwitchers()   end
    end

    function QuestsField:ToggleBorderVisible()
        self:SetBorderVisible(not self:IsBorderVisible())
    end

end, 360)

    function QuestsField:SetBorderless(v)
        self.borderless = v and true or false

        local lines = self.gridLines
        if lines then
            local show = not self.borderless
            for _, tex in ipairs(lines.v) do
                if show then tex:Show() else tex:Hide() end

        if self.frame and self.ReflowCells then
            self:ReflowCells()
        end

        self:Save()
    end

-- ============================================================================
-- NSQC4 / Quests / Field / Cells / Reflow
-- Одна логика: раскладка 20 линий сетки.
--
--   cellW = (fw - PAD*2 - (COLS-1)*GAP) / COLS
--   cellH = (fh - PAD*2 - (ROWS-1)*GAP) / ROWS
--
--   Вертикальная линия i (1..COLS+1) — на границе i-й колонки.
--   Горизонтальная линия j (1..ROWS+1) — на границе j-й строки.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS   = CONST.COLS, CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:ReflowCells()
        local f = self.frame
        if not f then return end

        local lines = self.gridLines
        if not lines then return end

        local fw, fh = f:GetWidth(), f:GetHeight()

        local cellW = (fw - GRID_PADDING * 2 - (COLS - 1) * CELL_GAP) / COLS
        local cellH = (fh - GRID_PADDING * 2 - (ROWS - 1) * CELL_GAP) / ROWS

        local gridW = fw - GRID_PADDING * 2
        local gridH = fh - GRID_PADDING * 2

        -- Вертикальные
        for i = 1, COLS + 1 do
            local tex = lines.v[i]
            local x = GRID_PADDING + (i - 1) * (cellW + CELL_GAP)
            tex:ClearAllPoints()
            tex:SetPoint("TOPLEFT",     f, "BOTTOMLEFT", x, GRID_PADDING + gridH)
            tex:SetPoint("BOTTOMLEFT",  f, "BOTTOMLEFT", x, GRID_PADDING)
            tex:SetWidth(1)
            tex:Show()
        end

        -- Горизонтальные
        for j = 1, ROWS + 1 do
            local tex = lines.h[j]
            local y = GRID_PADDING + (j - 1) * (cellH + CELL_GAP)
            tex:ClearAllPoints()
            tex:SetPoint("BOTTOMLEFT",  f, "BOTTOMLEFT", GRID_PADDING,            y)
            tex:SetPoint("BOTTOMRIGHT", f, "BOTTOMLEFT", GRID_PADDING + gridW,    y)
            tex:SetHeight(1)
            tex:Show()
        end
    end

end, 330)

        local c = {}
        c.__index = c

        if parent then
            setmetatable(c, { __index = parent })
        end

        function c.new(...)
            local o = setmetatable({}, c)
            if o.Init then
                o:Init(...)
            end
            return o
        end

        return c
    end

-- ============================================================================
-- NSQC4 / Quests / Field / Core / Storage
-- Одна логика: сохранение/загрузка состояния поля через NSDB.
--
--   QuestsField:Load()  -- NSDB:Get("field", ...) → self.saved + self.*
--   QuestsField:Save()  -- self.* → NSDB:Set("field", ...)
--
-- Таблица NSDB: "field".
-- Параметры:
--   x, y           — координаты BOTTOMLEFT (координаты поля)
--   cellSize       — размер клетки в px
--   borderless     — true | false (бордеры клеток)
--   borderVisible  — true | false (подложка: alpha 1 vs 0.01)
--
-- Никаких прямых обращений к nsDbc4 — только NSDB.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    local NSDB = NSQC4.NSDB
    if not NSDB then
        print("|cffff0000[NSQC4]|r Core/Storage: NSDB не загружен.")
        return
    end

    local SCOPE = "field"

    -- ========================================================================
    -- Load — NSDB → self.saved + self.*
    -- ========================================================================
    function QuestsField:Load()
        local borderVisible = NSDB:Get(SCOPE, "borderVisible")
        if borderVisible == nil then
            borderVisible = true          -- подложка видима по умолчанию
        else
            borderVisible = borderVisible and true or false
        end

        self.saved = {
            x             = NSDB:Get(SCOPE, "x"),
            y             = NSDB:Get(SCOPE, "y"),
            cellSize      = NSDB:Get(SCOPE, "cellSize"),
            borderless    = NSDB:Get(SCOPE, "borderless") and true or false,
            borderVisible = borderVisible,
        }

        self.borderless    = self.saved.borderless
        self.borderVisible = self.saved.borderVisible

        return self.saved
    end

    -- ========================================================================
    -- Save — self.* → NSDB
    -- ========================================================================
    function QuestsField:Save()
        if self.frame then
            NSDB:Set(SCOPE, "x", self.frame:GetLeft())
            NSDB:Set(SCOPE, "y", self.frame:GetBottom())
        end

        if self.cellSize then
            NSDB:Set(SCOPE, "cellSize", self.cellSize)
        end

        NSDB:Set(SCOPE, "borderless",    self.borderless and true or false)
        NSDB:Set(SCOPE, "borderVisible", self.borderVisible and true or false)
    end

end, 290)

-- ============================================================================
-- NSQC4 / Quests / Field / Frame / Create
-- Одна логика: создание главного frame поля.
--
-- Читает: QuestsFieldConst, self.saved.
-- Пишет:  self.frame, self.cellSize.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS     = CONST.COLS, CONST.ROWS
    local CELL_GAP       = CONST.CELL_GAP
    local GRID_PADDING   = CONST.GRID_PADDING
    local DEFAULT_CELL   = CONST.DEFAULT_CELL
    local MIN_CELL       = CONST.MIN_CELL
    local MAX_CELL       = CONST.MAX_CELL
    local SCREEN_PERCENT = CONST.SCREEN_PERCENT or 0.9

    -- ========================================================================
    -- FrameSizeForCell — размер frame по cellSize. Чистая функция.
    -- ========================================================================
    local function FrameSizeForCell(cs)
        return cs * COLS + (COLS - 1) * CELL_GAP + GRID_PADDING * 2,
               cs * ROWS + (ROWS - 1) * CELL_GAP + GRID_PADDING * 2
    end
    QuestsField._FrameSizeForCell = FrameSizeForCell

    -- ========================================================================
    -- MaxCellForScreen — предел cellSize по экрану (90% узкой стороны).
    -- ========================================================================
    function QuestsField:MaxCellForScreen()
        local uiW = (UIParent and UIParent:GetWidth())  or 0
        local uiH = (UIParent and UIParent:GetHeight()) or 0
        local minSide = math.min(uiW, uiH)
        if minSide <= 0 then return MAX_CELL end

        local maxSide = minSide * SCREEN_PERCENT
        local byW = (maxSide - (COLS - 1) * CELL_GAP) / COLS
        local byH = (maxSide - (ROWS - 1) * CELL_GAP) / ROWS
        local maxFromScreen = math.floor(math.min(byW, byH))

        if maxFromScreen < MIN_CELL then maxFromScreen = MIN_CELL end
        return math.min(MAX_CELL, maxFromScreen)
    end

    -- ========================================================================
    -- CreateFrame
    -- ========================================================================
    function QuestsField:CreateFrame()
        if self.frame then return self.frame end

        if not self.saved then self:Load() end
        local saved = self.saved or {}

        local effMaxCell = self:MaxCellForScreen()

        local cs = math.floor((saved.cellSize or DEFAULT_CELL) + 0.5)
        if cs < MIN_CELL then cs = MIN_CELL end
        if cs > effMaxCell then cs = effMaxCell end

        self.cellSize = cs

-- ============================================================================
-- NSQC4 / Quests / Field / Frame / Mouse
-- Одна логика: мышь. Только события OnMouseDown / OnMouseUp / OnMouseWheel
-- и координаты курсора в системе frame. Куда попал и что делать —
-- решают обработчики, зарегистрированные через API:
--   self:AddMouseHandler(fn)   -- fn(x, y, button, event) → true, если поглотил
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    -- ========================================================================
    -- API: регистрация обработчика мыши.
    --   fn(x, y, button, event)
    --     x, y   — координаты курсора в системе frame (BOTTOMLEFT = 0,0)
    --     button — "LeftButton" | "RightButton" | "MiddleButton"
    --     event  — "down" | "up" | "wheel"
    --   Возврат true — событие поглощено, следующие обработчики не зовутся.
    -- ========================================================================
    function QuestsField:AddMouseHandler(fn)
        if type(fn) ~= "function" then return end
        self._mouseHandlers = self._mouseHandlers or {}
        self._mouseHandlers[#self._mouseHandlers + 1] = fn
    end

    -- ========================================================================
    -- Координаты курсора в системе frame.
    -- ========================================================================
    function QuestsField:CursorInFrame()
        local f = self.frame
        if not f then return nil end
        local scale = UIParent:GetEffectiveScale()
        local cx, cy = GetCursorPosition()
        cx, cy = cx / scale, cy / scale
        local left, bottom = f:GetLeft(), f:GetBottom()
        if not left or not bottom then return nil end
        return cx - left, cy - bottom
    end

    local function Dispatch(self, button, event)
        local x, y = self:CursorInFrame()
        if not x then return end
        local handlers = self._mouseHandlers
        if not handlers then return end
        for _, fn in ipairs(handlers) do
            if fn(x, y, button, event) then return end
        end
    end

    -- ========================================================================
    -- SetupMouse — повесить скрипты на frame.
    -- ========================================================================
    function QuestsField:SetupMouse()
        local f = self.frame
        if not f then return end
        f:EnableMouse(true)
        f:EnableMouseWheel(true)

        f:SetScript("OnMouseDown", function(_, button)
            Dispatch(self, button, "down")
        end)

        f:SetScript("OnMouseUp", function(_, button)
            Dispatch(self, button, "up")
        end)

        f:SetScript("OnMouseWheel", function(_, delta)
            Dispatch(self, delta > 0 and "WheelUp" or "WheelDown", "wheel")
        end)
    end

-- ============================================================================
-- NSQC4 / Quests / Field / Resize / Apply
-- Одна логика: вычислить cellSize по позиции мыши и применить.
--
--   QuestsField:ApplyResize(cursorX, cursorY, anchorL, anchorT) → true | false
--
--   anchorL, anchorT — координаты левого-верхнего угла frame в момент
--     OnMouseDown (запоминает Handle.lua).
--   cursorX, cursorY — текущая позиция мыши (в координатах UIParent).
--
--   rawW = cursorX - anchorL
--   rawH = anchorT - cursorY
--   csX  = (rawW - (COLS-1)*GAP - 2*PAD) / COLS
--   csY  = (rawH - (ROWS-1)*GAP - 2*PAD) / ROWS
--   cs   = max(csX, csY)  (по большей стороне — тянем пропорционально)
--   → Resize(cs)
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS   = CONST.COLS, CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:ApplyResize(cursorX, cursorY, anchorL, anchorT)
        if not cursorX or not cursorY or not anchorL or not anchorT then
            return false
        end

        local rawW = cursorX - anchorL
        local rawH = anchorT - cursorY

        local csX = (rawW - (COLS - 1) * CELL_GAP - GRID_PADDING * 2) / COLS
        local csY = (rawH - (ROWS - 1) * CELL_GAP - GRID_PADDING * 2) / ROWS
        local cs  = math.max(csX, csY)

        return self:Resize(cs)
    end

end, 410)

    function QuestsField:Resize(cs)

        if cs < MIN_CELL then cs = MIN_CELL end

        if self.RefreshSwitchers then self:RefreshSwitchers() end

        if self.ApplyBorderAnchors then self:ApplyBorderAnchors() end

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / CellCoords
-- Одна логика: геометрия. CellCenterXY / PanelButtonXY.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS   = CONST.COLS, CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:CellCenterXY(index)
        local f = self.frame
        if not f then return nil end
        local fw, fh = f:GetWidth(), f:GetHeight()

        local i = (tonumber(index) or 0) - 1
        local col = i % COLS
        local row = math.floor(i / COLS)

        local cellW = (fw - GRID_PADDING * 2 - (COLS - 1) * CELL_GAP) / COLS
        local cellH = (fh - GRID_PADDING * 2 - (ROWS - 1) * CELL_GAP) / ROWS

        local cx = GRID_PADDING + col * (cellW + CELL_GAP) + cellW / 2
        local cy = GRID_PADDING + row * (cellH + CELL_GAP) + cellH / 2
        return cx, cy
    end

    function QuestsField:PanelButtonXY(side, index)
        local f = self.frame
        if not f then return nil end
        local fw, fh = f:GetWidth(), f:GetHeight()
        local cx, cy = self:CellCenterXY(index)
        if not cx then return nil end

        local half = GRID_PADDING / 2

        if side == "RIGHT" then
            return fw - half, cy
        elseif side == "LEFT" then
            return half, cy
        elseif side == "TOP" then
            return cx, fh - half
        elseif side == "BOTTOM" then
            return cx, half
        end
        return nil
    end

end, 440)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Init
-- Универсальный рендерер кнопок.
--
-- НИЧЕГО не знает о поле (QuestsField) и об окнах (FieldWindow).
-- Рисует кнопки на любом parent-фрейме, где скажут.
--
-- Публичный API (NSQC4.UIButton):
--   CreateRecord(label, onClick, opts)       → rec
--   Draw(parent, rec, y, availableW)         → height
--   HitTest(rec)                             → boolean
--   FindHit(items)                           → rec | nil
--   Press(rec)
--   Release()
--   SetupMouse(parent, getItems)
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local UIButton = {}
    NSQC4.UIButton = UIButton

    local BUTTON_H = 24
    local PIXELS   = 2

    -- ========================================================================
    -- CreateRecord — создать запись кнопки (без отрисовки).
    -- ========================================================================
    function UIButton:CreateRecord(label, onClick, opts)
        return {
            kind    = "button",
            label   = label,
            onClick = onClick,
            opts    = opts or {},
        }
    end

    -- ========================================================================
    -- Draw — отрисовать кнопку на parent в позиции y.
    -- Возвращает высоту блока (для раскладки).
    -- ========================================================================
    function UIButton:Draw(parent, rec, y, availableW)
        if not parent then return 0 end

        if not rec.frame then
            local tex = parent:CreateTexture(nil, "OVERLAY")
            tex:SetTexture("Interface\\Buttons\\WHITE8x8")
            tex:SetVertexColor(0.4, 0, 0, 1)

            local fs = parent:CreateFontString(nil, "OVERLAY", "GameFontNormal")
            fs:SetText(rec.label)
            fs:SetPoint("CENTER", tex, "CENTER", 0, 0)

            rec.frame = tex
            rec.text  = fs
        end

        rec.frame:ClearAllPoints()
        rec.frame:SetPoint("TOPLEFT", parent, "TOPLEFT", 8, y)
        rec.frame:SetSize(availableW, BUTTON_H)
        rec.frame:Show()

        if rec.text then rec.text:Show() end

        return BUTTON_H + 4
    end

    -- ========================================================================
    -- HitTest — попал ли курсор в кнопку.
    -- ========================================================================
    function UIButton:HitTest(rec)
        if not rec or not rec.frame or not rec.frame:IsShown() then
            return false
        end

        local left  = rec.frame:GetLeft()
        local right = rec.frame:GetRight()
        local top   = rec.frame:GetTop()
        local bot   = rec.frame:GetBottom()

        if not (left and right and top and bot) then return false end

        local scale = UIParent:GetEffectiveScale()
        local cx, cy = GetCursorPosition()
        cx, cy = cx / scale, cy / scale

        return cx >= left and cx <= right and cy <= top and cy >= bot
    end

    -- ========================================================================
    -- FindHit — найти кнопку под курсором в таблице items.
    -- ========================================================================
    function UIButton:FindHit(items)
        if not items then return nil end

        for _, rec in pairs(items) do
            if rec.kind == "button" and self:HitTest(rec) then
                return rec
            end
        end

        return nil
    end

    -- ========================================================================
    -- Press / Release — анимация нажатия.
    -- ========================================================================
    function UIButton:Press(rec)
        if not rec or not rec.frame then return end

        local tex = rec.frame
        local w, h = tex:GetWidth(), tex:GetHeight()
        if not w or not h then return end

        tex:SetSize(w - PIXELS * 2, h - PIXELS * 2)
        self._pressed = { tex = tex, w = w, h = h }
    end

    function UIButton:Release()
        local p = self._pressed
        if not p then return end

        self._pressed = nil
        if p.tex and p.w and p.h then
            p.tex:SetSize(p.w, p.h)
        end
    end

    -- ========================================================================
    -- SetupMouse — подключить мышь к parent.
    --   getItems() — функция, возвращающая таблицу кнопок для хит-теста.
    -- ========================================================================
    function UIButton:SetupMouse(parent, getItems)
        if not parent or parent._uiButtonMouseSetup then return end
        parent._uiButtonMouseSetup = true
        parent:EnableMouse(true)

        parent:SetScript("OnMouseDown", function(_, button)
            if button ~= "LeftButton" then return end

            local items = getItems()
            local rec   = UIButton:FindHit(items)

            if rec then
                UIButton:Press(rec)
                if rec.onClick then rec.onClick(button) end
            end
        end)

        parent:SetScript("OnMouseUp", function()
            UIButton:Release()
        end)
    end
end, 530)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Panel
-- Одна логика: текущий panelIndex стороны (из NSDB) + смена + переключение.
--
-- Хранилище: NSDB scope "panel", ключ "side" → число (1..N).

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local NSDB        = NSQC4.NSDB
    if not QuestsField or not NSDB then return end

    local SCOPE = "panel"
    local SIDES = { "TOP", "BOTTOM", "LEFT", "RIGHT" }

    function QuestsField:GetPanelIndex(side)
        local n = NSDB:Get(SCOPE, side)
        if n == nil then
            NSDB:Set(SCOPE, side, 1)
            return 1
        end
        return n
    end

    function QuestsField:SetPanelIndex(side, n)
        n = tonumber(n) or 1
        NSDB:Set(SCOPE, side, n)
        if self.RefreshAllButtons then self:RefreshAllButtons() end
        if self.RefreshSwitchers   then self:RefreshSwitchers()   end
    end

    function QuestsField:GetPanelCount(side)
        local maxN = 0
        local buttons = NSDB:Table("buttons")
        if buttons then
            for _, data in pairs(buttons) do
                if data.side == side and data.panelIndex and data.panelIndex > maxN then
                    maxN = data.panelIndex
                end
            end
        end
        if maxN < 1 then maxN = 1 end
        return maxN
    end

    local RADIO_H, RADIO_GAP, MARGIN, ICON_W, ICON_GAP = 24, 6, 8, 24, 12

    local function SafeName(windowId, groupId, value)
        local raw = string.format("NSQC4FieldRadio_%s_%s_%s",
            tostring(windowId), tostring(groupId), tostring(value))
        return (raw:gsub("[^%w_]", "_"))
    end

    function W:EnsureRadioGroup(windowId, groupId)
        local win = self[windowId]
        if not win then return nil end
        local grp = self._radioGroups and self._radioGroups[windowId]
                        and self._radioGroups[windowId][groupId]
        if not grp then return nil end
        if grp._buttons then return grp._buttons end

        grp._buttons = {}
        for _, opt in ipairs(grp.options or {}) do
            local name = SafeName(windowId, groupId, opt.value)
            local rb = _G[name] or CreateFrame("CheckButton", name, win, "UIRadioButtonTemplate")
            rb:SetParent(win)
            rb:SetFrameLevel(win:GetFrameLevel() + 5)
            rb:EnableMouse(true)
            rb.value = opt.value

            if fs then fs:Hide() end
            local fsG = _G[name .. "Text"]
            if fsG then fsG:Hide() end

                rb:SetScript("OnEnter", function(s)
                    GameTooltip:SetOwner(s, "ANCHOR_RIGHT")
                    GameTooltip:ClearLines()
                    GameTooltip:AddLine(opt.tooltip, 1, 1, 1, true)
                    GameTooltip:Show()
                end)
                rb:SetScript("OnLeave", function() GameTooltip:Hide() end)
            end

            rb:SetScript("OnClick", function(s)
                for _, o in pairs(grp._buttons) do o:SetChecked(o == s) end
                if grp.onSelect then grp.onSelect(s.value) end
            end)

            grp._buttons[opt.value] = rb
        end
        return grp._buttons
    end

        if not grp then return 0 end

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Window / Layout
-- Одна логика: управление расположением элементов окна по вертикали (Y).
--
-- Не создает фреймы, текстуры или текст. Только двигает.
-- Порядок — строго по индексу (table.sort по числовым ключам).
-- Кнопки окна рисует универсальный рендерер NSQC4.UIButton:Draw.
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local W = NSQC4.FieldWindow
    if not W then return end

    local START_Y   = -8
    local CONTENT_W = 16 -- отступы 8 слева + 8 справа

    function W:LayoutWindow(windowId)
        local win = self[windowId]
        if not win then return end

        local items = self._items and self._items[windowId]
        if not items then return end

        -- 1. Собираем и сортируем индексы (числовые — как числа)
        local indices = {}
        for i in pairs(items) do
            indices[#indices + 1] = i
        end
        table.sort(indices, function(a, b)
            if type(a) == "number" and type(b) == "number" then
                return a < b
            end
            return tostring(a) < tostring(b)
        end)

        local y          = START_Y
        local availableW = win:GetWidth() - CONTENT_W
        local UIButton   = NSQC4.UIButton

        -- 2. Проходим сверху вниз
        for _, i in ipairs(indices) do
            local rec = items[i]
            if rec then
                local h = 0

                if rec.kind == "label" and self.DrawLabel then
                    h = self:DrawLabel(windowId, rec, y, availableW)

                elseif rec.kind == "button" and UIButton and UIButton.Draw then
                    h = UIButton:Draw(win, rec, y, availableW)

                elseif rec.kind == "radio" and self.DrawRadio then
                    h = self:DrawRadio(windowId, rec, y, availableW)
                end

                y = y - (h or 0)
            end
        end
    end

-- NSQC4 / Quests / NSDB / Debug
-- Одна логика: /ns_db tables | dump <scope> | clear <scope> | drop <scope>.

NSQC4.RegisterModule("quests", function()

    local NSDB = NSQC4.NSDB
    if not NSDB then return end

    local function DumpTree(t, prefix, depth)
        if depth > 6 then return end
        local keys = {}
        for k in pairs(t) do keys[#keys + 1] = k end
        table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
        for _, k in ipairs(keys) do
            local v = t[k]
            if type(v) == "table" then
                print(string.format("  %s%s = {", prefix, tostring(k)))
                DumpTree(v, prefix .. "  ", depth + 1)
                print(string.format("  %s}", prefix))
            else
                print(string.format("  %s%s = %s", prefix, tostring(k), tostring(v)))
            end
        end
    end

    SLASH_NSDB1 = "/ns_db"
    SlashCmdList["NSDB"] = function(msg)
        local cmd, arg = (msg or ""):match("^(%S*)%s*(.-)$")
        cmd = (cmd or ""):lower()
        arg = arg or ""

        if cmd == "" or cmd == "tables" then
            local root = NSDB:Root()
            local names = {}
            for n in pairs(root) do names[#names + 1] = n end
            table.sort(names)
            print(string.format("[NSDB] таблиц верхнего уровня: %d", #names))
            for _, n in ipairs(names) do
                print(string.format("  %s", n))
            end

        elseif cmd == "dump" then
            if arg == "" then
                print("[NSDB] /ns_db dump <scope>")
                return
            end
            local t = NSDB:Walk(arg)
            if not t then
                print("[NSDB] нет таблицы: " .. arg)
                return
            end
            print(string.format("[NSDB] %s = {", arg))
            DumpTree(t, "  ", 0)
            print("}")

        elseif cmd == "clear" then
            if arg == "" then
                print("[NSDB] /ns_db clear <scope>")
                return
            end
            print(string.format("[NSDB] clear %s → %s", arg,
                tostring(NSDB:Clear(arg))))

        elseif cmd == "drop" then
            if arg == "" then
                print("[NSDB] /ns_db drop <scope>")
                return
            end
            print(string.format("[NSDB] drop %s → %s", arg,
                tostring(NSDB:DropTable(arg))))

        else
            print("[NSDB] /ns_db tables | dump <scope> | clear <scope> | drop <scope>")
        end
    end

    print("|cff00ffff[NSDB]|r База данных загружена. /ns_db tables | dump <scope>")

-- ============================================================================
-- NSQC4 / Quests / NSDB / Root
-- Одна логика: корень базы (nsDbc4.quests) + навигация по пути scope.
--
--   Root()              -- nsDbc4.quests (гарантирует)
--   Walk(scope)         -- таблица по пути, БЕЗ создания промежуточных
--   WalkCreate(scope)   -- таблица по пути, С созданием промежуточных
--
-- scope = "field.pos" → nsDbc4.quests.field.pos
-- scope = "field"     → nsDbc4.quests.field
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local NSDB = NSQC4.NSDB
    if not NSDB then return end

    function NSDB:Root()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        return nsDbc4.quests
    end

    -- Walk(scope) → таблица | nil (не создаёт).
    function NSDB:Walk(scope)
        if type(scope) ~= "string" or scope == "" then return nil end
        local cur = self:Root()
        for part in scope:gmatch("[^%.]+") do
            cur = cur[part]
            if cur == nil then return nil end
            if type(cur) ~= "table" then return nil end
        end
        return cur
    end

    -- WalkCreate(scope) → таблица (создаёт промежуточные).
    function NSDB:WalkCreate(scope)
        if type(scope) ~= "string" or scope == "" then return nil end
        local cur = self:Root()
        for part in scope:gmatch("[^%.]+") do
            if type(cur[part]) ~= "table" then
                cur[part] = {}
            end
            cur = cur[part]
        end
        return cur
    end

end, 180)

-- ============================================================================
-- NSQC4 / Quests / NSDB / Set
-- Одна логика: Set(scope, key, value) — записать параметр.
--
--   scope — путь таблицы ("field", "field.pos")
--   key   — имя параметра ("x", "cellSize")
--   value = nil → Delete(scope, key).
--   Таблица по пути scope создаётся, если нет.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local NSDB = NSQC4.NSDB
    if not NSDB then return end

    function NSDB:Set(scope, key, value)
        if not self:ValidName(scope) then return false end
        if key == nil then return false end

        local t = self:WalkCreate(scope)
        if not t then return false end

        if value == nil then
            t[key] = nil
        else
            t[key] = value
        end
        return true
    end

end, 200)
