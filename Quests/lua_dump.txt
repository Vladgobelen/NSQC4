-- NSQC4 / Quests / Field / Bootstrap / Init

NSQC4.RegisterModule("quests", function()

    local field = NSQC4.Field
    if not field then
        print("|cffff0000[NSQC4]|r Bootstrap/Init: NSQC4.Field не создан.")
        return
    end

    -- Безопасный вызов метода
    local function call(method, ...)
        if type(field[method]) == "function" then
            local ok, err = pcall(field[method], field, ...)
            if not ok then
                print("|cffff0000[NSQC4]|r Bootstrap/Init:", method, "УПАЛ:", tostring(err))
            end
            return ok
        else
            print("|cffff0000[NSQC4]|r Bootstrap/Init: метод", method, "НЕ ОПРЕДЕЛЁН")
            return false
        end
    end

    -- ========================================================================
    -- Инициализация поля
    -- ========================================================================
    call("Load")
    call("CreateFrame")
    call("CreateCells")
    call("ReflowCells")
    call("InitPanels")

    -- Свитчеры панелей
    call("CreateSwitchers")

    -- Отрисовка API-кнопок
    call("RefreshAllButtons")

    -- Раскладка свитчеров
    call("RefreshSwitchers")

    -- ========================================================================
    -- Ввод (мышь)
    -- ПКМ регистрируется в Border/Click.lua через AddMouseHandler.
    -- ========================================================================
    call("SetupMouse")

    -- ========================================================================
    -- Оформление
    -- ========================================================================
    call("CreateBackdrop")
    call("SetupFrameDrag")
    call("CreateCloseButton")
    call("CreateResizeHandle")

-- ============================================================================
-- NSQC4 / Quests / Field / Border / Click
-- Одна логика: ПКМ по frame → ToggleBorderVisible().
--
-- Регистрация — через единый регистратор мыши поля
-- (Field/Frame/Mouse.lua → AddMouseHandler).
--
-- ВАЖНО: вес 395 > 390 (Frame/Mouse.lua), иначе AddMouseHandler ещё не создан.
--
-- ОТЛАДКА: [CLICK]. Отключить — CLICK_DEBUG = false.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local CLICK_DEBUG = true
    local function dbg(...)
        if CLICK_DEBUG then print("|cffffaa00[CLICK]|r", ...) end
    end

    local field = NSQC4.Field
    if not field then
        print("|cffff0000[CLICK]|r NSQC4.Field == nil — обработчик не зарегистрирован")
        return
    end

    if type(field.AddMouseHandler) ~= "function" then
        print("|cffff0000[CLICK]|r field:AddMouseHandler == nil — Frame/Mouse.lua не загружен?")
        return
    end

    dbg("Регистрирую обработчик ПКМ через AddMouseHandler")

    field:AddMouseHandler(function(x, y, button, event)
        dbg(string.format("handler: event=%s button=%s x=%s y=%s",
            tostring(event), tostring(button),
            tostring(x), tostring(y)))

        if event ~= "down" then return false end
        if button ~= "RightButton" then return false end

        dbg("ПКМ down — вызываю ToggleBorderVisible")

        if type(field.ToggleBorderVisible) ~= "function" then
            print("|cffff0000[CLICK]|r field:ToggleBorderVisible == nil — Border/Visible.lua не загружен?")
            return false
        end

        local before = field:IsBorderVisible()
        field:ToggleBorderVisible()
        local after = field:IsBorderVisible()

        dbg(string.format("borderVisible: %s → %s", tostring(before), tostring(after)))

        return true
    end)

    dbg("Обработчик зарегистрирован. Всего обработчиков:",
        field._mouseHandlers and #field._mouseHandlers or "nil")

    if field._mouseHandlers then
        for i, fn in ipairs(field._mouseHandlers) do
            dbg(string.format("  handler[%d] = %s", i, tostring(fn)))
        end
    end

end, 395)

-- ============================================================================
-- NSQC4 / Quests / Field / Border / Drag / Clamp
-- Одна логика: удержание frame в пределах экрана.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:ClampFrameToScreen()
        local frame = self.frame
        if not frame then return end

        local left   = frame:GetLeft()   or 0
        local right  = frame:GetRight()  or 0
        local top    = frame:GetTop()    or 0
        local bottom = frame:GetBottom() or 0

        local pad = GRID_PADDING
        local uiW = UIParent:GetWidth()
        local uiH = UIParent:GetHeight()

        local dx, dy = 0, 0
        if left + pad < 0 then dx = -(left + pad) end
        if right - pad > uiW then dx = uiW - (right - pad) end
        if bottom + pad < 0 then dy = -(bottom + pad) end
        if top - pad > uiH then dy = uiH - (top - pad) end

        if dx ~= 0 or dy ~= 0 then
            local x = frame:GetLeft()   or 0
            local y = frame:GetBottom() or 0
            frame:ClearAllPoints()
            frame:SetPoint("BOTTOMLEFT", UIParent, "BOTTOMLEFT", x + dx, y + dy)
        end
    end
end, 371)

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS = CONST.COLS, CONST.ROWS

    local LINE_COLOR  = { 0.5, 0.5, 0.5, 1 }
    local PRESS_ALPHA = 0.35

    function QuestsField:CreateCells()
        if self.gridLines then return self.gridLines end

        local f = self.frame
        if not f then return nil end

        local lines = { v = {}, h = {} }

        -- Вертикальные линии: COLS + 1
        for i = 1, COLS + 1 do
            local tex = f:CreateTexture(nil, "ARTWORK", nil, 4)
            tex:SetTexture("Interface\\Buttons\\WHITE8x8")
            tex:SetVertexColor(LINE_COLOR[1], LINE_COLOR[2], LINE_COLOR[3], LINE_COLOR[4])
            tex:SetWidth(1)
            tex:Hide()
            lines.v[i] = tex
        end

        -- Горизонтальные линии: ROWS + 1
        for i = 1, ROWS + 1 do
            local tex = f:CreateTexture(nil, "ARTWORK", nil, 4)
            tex:SetTexture("Interface\\Buttons\\WHITE8x8")
            tex:SetVertexColor(LINE_COLOR[1], LINE_COLOR[2], LINE_COLOR[3], LINE_COLOR[4])

        press:SetTexture("Interface\\Buttons\\WHITE8x8")
        press:SetVertexColor(0, 0, 0, PRESS_ALPHA)

-- ============================================================================
-- NSQC4 / Quests / Field / Cells / Reflow / Refresh
-- Одна логика: оркестрация раскладки сетки (вертикальные + горизонтальные).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS         = CONST.COLS
    local ROWS         = CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:ReflowCells()
        local f = self.frame
        if not f then return end

        local lines = self.gridLines
        if not lines then return end

        local fw, fh = f:GetWidth(), f:GetHeight()

        local cellW = (fw - GRID_PADDING * 2 - (COLS - 1) * CELL_GAP) / COLS
        local cellH = (fh - GRID_PADDING * 2 - (ROWS - 1) * CELL_GAP) / ROWS

        local gridW = fw - GRID_PADDING * 2
        local gridH = fh - GRID_PADDING * 2

        -- Вызов вертикалей и горизонталей
        self:ReflowVerticalLines(f, cellW, gridH)
        self:ReflowHorizontalLines(f, cellH, gridW)
    end

end, 333)

-- NSQC4 / Quests / Field / Core / Class
-- Мини-система классов. Одна логика: создание класса по прототипу.
--
--   local MyClass = NSQC4.Class()          -- без родителя
--   local MyClass = NSQC4.Class(Parent)    -- с родителем

-- ============================================================================

NSQC4.RegisterModule("quests", function()

    function NSQC4.Class(parent)
        local c = {}
        c.__index = c

        if parent then
            setmetatable(c, { __index = parent })
        end

        function c.new(...)
            local o = setmetatable({}, c)
            if o.Init then
                o:Init(...)
            end
            return o
        end

        return c
    end

-- ============================================================================
-- NSQC4 / Quests / Field / Frame / Create
-- Одна логика: создание главного frame поля.
--
-- Читает: QuestsFieldConst, self.saved.
-- Пишет:  self.frame, self.cellSize.
--
-- ФОН НЕ создаётся здесь — его рисует Border/Backdrop.lua
-- (текстура BACKGROUND sublevel -1, ниже сетки/кнопок/свитчеров).
--
-- ОТЛАДКА: [FRAME]. Отключить — FRAME_DEBUG = false.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local FRAME_DEBUG = true
    local function dbg(...)
        if FRAME_DEBUG then print("|cffffaa00[FRAME]|r", ...) end
    end

    local COLS, ROWS     = CONST.COLS, CONST.ROWS
    local CELL_GAP       = CONST.CELL_GAP
    local GRID_PADDING   = CONST.GRID_PADDING
    local DEFAULT_CELL   = CONST.DEFAULT_CELL
    local MIN_CELL       = CONST.MIN_CELL
    local MAX_CELL       = CONST.MAX_CELL
    local SCREEN_PERCENT = CONST.SCREEN_PERCENT or 0.9

    -- ========================================================================
    -- FrameSizeForCell — размер frame по cellSize. Чистая функция.
    -- ========================================================================
    local function FrameSizeForCell(cs)
        return cs * COLS + (COLS - 1) * CELL_GAP + GRID_PADDING * 2,
               cs * ROWS + (ROWS - 1) * CELL_GAP + GRID_PADDING * 2
    end
    QuestsField._FrameSizeForCell = FrameSizeForCell

    -- ========================================================================
    -- MaxCellForScreen — предел cellSize по экрану (90% узкой стороны).
    -- ========================================================================
    function QuestsField:MaxCellForScreen()
        local uiW = (UIParent and UIParent:GetWidth())  or 0
        local uiH = (UIParent and UIParent:GetHeight()) or 0
        local minSide = math.min(uiW, uiH)
        if minSide <= 0 then return MAX_CELL end

        local maxSide = minSide * SCREEN_PERCENT
        local byW = (maxSide - (COLS - 1) * CELL_GAP) / COLS
        local byH = (maxSide - (ROWS - 1) * CELL_GAP) / ROWS
        local maxFromScreen = math.floor(math.min(byW, byH))

        if maxFromScreen < MIN_CELL then maxFromScreen = MIN_CELL end
        return math.min(MAX_CELL, maxFromScreen)
    end

    -- ========================================================================
    -- CreateFrame
    -- ========================================================================
    function QuestsField:CreateFrame()
        if self.frame then
            dbg("frame уже существует, возвращаю")
            return self.frame
        end

        if not self.saved then self:Load() end
        local saved = self.saved or {}

        local effMaxCell = self:MaxCellForScreen()

        local cs = math.floor((saved.cellSize or DEFAULT_CELL) + 0.5)
        if cs < MIN_CELL then cs = MIN_CELL end
        if cs > effMaxCell then cs = effMaxCell end

        self.cellSize = cs

        local w0, h0 = FrameSizeForCell(cs)

        dbg(string.format("создаю frame: cs=%d w=%d h=%d", cs, w0, h0))

        local f = CreateFrame("Frame", "NSQC4FieldFrame", UIParent)
        self.frame = f
        f:SetFrameStrata("MEDIUM")
        f:SetMovable(true)
        f:SetClampedToScreen(false)
        f:SetSize(w0, h0)

        -- ВАЖНО: фон НЕ создаётся здесь.
        -- Фон рисует Border/Backdrop.lua:
        --   local tex = f:CreateTexture(nil, "BACKGROUND", nil, -1)
        --   tex:SetAllPoints(f)
        --   tex:SetVertexColor(0, 0, 0, 0.85)
        -- sublevel -1 — ниже сетки (BACKGROUND, sublevel 1),
        -- кнопок (OVERLAY, 10) и свитчеров (OVERLAY, 11).

        f:EnableMouse(false)

        if saved.x and saved.y then
            f:SetPoint("BOTTOMLEFT", UIParent, "BOTTOMLEFT", saved.x, saved.y)
            dbg(string.format("позиция из saved: %s, %s", tostring(saved.x), tostring(saved.y)))
        else
            f:SetPoint("CENTER")
            dbg("позиция: CENTER (по умолчанию)")
        end

        f:Hide()
        dbg("frame создан и скрыт")
        return f
    end

--   csY  = (rawH - (ROWS-1)*GAP - 2*PAD) / ROWS
--   cs   = max(csX, csY)  (по большей стороне — тянем пропорционально)

    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:ApplyResize(cursorX, cursorY, anchorL, anchorT)
        if not cursorX or not cursorY or not anchorL or not anchorT then
            return false
        end

        local rawW = cursorX - anchorL
        local rawH = anchorT - cursorY

        local csX = (rawW - (COLS - 1) * CELL_GAP - GRID_PADDING * 2) / COLS
        local csY = (rawH - (ROWS - 1) * CELL_GAP - GRID_PADDING * 2) / ROWS
        local cs  = math.max(csX, csY)

        return self:Resize(cs)
    end

-- ============================================================================
-- NSQC4 / Quests / Field / Switcher / Init
-- Одна логика: создание свитчеров панелей.
-- БЕЗ таймеров, без OnUpdate.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local NSDB        = NSQC4.NSDB

    if not QuestsField then return end
    if not NSDB then return end

    local SCOPE       = "switchers"
    local SWITCH_SLOT = 1
    local SIDES       = { "TOP", "BOTTOM", "LEFT", "RIGHT" }

    local function Key(side)
        return side .. ":" .. SWITCH_SLOT
    end

    local function Defaults(side)
        return {
            icon    = "Interface\\Buttons\\UI-SpellbookIcon-NextPage-Up",
            color   = { 1, 1, 1, 1 },
            press   = 2,
            visible = 1,
            tooltip = "Панель " .. side .. ": ЛКМ вперёд, ПКМ назад",
        }
    end

    local function Ensure(side)
        local key = Key(side)
        local existing = NSDB:Get(SCOPE, key)
        if existing then return existing end
        local defaults = Defaults(side)
        NSDB:Set(SCOPE, key, defaults)
        return defaults
    end

    function QuestsField:HasPanelButtons(side)
        local buttons = NSDB:Table("buttons")
        if not buttons then return false end
        local cur = self:GetPanelIndex(side)
        for _, data in pairs(buttons) do
            if data.side == side and data.panelIndex == cur then
                return true
            end
        end
        return false
    end

    function QuestsField:CreateSwitchers()
        if self._switchers then
            return self._switchers
        end
        if not self.frame then
            return nil
        end

        self._switchers = {}
        local bs = self:GetButtonSize()

        for _, side in ipairs(SIDES) do
            local settings = Ensure(side)
            local tex = self.frame:CreateTexture(nil, "OVERLAY", nil, 11)
            tex:SetTexture(settings.icon)
            tex:SetSize(bs, bs)
            tex:Hide()

            self._switchers[#self._switchers + 1] = {
                texture    = tex,
                side       = side,
                slot       = SWITCH_SLOT,
                settings   = settings,
                isSwitcher = true,
            }
        end

        if self.RefreshSwitchers then
            self:RefreshSwitchers()
        end

        return self._switchers
    end

    local inst = NSQC4.Field
    if inst and inst.frame and not inst._switchers then
        inst:CreateSwitchers()
    end

end, 605)

--
--   event "up"   → ничего не делаем (анимацию снимает Buttons/Mouse.lua
--                  через field:OnMouseUp() → ReleaseTexture)
--   event "down" → HitSwitcher → MiddleButton: ToggleTooltipSwitcher
--                               → LeftButton / RightButton:
--                                   PressTexture + OnSwitcherClick
--
-- НЕ рисует (это делает Draw.lua).
-- НЕ создаёт текстуры (это делает Init.lua).
-- НЕ хит-тестит (это делает Click.lua).
-- Анимация нажатия — через Press.lua (PressTexture / ReleaseTexture).
--
-- ВАЖНО: вес 615 < 620 (Bootstrap/Init.lua) — обработчик регистрируется
-- ДО того, как Bootstrap вызовет SetupMouse. Иначе коллизия весов 620
-- с Bootstrap/Init.lua делает порядок недетерминированным.
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local field = NSQC4.Field
    if not field then return end

    field:AddMouseHandler(function(x, y, button, event)
        -- Отпускание мыши — анимацию уже снял Buttons/Mouse.lua
        -- через field:OnMouseUp() → ReleaseTexture().
        -- Возвращаем false, чтобы не блокировать другие обработчики.
        if event == "up" then
            return false
        end

        if event ~= "down" then return false end

        local sw = field.HitSwitcher and field:HitSwitcher(x, y)
        if sw then
            if button == "MiddleButton" then
                field:ToggleTooltipSwitcher(sw)
            else
                field:PressTexture(sw.texture)
                field:OnSwitcherClick(sw, button)
            end

            return true  -- событие поглощено
        end

        return false
    end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / CellCoords
-- Одна логика: геометрия. CellCenterXY / PanelButtonXY.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS   = CONST.COLS, CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    function QuestsField:CellCenterXY(index)
        local f = self.frame
        if not f then return nil end
        local fw, fh = f:GetWidth(), f:GetHeight()

        local i = (tonumber(index) or 0) - 1
        local col = i % COLS
        local row = math.floor(i / COLS)

        local cellW = (fw - GRID_PADDING * 2 - (COLS - 1) * CELL_GAP) / COLS
        local cellH = (fh - GRID_PADDING * 2 - (ROWS - 1) * CELL_GAP) / ROWS

        local cx = GRID_PADDING + col * (cellW + CELL_GAP) + cellW / 2
        local cy = GRID_PADDING + row * (cellH + CELL_GAP) + cellH / 2
        return cx, cy
    end

    function QuestsField:PanelButtonXY(side, index)
        local f = self.frame
        if not f then return nil end
        local fw, fh = f:GetWidth(), f:GetHeight()
        local cx, cy = self:CellCenterXY(index)
        if not cx then return nil end

        local half = GRID_PADDING / 2

        if side == "RIGHT" then
            return fw - half, cy
        elseif side == "LEFT" then
            return half, cy
        elseif side == "TOP" then
            return cx, fh - half
        elseif side == "BOTTOM" then
            return cx, half
        end
        return nil
    end

end, 440)

-- ============================================================================

-- Вызывает:
--   ShouldShowButton       (Visible.lua)

--   ApplyButtonAppearance  (Appearance.lua)

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local NSDB        = NSQC4.NSDB
    if not QuestsField or not NSDB then return end

    local STEP = {
        TOP    =  1,
        RIGHT  = -10,
        LEFT   = -10,
        BOTTOM =  1,
    }

    local FIRST_CELL = {
        TOP    = 91,
        RIGHT  = 100,
        LEFT   = 91,
        BOTTOM = 1,
    }

    local function CellForSlot(side, slot)
        local step  = STEP[side]
        local first = FIRST_CELL[side]
        if not step or not first or not slot then return nil end
        return first + step * slot

        if not f then return end

        self._panelButtons = self._panelButtons or {}
        local buttons = NSDB:Table("buttons")
        if not buttons then return end

        for key, data in pairs(buttons) do
            local side = data.side
            local cellIndex = data.slot and CellForSlot(side, data.slot)

            if cellIndex then
                -- Создаем текстуру, если ее еще нет
                local rec = self._panelButtons[key]
                if not rec then
                    rec = { 
                        texture = f:CreateTexture(nil, "OVERLAY", nil, 10),
                        side = side, 
                        slot = data.slot 
                    }
                    self._panelButtons[key] = rec
                end

                rec.settings   = data
                rec.panelIndex = data.panelIndex
                rec.slot       = data.slot

                -- 1. Видимость
                local show = self:ShouldShowButton(data)
                if show then
                    -- 2. Позиция
                    local posOk = self:SetButtonPosition(rec, cellIndex)
                    if posOk then
                        -- 3. Внешний вид
                        self:ApplyButtonAppearance(rec, data)
                        -- 4. Размер
                        self:ApplyButtonSize(rec)
                        
                        rec.texture:Show()
                    else
                        rec.texture:Hide()
                    end
                else
                    rec.texture:Hide()
                end
            end
        end
    end

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Mouse
-- Одна логика: регистрация обработчика мыши для кнопок панелей.
--
--   event "up"   → OnMouseUp()  (ReleaseTexture)
--   event "down" → HitPanelButton → LeftButton:   OnPanelButtonClick
--                                    MiddleButton: ToggleTooltipButton
--
-- НЕ рисует, НЕ создаёт, НЕ хит-тестит (это делают Draw / Hit / Click).
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local field = NSQC4.Field
    if not field then return end

    field:AddMouseHandler(function(x, y, button, event)
        -- Отпускание мыши — снять анимацию нажатия.
        -- Возвращаем false, чтобы другие обработчики (свитчеры) тоже
        -- могли снять свою анимацию.
        if event == "up" then
            field:OnMouseUp()
            return false
        end

        if event ~= "down" then return false end

        local rec = field.HitPanelButton and field:HitPanelButton(x, y)
        if rec then
            if button == "MiddleButton" then
                field:ToggleTooltipButton(rec)
            elseif button == "LeftButton" then
                field:OnPanelButtonClick(rec, button)
            end

            return true  -- событие поглощено
        end

        return false
    end)
end, 468)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Panel
-- Одна логика: текущий panelIndex стороны (из NSDB) + смена + переключение.
--
-- Хранилище: NSDB scope "panel", ключ "side" → число (1..N).
-- НЕ рисует, НЕ регистрирует, НЕ создаёт текстуры.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local NSDB        = NSQC4.NSDB
    if not QuestsField or not NSDB then return end

    local SCOPE = "panel"
    local SIDES = { "TOP", "BOTTOM", "LEFT", "RIGHT" }

    function QuestsField:GetPanelIndex(side)
        local n = NSDB:Get(SCOPE, side)
        if n == nil then
            NSDB:Set(SCOPE, side, 1)
            return 1
        end
        return n
    end

    function QuestsField:SetPanelIndex(side, n)
        n = tonumber(n) or 1
        NSDB:Set(SCOPE, side, n)
        if self.RefreshAllButtons then self:RefreshAllButtons() end
        if self.RefreshSwitchers   then self:RefreshSwitchers()   end
    end

    function QuestsField:GetPanelCount(side)
        local maxN = 0
        local buttons = NSDB:Table("buttons")
        if buttons then
            for _, data in pairs(buttons) do
                if data.side == side and data.panelIndex and data.panelIndex > maxN then
                    maxN = data.panelIndex
                end
            end
        end
        if maxN < 1 then maxN = 1 end
        return maxN
    end

    function QuestsField:SwitchPanel(side, dir)
        local cur  = self:GetPanelIndex(side)
        local maxN = self:GetPanelCount(side)
        local n    = cur + (tonumber(dir) or 1)
        if n > maxN then n = 1    end
        if n < 1    then n = maxN end
        self:SetPanelIndex(side, n)
    end

    function QuestsField:InitPanels()
        for _, side in ipairs(SIDES) do
            if NSDB:Get(SCOPE, side) == nil then
                NSDB:Set(SCOPE, side, 1)
            end
        end
    end

end, 450)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Press
-- Одна логика: анимация нажатия текстур-кнопок (панели, свитчеры).
--   PressTexture(tex)   — уменьшить на PIXELS*2, запомнить исходный размер
--   ReleaseTexture()    — вернуть исходный размер
--   IsPressed(tex)      — true, если эта текстура сейчас нажата
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    local PIXELS = 3

    function QuestsField:PressTexture(tex)
        if not tex then return end
        local w, h = tex:GetWidth(), tex:GetHeight()
        if not w or not h then return end
        tex:SetSize(w - PIXELS * 2, h - PIXELS * 2)
        self._pressedTex = { tex = tex, w = w, h = h }
    end

    function QuestsField:ReleaseTexture()
        local p = self._pressedTex
        if not p then return end
        self._pressedTex = nil
        if p.tex and p.w and p.h then
            p.tex:SetSize(p.w, p.h)
        end
    end

    function QuestsField:IsPressed(tex)
        local p = self._pressedTex
        return p ~= nil and p.tex == tex
    end

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Buttons / Register
-- Одна логика: API RegisterButton — валидация + запись в NSDB.
-- Не рисует, не раскладывает, не создаёт текстуры.
-- Хранилище: NSDB scope "buttons", ключ "side:panelIndex:slot".
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    local NSDB        = NSQC4.NSDB
    if not QuestsField or not NSDB then return end

    local SCOPE    = "buttons"
    local MIN_SLOT = 1
    local MAX_SLOT = 9
    local SIDES    = { LEFT = true, RIGHT = true, TOP = true, BOTTOM = true }

    function QuestsField:RegisterButton(side, panelIndex, slot, icon, onClick, opts)
        if not SIDES[side] then return nil end

        slot       = tonumber(slot) or 0
        panelIndex = tonumber(panelIndex) or 1
        opts       = opts or {}

        if slot < MIN_SLOT or slot > MAX_SLOT then return nil end

        local key = side .. ":" .. panelIndex .. ":" .. slot

        if not self._callbacks then self._callbacks = {} end
        self._callbacks[key] = onClick

        if NSDB:Has(SCOPE, key) then
            return NSDB:Get(SCOPE, key)
        end

        local data = {
            side       = side,
            panelIndex = panelIndex,
            slot       = slot,
            icon       = icon,
            tooltip    = opts.tooltip,
            opts       = opts,
        }
        NSDB:Set(SCOPE, key, data)
        return data
    end

end, 460)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Label / Refresh
-- Отрисовка лейблов. Положение — по rec.align (LEFT / CENTER / RIGHT).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local W = NSQC4.FieldWindow
    if not W then return end

    local LABEL_H  = 24
    local DESC_GAP = 4
    local TITLE_R, TITLE_G, TITLE_B = 1, 0.82, 0
    local DESC_R,  DESC_G,  DESC_B  = 0.7, 0.7, 0.7

    local function ApplyAlign(fs, win, y, align)
        fs:ClearAllPoints()
        if align == "CENTER" then
            fs:SetPoint("TOP", win, "TOP", 0, y)
            fs:SetJustifyH("CENTER")
        elseif align == "RIGHT" then
            fs:SetPoint("TOPRIGHT", win, "TOPRIGHT", -8, y)
            fs:SetJustifyH("RIGHT")
        else
            fs:SetPoint("TOPLEFT", win, "TOPLEFT", 8, y)
            fs:SetJustifyH("LEFT")
        end
    end

    function W:DrawLabel(windowId, rec, y, availableW)
        local win = self[windowId]
        if not win then return 0 end

        local align = rec.align or "LEFT"

        if not rec.frame then
            local title = win:CreateFontString(nil, "OVERLAY", "GameFontNormal")
            title:SetText(rec.title)
            title:SetTextColor(TITLE_R, TITLE_G, TITLE_B)
            rec.frame = title

            if rec.desc then
                local desc = win:CreateFontString(nil, "OVERLAY", "GameFontHighlightSmall")
                desc:SetText(rec.desc)
                desc:SetTextColor(DESC_R, DESC_G, DESC_B)
                desc:SetNonSpaceWrap(true)
                rec.descFrame = desc
            end
        end

        ApplyAlign(rec.frame, win, y, align)
        rec.frame:Show()

        local h = LABEL_H
        if rec.descFrame then
            rec.descFrame:SetWidth(availableW)
            ApplyAlign(rec.descFrame, win, y - LABEL_H, align)
            rec.descFrame:Show()
            h = h + (rec.descFrame:GetStringHeight() or 0) + DESC_GAP
        end
        return h
    end

end, 520)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Radio / Create
-- Одна логика: создание CheckButton для радио-группы.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local W = NSQC4.FieldWindow
    if not W then return end

    local function SafeName(windowId, groupId, value)
        local raw = string.format("NSQC4FieldRadio_%s_%s_%s",
            tostring(windowId), tostring(groupId), tostring(value))
        return (raw:gsub("[^%w_]", "_"))
    end

    function W:EnsureRadioGroup(windowId, groupId)
        local win = self[windowId]
        if not win then return nil end
        local grp = self._radioGroups and self._radioGroups[windowId]
                        and self._radioGroups[windowId][groupId]
        if not grp then return nil end
        if grp._buttons then return grp._buttons end

        grp._buttons = {}
        for _, opt in ipairs(grp.options or {}) do
            local name = SafeName(windowId, groupId, opt.value)
            local rb = _G[name] or CreateFrame("CheckButton", name, win, "UIRadioButtonTemplate")
            rb:SetParent(win)
            rb:SetFrameLevel(win:GetFrameLevel() + 5)
            rb:EnableMouse(true)
            rb.value = opt.value
            rb:Hide()

            local fs = rb:GetFontString()
            if fs then fs:Hide() end
            local fsG = _G[name .. "Text"]
            if fsG then fsG:Hide() end

            if opt.tooltip then
                rb:SetScript("OnEnter", function(s)
                    GameTooltip:SetOwner(s, "ANCHOR_RIGHT")
                    GameTooltip:ClearLines()
                    GameTooltip:AddLine(opt.tooltip, 1, 1, 1, true)
                    GameTooltip:Show()
                end)
                rb:SetScript("OnLeave", function() GameTooltip:Hide() end)
            end

            rb:SetScript("OnClick", function(s)
                for _, o in pairs(grp._buttons) do o:SetChecked(o == s) end
                if grp.onSelect then grp.onSelect(s.value) end
            end)

            grp._buttons[opt.value] = rb
        end
        return grp._buttons
    end
end, 553)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Radio / Draw
-- Одна логика: раскладка кнопок радио-группы.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local W = NSQC4.FieldWindow
    if not W then return end

    local RADIO_H, RADIO_GAP, MARGIN, ICON_W, ICON_GAP = 24, 6, 8, 24, 12

    function W:DrawRadio(windowId, rec, y, availableW)
        local win = self[windowId]
        if not win then return 0 end
        local btns = self:EnsureRadioGroup(windowId, rec.groupId)
        if not btns then return 0 end
        local grp = self._radioGroups[windowId][rec.groupId]
        if not grp then return 0 end

        local align = grp.align or "LEFT"
        local winW = win:GetWidth()
        local n = 0
        for _, opt in ipairs(grp.options or {}) do
            if btns[opt.value] then n = n + 1 end
        end
        local totalW = ICON_W * n + (n > 1 and ICON_GAP * (n - 1) or 0)

        local x = (align == "CENTER") and (winW - totalW) / 2
               or (align == "RIGHT") and (winW - MARGIN - totalW)
               or MARGIN

        for _, opt in ipairs(grp.options or {}) do
            local rb = btns[opt.value]
            if rb then
                rb:ClearAllPoints()
                rb:SetPoint("TOPLEFT", win, "TOPLEFT", x, y)
                rb:Show()
                x = x + ICON_W + ICON_GAP
            end
        end
        return RADIO_H + RADIO_GAP
    end
end, 554)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Window / API / Buttons / Hit
-- Одна логика: хит-тест кнопок окна по геометрии + регистрация обработчика.
--
--   W:AttachButtonMouse(win, windowId)
--
-- event "up"   → UIButton:Release()
-- event "down" → хит-тест по кнопкам windowId → Press + onClick
--
-- Использует W:AddWindowMouseHandler (API/Mouse/Register.lua, 494).
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local W        = NSQC4.FieldWindow
    local UIButton = NSQC4.UIButton
    if not W or not UIButton then return end

    function W:AttachButtonMouse(win, windowId)
        if not win or win._buttonsMouseAttached then return end
        win._buttonsMouseAttached = true

        self:AddWindowMouseHandler(win, function(x, y, button, event)
            if event ~= "down" then
                if event == "up" then
                    UIButton:Release()
                end
                return false
            end

            if button ~= "LeftButton" then return false end

            local items = self._items and self._items[windowId]
            if not items then return false end

            local winLeft, winBottom = win:GetLeft(), win:GetBottom()
            if not winLeft or not winBottom then return false end
            local screenX = winLeft + x
            local screenY = winBottom + y

            for _, rec in pairs(items) do
                if rec.kind == "button" and rec.frame and rec.frame:IsShown() then
                    local l = rec.frame:GetLeft()
                    local r = rec.frame:GetRight()
                    local t = rec.frame:GetTop()
                    local b = rec.frame:GetBottom()
                    if l and r and t and b
                       and screenX >= l and screenX <= r
                       and screenY >= b and screenY <= t then
                        UIButton:Press(rec)
                        if rec.onClick then rec.onClick(button) end
                        return true
                    end
                end
            end

            return false
        end)
    end

end, 561)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Window / API / Create
-- Одна логика: создание окна (Register) + отложенные onShow-хуки.
--
-- После создания окна:
--   - навешивает мышь (W:SetupWindowMouse, 497)
--   - навешивает хит-тест кнопок (W:AttachButtonMouse, 584), если есть items
--   - вызывает отложенные onShow-хуки (W:RegisterOnShow)
--
-- Это заменяет манки-патчинг W:Register из Content.lua и W:Show/W:Toggle
-- из Radio.lua.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local W = NSQC4.FieldWindow
    local C = NSQC4.FieldWindowConst
    if not W or not C then return end

    local function ResolveSize(val, fieldDim, default)
        if val == "FIELD" then return fieldDim end
        if type(val) == "number" then return val end
        return default
    end

    function W:Register(desc)
        if not desc or not desc.id then return nil end
        if self[desc.id] then return self[desc.id] end

        local field = NSQC4.Field
        if not field or not field.frame then return nil end

        local f   = field.frame
        local key = desc.id

        local w = ResolveSize(desc.width,  f:GetWidth(),  C.DEFAULT_WIDTH)
        local h = ResolveSize(desc.height, f:GetHeight(), f:GetHeight())

        local win = CreateFrame("Frame", "NSQC4FieldWindow_"..key, f)
        win:SetFrameStrata(f:GetFrameStrata())
        win:SetFrameLevel(f:GetFrameLevel() + C.FRAME_LEVEL_ADD)
        win:SetBackdrop({ bgFile = "Interface\\Buttons\\WHITE8x8" })
        win:SetBackdropColor(0, 0, 0, 0.9)
        win:EnableMouse(true)

 FILE: Field/UI/Window/Layout.lua

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Window / Layout
-- Одна логика: управление расположением элементов окна по вертикали (Y).
--
-- Не создает фреймы, текстуры или текст. Только двигает.
-- Порядок — строго по индексу (table.sort по числовым ключам).
-- Кнопки окна рисует универсальный рендерер NSQC4.UIButton:Draw.
-- ============================================================================
NSQC4.RegisterModule("quests", function()
    local W = NSQC4.FieldWindow
    if not W then return end

    local START_Y   = -8
    local CONTENT_W = 16 -- отступы 8 слева + 8 справа

    function W:LayoutWindow(windowId)
        local win = self[windowId]
        if not win then return end

        local items = self._items and self._items[windowId]
        if not items then return end

        -- 1. Собираем и сортируем индексы (числовые — как числа)
        local indices = {}
        for i in pairs(items) do
            indices[#indices + 1] = i
        end
        table.sort(indices, function(a, b)
            if type(a) == "number" and type(b) == "number" then
                return a < b
            end
            return tostring(a) < tostring(b)
        end)

        local y          = START_Y
        local availableW = win:GetWidth() - CONTENT_W
        local UIButton   = NSQC4.UIButton

        -- 2. Проходим сверху вниз
        for _, i in ipairs(indices) do
            local rec = items[i]
            if rec then
                local h = 0

                if rec.kind == "label" and self.DrawLabel then
                    h = self:DrawLabel(windowId, rec, y, availableW)

                elseif rec.kind == "button" and UIButton and UIButton.Draw then
                    h = UIButton:Draw(win, rec, y, availableW)

                elseif rec.kind == "radio" and self.DrawRadio then
                    h = self:DrawRadio(windowId, rec, y, availableW)
                end

                y = y - (h or 0)
            end
        end
    end
end, 500)

-- ============================================================================
-- NSQC4 / Quests / NSDB / Debug
-- Одна логика: /ns_db tables | dump <scope> | clear <scope> | drop <scope>.
-- dump рекурсивный: показывает вложенные таблицы.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local NSDB = NSQC4.NSDB
    if not NSDB then return end

    local function DumpTree(t, prefix, depth)
        if depth > 6 then return end
        local keys = {}
        for k in pairs(t) do keys[#keys + 1] = k end
        table.sort(keys, function(a, b) return tostring(a) < tostring(b) end)
        for _, k in ipairs(keys) do
            local v = t[k]
            if type(v) == "table" then
                print(string.format("  %s%s = {", prefix, tostring(k)))
                DumpTree(v, prefix .. "  ", depth + 1)
                print(string.format("  %s}", prefix))
            else
                print(string.format("  %s%s = %s", prefix, tostring(k), tostring(v)))
            end
        end
    end

    SLASH_NSDB1 = "/ns_db"
    SlashCmdList["NSDB"] = function(msg)
        local cmd, arg = (msg or ""):match("^(%S*)%s*(.-)$")
        cmd = (cmd or ""):lower()
        arg = arg or ""

        if cmd == "" or cmd == "tables" then
            local root = NSDB:Root()
            local names = {}
            for n in pairs(root) do names[#names + 1] = n end
            table.sort(names)
            print(string.format("[NSDB] таблиц верхнего уровня: %d", #names))
            for _, n in ipairs(names) do
                print(string.format("  %s", n))
            end

        elseif cmd == "dump" then
            if arg == "" then
                print("[NSDB] /ns_db dump <scope>")
                return
            end
            local t = NSDB:Walk(arg)
            if not t then
                print("[NSDB] нет таблицы: " .. arg)
                return
            end
            print(string.format("[NSDB] %s = {", arg))
            DumpTree(t, "  ", 0)
            print("}")

        elseif cmd == "clear" then
            if arg == "" then
                print("[NSDB] /ns_db clear <scope>")
                return
            end
            print(string.format("[NSDB] clear %s → %s", arg,
                tostring(NSDB:Clear(arg))))

        elseif cmd == "drop" then
            if arg == "" then
                print("[NSDB] /ns_db drop <scope>")
                return
            end
            print(string.format("[NSDB] drop %s → %s", arg,
                tostring(NSDB:DropTable(arg))))

        else
            print("[NSDB] /ns_db tables | dump <scope> | clear <scope> | drop <scope>")
        end
    end

    print("|cff00ffff[NSDB]|r База данных загружена. /ns_db tables | dump <scope>")

end, 270)

-- Одна логика: DropTable(scope) — удалить таблицу по пути.
--
--   DropTable("field")       → nsDbc4.quests.field = nil
--   DropTable("field.pos")   → nsDbc4.quests.field.pos = nil

NSQC4.RegisterModule("quests", function()

    local NSDB = NSQC4.NSDB

        -- Разбить на родителя и последний сегмент.

            parent = self:Walk(parentPath)
        else
            parent = self:Root()
            lastName = scope
        end

        if not parent then return false end
        if parent[lastName] == nil then return false end
        parent[lastName] = nil
        return true
    end
