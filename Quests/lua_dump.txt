-- Класс поля квестов 10×10 + метаданные модуля "quests".

NSQC4.RegisterModule("quests", function()

    local COLS, ROWS   = 10, 10
    local CELL_GAP     = 0
    local GRID_PADDING = 40
    local DEFAULT_CELL = 32
    local MIN_CELL     = 10
    local MAX_CELL     = 64
    local SAVE_KEY     = "questsField"

    local QuestsField = {}
    QuestsField.__index = QuestsField

    NSQC4.QuestsFieldClass = QuestsField
    NSQC4.QuestsFieldConst = {
        COLS = COLS, ROWS = ROWS,
        CELL_GAP = CELL_GAP, GRID_PADDING = GRID_PADDING,
        DEFAULT_CELL = DEFAULT_CELL, MIN_CELL = MIN_CELL, MAX_CELL = MAX_CELL,
        SAVE_KEY = SAVE_KEY,
    }

    function QuestsField.new()
        local self = setmetatable({}, QuestsField)
        self.frame         = nil
        self.cells         = {}
        self.borderFrame   = nil
        self.resizeHandle  = nil
        self.closeBtn      = nil
        self.borderVisible = false
        self.cellSize      = nil
        self.saved         = nil
        self.contextMenus  = {} -- Инициализация таблицы контекстных меню
        return self
    end

    NSQC4.QuestsField = QuestsField.new()

    NSQC4.SetModuleMeta("quests", {
        label   = "Квесты",
        tooltip = "Поле квестов 10×10.",
        order   = 500,
    })

    SLASH_NSFIELD1 = "/ns_field"
    SlashCmdList["NSFIELD"] = function()
        local inst = NSQC4.QuestsField
        if not inst or not inst.Toggle then
            print("|cffff0000[NSQC4]|r Модуль «Квесты» выключен или не загружен.")
            return
        end
        inst:Toggle()
    end

    if NSQC4.SlashPanel_RegisterSectionButton then
        NSQC4.SlashPanel_RegisterSectionButton(
            "Квесты",
            "П",
            "/ns_field",

-- ============================================================================
-- NSQC4 / Quests / Field / Storage
-- Хранилище nsDbc4.quests.field + API флага и шаблона заполнения.
-- Модуль: quests.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    -- ========================================================================
    -- Гарантирует, что таблица существует
    -- ========================================================================
    function QuestsField:EnsureStorage()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.field = nsDbc4.quests.field or {}
        return nsDbc4.quests.field
    end

    -- ========================================================================
    -- Загрузка из nsDbc4 → self
    -- ========================================================================
    function QuestsField:Load()
        local s = self:EnsureStorage()
        self.saved = {
            x             = s.x,
            y             = s.y,
            cellSize      = s.cellSize,
            borderVisible = s.borderVisible,
        }
    end

    -- ========================================================================
    -- Сохранение self → nsDbc4
    -- ========================================================================
    function QuestsField:Save()
        local s = self:EnsureStorage()

        if self.frame then
            s.x = self.frame:GetLeft()
            s.y = self.frame:GetBottom()
        end

        if self.cellSize then
            s.cellSize = self.cellSize
        end

        s.borderVisible = self.borderVisible and true or false
    end

    -- ========================================================================
    -- API: флаг «поле заполнено» — един на всё поле
    -- ========================================================================
    function QuestsField:IsFilled()
        local s = self:EnsureStorage()
        return s.filled and true or false
    end

    function QuestsField:SetFilled(v)
        local s = self:EnsureStorage()
        s.filled = v and true or false
    end

    -- ========================================================================
    -- API: шаблон заполнения поля
    -- ========================================================================
    function QuestsField:GetFillTemplate()
        local s = self:EnsureStorage()
        return s.template
    end

    function QuestsField:SetFillTemplate(tpl)
        local s = self:EnsureStorage()
        s.template = tpl
        return s.template
    end

end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Anchors
-- Управление иконками (маркерами) в клетке.
-- Только визуал: SetAnchorTexture, SetAnchorExpanded,
-- ClearAnchorTexture, ClearAnchorTextures.
--
-- Клики НЕ здесь — регистрация кликов только через Mouse.lua
-- (SetCellOnMouseClick / ClearCellMouse). Реестр подмен кликов удалён:
-- TerrainUIApi сразу проверяет существо и вешает нужный клик.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    local ANCHOR_POSITIONS = {
        "TOPLEFT", "TOPCENTER", "TOPRIGHT",
        "MIDLEFT", "CENTER",    "MIDRIGHT",
        "BOTLEFT", "BOTCENTER", "BOTRIGHT",
    }

    local ANCHOR_SUBLEVEL = 7

    local function GetOrCreateAnchor(btn, pos)
        if not btn.anchors then btn.anchors = {} end
        local anchor = btn.anchors[pos]
        if not anchor then
            anchor = btn:CreateTexture(nil, "ARTWORK", nil, ANCHOR_SUBLEVEL)
            anchor:SetDrawLayer("ARTWORK", ANCHOR_SUBLEVEL)
            btn.anchors[pos] = anchor
        else
            anchor:SetDrawLayer("ARTWORK", ANCHOR_SUBLEVEL)
        end
        return anchor
    end

    local function ApplyAnchorPosition(texture, btn, pos, expanded)
        texture:ClearAllPoints()
        if expanded then
            texture:SetAllPoints(btn)
        else
            local size = btn:GetWidth() / 3
            texture:SetSize(size, size)
            if pos == "TOPLEFT" then
                texture:SetPoint("TOPLEFT", btn, "TOPLEFT", 0, 0)
            elseif pos == "TOPCENTER" then
                texture:SetPoint("TOP", btn, "TOP", 0, 0)
            elseif pos == "TOPRIGHT" then
                texture:SetPoint("TOPRIGHT", btn, "TOPRIGHT", 0, 0)
            elseif pos == "MIDLEFT" then
                texture:SetPoint("LEFT", btn, "LEFT", 0, 0)
            elseif pos == "CENTER" then
                texture:SetPoint("CENTER", btn, "CENTER", 0, 0)
            elseif pos == "MIDRIGHT" then
                texture:SetPoint("RIGHT", btn, "RIGHT", 0, 0)
            elseif pos == "BOTLEFT" then
                texture:SetPoint("BOTTOMLEFT", btn, "BOTTOMLEFT", 0, 0)
            elseif pos == "BOTCENTER" then
                texture:SetPoint("BOTTOM", btn, "BOTTOM", 0, 0)
            elseif pos == "BOTRIGHT" then
                texture:SetPoint("BOTTOMRIGHT", btn, "BOTTOMRIGHT", 0, 0)
            end
        end
    end

    -- ========================================================================
    -- Визуал якоря: текстура 1/3 клетки (expanded = false)
    -- ========================================================================
    function QuestsField:SetAnchorTexture(index, pos, texture)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end
        local anchor = GetOrCreateAnchor(btn, pos)
        if texture then
            ApplyAnchorPosition(anchor, btn, pos, false)
            anchor:SetTexture(texture)
            anchor:SetTexCoord(0, 1, 0, 1)
            anchor:SetVertexColor(1, 1, 1, 1)
            anchor:SetAlpha(1)
            anchor:Show()
        else
            anchor:SetTexture(nil)
            anchor:Hide()
        end
    end

    -- ========================================================================
    -- Визуал якоря с явным expanded (true → на всю клетку)
    -- ========================================================================
    function QuestsField:SetAnchorExpanded(index, pos, texture, expanded)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end
        local anchor = GetOrCreateAnchor(btn, pos)
        if texture then
            ApplyAnchorPosition(anchor, btn, pos, expanded)
            anchor:SetTexture(texture)
            anchor:SetTexCoord(0, 1, 0, 1)
            anchor:SetVertexColor(1, 1, 1, 1)
            anchor:SetAlpha(1)
            anchor:Show()
        else
            anchor:SetTexture(nil)
            anchor:Hide()
        end
    end

    -- ========================================================================
    -- Очистка одного якоря
    -- ========================================================================
    function QuestsField:ClearAnchorTexture(index, pos)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn or not btn.anchors or not btn.anchors[pos] then return end
        local anchor = btn.anchors[pos]
        anchor:SetTexture(nil)
        anchor:Hide()
    end

    -- ========================================================================
    -- Очистка всех якорей клетки
    -- ========================================================================
    function QuestsField:ClearAnchorTextures(index)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn or not btn.anchors then return end
        for _, pos in ipairs(ANCHOR_POSITIONS) do
            if btn.anchors[pos] then
                btn.anchors[pos]:SetTexture(nil)
                btn.anchors[pos]:Hide()
            end
        end
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Clear
-- Полная очистка визуала клетки и всего поля.
-- Клики НЕ трогает — их вешают/снимают модули через Mouse.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS = CONST.COLS, CONST.ROWS

    function QuestsField:ClearCell(index)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        -- 1. Основная текстура (ландшафт)
        if btn.texture then
            btn.texture:SetTexture(nil)
            btn.texture:SetVertexColor(1, 1, 1, 1)
            btn.texture:Hide()
        end

        -- 2. Ховер-текстура
        if btn.hoverTexture then
            btn.hoverTexture:SetTexture(nil)
            btn.hoverTexture:Hide()
        end

        -- 3. Текст (основной и ховер)
        if btn.text_main then
            btn.text_main:SetText("")
            btn.text_main:Hide()
        end
        if btn.text_hover then
            btn.text_hover:SetText("")
            btn.text_hover:Hide()
        end

        -- 4. Иконки-якоря (визуал, не _clicks)
        if btn.anchors then
            for _, anchor in pairs(btn.anchors) do
                anchor:SetTexture(nil)
                anchor:Hide()
            end
        end

        -- 5. Рамка кнопки — стандартный вид
        btn:SetBackdropBorderColor(0.25, 0.25, 0.25, 1)

        -- 6. Ховер-обработчики
        btn._onEnter = nil
        btn._onLeave = nil
    end

    function QuestsField:ClearAll()
        for i = 1, COLS * ROWS do
            self:ClearCell(i)
        end
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / ContextMenu
-- Отрисовка контекстного меню под мышкой.
--
-- Фрейм меню создаётся ОДИН раз (лениво) и переиспользуется.
-- Список пунктов хранится в self._contextMenuItems и читается InitMenu
-- при каждом показе — поэтому повторные вызовы не плодят фреймы.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    -- ========================================================================
    -- Показ меню
    -- menuItems = { { text = "Текст", func = function() ... end }, ... }
    -- btn = кнопка (клетка), к которой привязано меню
    -- ========================================================================
    function QuestsField:ShowContextMenu(menuItems, btn)
        if not menuItems or #menuItems == 0 then return end

        -- 1. Фрейм меню — один раз на всё поле
        local menuFrame = self._contextMenuFrame
        if not menuFrame then
            menuFrame = CreateFrame("Frame", "NSQC4ContextMenu", UIParent, "UIDropDownMenuTemplate")
            self._contextMenuFrame = menuFrame
        end

        -- 2. Актуальный список пунктов для текущего показа
        self._contextMenuItems = menuItems

        -- 3. InitMenu читает self._contextMenuItems (не замыкание на аргумент)
        local function InitMenu(_, level)
            local items = self._contextMenuItems
            if not items then return end
            for _, item in ipairs(items) do
                local info = UIDropDownMenu_CreateInfo()
                info.text = item.text
                info.notCheckable = true
                info.func = function()
                    UIDropDownMenu_Initialize(menuFrame, nil)
                    CloseDropDownMenus()
                    if item.func then item.func() end
                end
                UIDropDownMenu_AddButton(info, level)
            end
        end

        UIDropDownMenu_Initialize(menuFrame, InitMenu)
        ToggleDropDownMenu(1, nil, menuFrame, btn, 0, 0)
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Hover
-- Управление поведением при наведении мыши.
--
-- OnEnter / OnLeave ставят btn._hovered = true/false.
-- Это использует AI/Work.lua, чтобы играть звук работы только когда
-- мышь на существе. Тултип тоже идёт через _onEnter/_onLeave.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    local function ApplyHover(btn)
        if btn._hoverTexture and btn.hoverTexture then
            btn.hoverTexture:Show()
        end
        if btn._hoverHighlight then
            local r, g, b, a = unpack(btn._hoverHighlight)
            btn:SetBackdropBorderColor(r, g, b, a)
        end
    end

    local function ResetHover(btn)
        if btn.hoverTexture then
            btn.hoverTexture:Hide()
        end
        if btn._hoverHighlight then
            btn:SetBackdropBorderColor(0.25, 0.25, 0.25, 1)
        end
    end

    function QuestsField:InitCellHover(index)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        if not btn._hoverInitialized then
            btn:SetScript("OnEnter", function(self2)
                self2._hovered = true
                ApplyHover(self2)
                if self2._onEnter then self2._onEnter(self2, index) end
            end)
            btn:SetScript("OnLeave", function(self2)
                self2._hovered = false
                ResetHover(self2)
                if self2._onLeave then self2._onLeave(self2, index) end
            end)
            btn._hoverInitialized = true
        end
    end

    function QuestsField:SetCellOnEnter(index, fn)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end
        btn._onEnter = fn
        self:InitCellHover(index)
    end

    function QuestsField:SetCellOnLeave(index, fn)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end
        btn._onLeave = fn
        self:InitCellHover(index)
    end

    function QuestsField:SetCellHoverTexture(index, texture, alpha)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        if not btn.hoverTexture then
            btn.hoverTexture = btn:CreateTexture(nil, "BORDER", nil, 2)
            btn.hoverTexture:SetDrawLayer("BORDER", 2)
            btn.hoverTexture:SetPoint("TOPLEFT", btn, "TOPLEFT", 1, -1)
            btn.hoverTexture:SetPoint("BOTTOMRIGHT", btn, "BOTTOMRIGHT", -1, 1)
            btn.hoverTexture:Hide()
        else
            btn.hoverTexture:SetDrawLayer("BORDER", 2)
        end

        if texture then
            btn._hoverTexture = texture
            btn.hoverTexture:SetTexture(texture)
            btn.hoverTexture:SetTexCoord(0, 1, 0, 1)
            btn.hoverTexture:SetAlpha(alpha or 1.0)
        else
            btn._hoverTexture = nil
            btn.hoverTexture:SetTexture(nil)
            btn.hoverTexture:Hide()
        end

        self:InitCellHover(index)
    end

    function QuestsField:SetCellHoverHighlight(index, r, g, b, a)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        btn._hoverHighlight = { r, g, b, a or 1 }
        self:InitCellHover(index)
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Mouse
-- Абстрактный регистратор кликов мыши на клетке.
-- Ничего не знает ни о ландшафте, ни о существах, ни о хайлайте.
-- API: регистрация действия на конкретную кнопку мыши.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    -- ========================================================================
    -- Регистрация клика на конкретную кнопку
    --   index  — номер клетки 1..100
    --   which  — "LeftButton" | "RightButton" | "MiddleButton" | ...
    --   fn     — function(btn, button, index) или nil (снять)
    -- ========================================================================
    function QuestsField:SetCellOnMouseClick(index, which, fn)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        if not btn._mouseHandlers then
            btn._mouseHandlers = {}
        end

        if fn then
            btn._mouseHandlers[which] = fn
        else
            btn._mouseHandlers[which] = nil
        end

        btn:RegisterForClicks("AnyUp")
        btn:SetScript("OnClick", function(self2, button)
            local handler = self2._mouseHandlers and self2._mouseHandlers[button]
            if handler then
                handler(self2, button, index)
            end
        end)
    end

    -- ========================================================================
    -- Снятие клика
    --   index — номер клетки
    --   which — конкретная кнопка (nil = снять все)
    -- ========================================================================
    function QuestsField:ClearCellMouse(index, which)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        if not btn._mouseHandlers then return end

        if which then
            btn._mouseHandlers[which] = nil
        else
            btn._mouseHandlers = {}
        end

        -- Если обработчиков не осталось — снимаем OnClick
        local hasAny = false
        for _ in pairs(btn._mouseHandlers) do hasAny = true; break end
        if not hasAny then
            btn:SetScript("OnClick", nil)
            btn:RegisterForClicks()
        end
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Text
-- Управление текстом клетки (основной и при наведении).
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    function QuestsField:SetCellText(index, text_main, text_hover, r, g, b, size)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end

        -- Ленивое создание основного текста (слой OVERLAY, subLevel 1)
        if not btn.text_main then
            btn.text_main = btn:CreateFontString(nil, "OVERLAY")
            btn.text_main:SetDrawLayer("OVERLAY", 1) -- жёсткая фиксация слоя
            btn.text_main:SetPoint("CENTER", 0, 0)
            btn.text_main:SetFont("Fonts\\FRIZQT__.TTF", 10, "OUTLINE")
        else
            btn.text_main:SetDrawLayer("OVERLAY", 1)
        end

        -- Ленивое создание текста при наведении (слой OVERLAY, subLevel 2 — над основным)
        if not btn.text_hover then
            btn.text_hover = btn:CreateFontString(nil, "OVERLAY")
            btn.text_hover:SetDrawLayer("OVERLAY", 2) -- жёсткая фиксация слоя
            btn.text_hover:SetPoint("CENTER", 0, 0)
            btn.text_hover:SetFont("Fonts\\FRIZQT__.TTF", 10, "OUTLINE")
        else
            btn.text_hover:SetDrawLayer("OVERLAY", 2)
        end

        -- Настройка основного текста
        if text_main and text_main ~= "" then
            btn.text_main:SetText(text_main)
            if r and g and b then btn.text_main:SetTextColor(r, g, b, 1) end
            if size then
                local path = btn.text_main:GetFont()
                if path then btn.text_main:SetFont(path, size, "OUTLINE") end
            end
            btn.text_main:Show()
        else
            btn.text_main:SetText("")
            btn.text_main:Hide()
        end

        -- Настройка текста при наведении
        if text_hover and text_hover ~= "" then
            btn.text_hover:SetText(text_hover)
            if r and g and b then btn.text_hover:SetTextColor(r, g, b, 1) end
            if size then
                local path = btn.text_hover:GetFont()
                if path then btn.text_hover:SetFont(path, size, "OUTLINE") end
            end
        else
            btn.text_hover:SetText("")
        end

        -- Изначально текст при наведении скрыт
        btn.text_hover:Hide()
    end

    function QuestsField:ClearCellText(index)
        self:EnsureFrame()
        local btn = self.cells[index]
        if not btn then return end
        if btn.text_main then btn.text_main:SetText(""); btn.text_main:Hide() end
        if btn.text_hover then btn.text_hover:SetText(""); btn.text_hover:Hide() end
    end
end)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / API / Tooltip
-- API поля: регистрация/снятие тултипа на клетке.
--
-- Вызывающий код знает клетку и код тултипа:
--   QuestsField:SetCellTooltip(index, function(btn)
--       GameTooltip:AddLine("Имя", 1, 1, 0)
--       ...
--   end)
--
-- API сам:
--   - делает GameTooltip:SetOwner(btn, "ANCHOR_RIGHT")
--   - зовёт tooltipFn(btn) — код тултипа только AddLine'ит
--   - делает GameTooltip:Show() на OnEnter
--   - делает GameTooltip:Hide() на OnLeave
--
-- Снятие:
--   QuestsField:ClearCellTooltip(index)
--
-- Хендлеры OnEnter/OnLeave вешаются через Hover.lua
-- (SetCellOnEnter / SetCellOnLeave) — не дублируем скрипты.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then return end

    -- ========================================================================
    -- Повесить тултип на клетку.
    --   index     — номер клетки 1..100
    --   tooltipFn — function(btn), рисует AddLine'ы; или nil (снять)
    -- ========================================================================
    function QuestsField:SetCellTooltip(index, tooltipFn)
        self:EnsureFrame()
        local btn = self.cells and self.cells[index]
        if not btn then return false end

        if tooltipFn then
            self:SetCellOnEnter(index, function(b)
                GameTooltip:SetOwner(b, "ANCHOR_RIGHT")
                tooltipFn(b)
                GameTooltip:Show()
            end)
            self:SetCellOnLeave(index, function()
                GameTooltip:Hide()
            end)
        else
            self:SetCellOnEnter(index, nil)
            self:SetCellOnLeave(index, nil)
        end

        return true
    end

    -- ========================================================================
    -- Снять тултип с клетки.
    -- ========================================================================
    function QuestsField:ClearCellTooltip(index)
        self:EnsureFrame()
        local btn = self.cells and self.cells[index]
        if not btn then return false end

        self:SetCellOnEnter(index, nil)
        self:SetCellOnLeave(index, nil)

        -- На случай, если тултип был открыт в момент снятия
        GameTooltip:Hide()

        return true
    end

    -- ========================================================================
    -- Есть ли тултип на клетке (для диагностики / отладки).
    -- ========================================================================
    function QuestsField:HasCellTooltip(index)
        local btn = self.cells and self.cells[index]
        if not btn then return false end
        return btn._onEnter ~= nil
    end
end)

    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst

    local COLS, ROWS   = CONST.COLS, CONST.ROWS
    local CELL_GAP     = CONST.CELL_GAP
    local GRID_PADDING = CONST.GRID_PADDING

    local function IndexToColRow(index)
        local i = index - 1
        return i % COLS, math.floor(i / COLS)
    end

    function QuestsField:ReflowCells()
        local f = self.frame
        if not f then return end
        local fw, fh = f:GetWidth(), f:GetHeight()

        for index, btn in pairs(self.cells) do
            local col, row = IndexToColRow(index)
            btn:ClearAllPoints()
            btn:SetSize(cellW, cellH)
            btn:SetPoint("BOTTOMLEFT", f, "BOTTOMLEFT",
                GRID_PADDING + col * (cellW + CELL_GAP),
                GRID_PADDING + row * (cellH + CELL_GAP))
        end
    end

    function QuestsField:CreateCell(index)

        local btn = CreateFrame("Button", "NSQC4FieldCell" .. index, f)

        btn:SetFrameLevel(f:GetFrameLevel() + 10)

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Frame
-- Создание главного фрейма поля + клетки + рамка + ресайз + крестик.
-- Клики по клеткам НЕ вешает — это делают модули (TerrainActions → Mouse).
-- Здесь только: drag фрейма, ресайз, крестик, раскладка.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS     = CONST.COLS, CONST.ROWS
    local CELL_GAP       = CONST.CELL_GAP
    local GRID_PADDING   = CONST.GRID_PADDING
    local DEFAULT_CELL   = CONST.DEFAULT_CELL
    local MIN_CELL       = CONST.MIN_CELL
    local MAX_CELL       = CONST.MAX_CELL

    local function FrameSizeForCell(cs)
        return cs * COLS + (COLS - 1) * CELL_GAP + GRID_PADDING * 2,
               cs * ROWS + (ROWS - 1) * CELL_GAP + GRID_PADDING * 2
    end
    QuestsField._FrameSizeForCell = FrameSizeForCell

    -- ========================================================================
    -- EnsureFrame — создать фрейм и всё вокруг (идемпотентно)
    -- ========================================================================
    function QuestsField:EnsureFrame()
        if self.frame then return self.frame end

        if not self.saved then self:Load() end
        local saved = self.saved or {}

        local cs = math.floor((saved.cellSize or DEFAULT_CELL) + 0.5)
        if cs < MIN_CELL then cs = MIN_CELL end
        if cs > MAX_CELL then cs = MAX_CELL end
        self.cellSize = cs

        local w0, h0 = FrameSizeForCell(cs)

        local f = CreateFrame("Frame", "NSQC4FieldFrame", UIParent)
        self.frame = f
        f:SetFrameStrata("MEDIUM")
        f:SetMovable(true)
        f:SetClampedToScreen(true)
        f:SetSize(w0, h0)
        f:SetBackdrop({ bgFile = "Interface\\Buttons\\WHITE8x8" })
        f:SetBackdropColor(0, 0, 0, 0)
        f:EnableMouse(false)

        if saved.x and saved.y then
            f:SetPoint("BOTTOMLEFT", UIParent, "BOTTOMLEFT", saved.x, saved.y)
        else
            f:SetPoint("CENTER")
        end

        -- Клетки
        for index = 1, COLS * ROWS do
            self.cells[index] = self:CreateCell(index)
        end
        self:ReflowCells()

        -- UI-механика фрейма (не клики клеток)
        self:CreateBorder()
        self:CreateResizeHandle()
        self:CreateCloseButton()

        if saved.borderVisible then
            self.borderVisible = false
            self:SetBorderVisible(true)
        end

        f:HookScript("OnSizeChanged", function()
            self:ReflowCells()
            self:ApplyBorderAnchors()
        end)

        f:Hide()
        return f
    end

    -- ========================================================================
    -- Show / Hide / Toggle
    -- ========================================================================
    function QuestsField:Show()
        self:EnsureFrame()
        self.frame:Show()

        if NSQC4.QuestsDfTerrain then NSQC4.QuestsDfTerrain:Render() end
        if NSQC4.QuestsDfEntities then NSQC4.QuestsDfEntities:Render() end
    end

    function QuestsField:Hide()
        if self.frame then self.frame:Hide() end
    end

    function QuestsField:Toggle()
        self:EnsureFrame()
        if self.frame:IsShown() then self.frame:Hide() else self:Show() end
    end

    -- ========================================================================
    -- Крестик
    -- ========================================================================
    function QuestsField:CreateCloseButton()
        local f = self.frame
        local btn = CreateFrame("Button", "NSQC4FieldClose", f, "UIPanelCloseButton")
        btn:SetSize(32, 32)
        btn:SetPoint("TOPRIGHT", f, "TOPRIGHT", -2, -2)

        end)

        self.closeBtn = btn
        return btn
    end

-- ============================================================================
-- NSQC4 / Quests / Field / UI / Resize
-- Хэндл ресайза в правом нижнем углу.
-- ============================================================================

NSQC4.RegisterModule("quests", function()
    local QuestsField = NSQC4.QuestsFieldClass
    local CONST       = NSQC4.QuestsFieldConst
    if not QuestsField or not CONST then return end

    local COLS, ROWS     = CONST.COLS, CONST.ROWS
    local CELL_GAP       = CONST.CELL_GAP
    local GRID_PADDING   = CONST.GRID_PADDING
    local MIN_CELL       = CONST.MIN_CELL
    local MAX_CELL       = CONST.MAX_CELL

    function QuestsField:CreateResizeHandle()
        local f = self.frame
        local h = CreateFrame("Button", "NSQC4FieldResize", f)
        h:SetSize(16, 16)
        h:SetPoint("BOTTOMRIGHT", f, "BOTTOMRIGHT", 4, -4)
        h:SetFrameLevel(f:GetFrameLevel() + 30)
        h:SetNormalTexture("Interface\\ChatFrame\\UI-ChatIM-SizeGrabber-Up")
        h:SetPushedTexture("Interface\\ChatFrame\\UI-ChatIM-SizeGrabber-Down")
        h:SetHighlightTexture("Interface\\ChatFrame\\UI-ChatIM-SizeGrabber-Highlight")
        h:EnableMouse(true)

        local anchorL, anchorT
        local lastCell
        local inst = self

        local function apply()
            local scale = UIParent:GetEffectiveScale()
            local cx, cy = GetCursorPosition()
            cx, cy = cx / scale, cy / scale
            local rawW = cx - anchorL
            local rawH = anchorT - cy
            local csX = (rawW - (COLS - 1) * CELL_GAP - GRID_PADDING * 2) / COLS
            local csY = (rawH - (ROWS - 1) * CELL_GAP - GRID_PADDING * 2) / ROWS
            local cs = math.max(csX, csY)
            cs = math.floor(cs + 0.5)
            if cs < MIN_CELL then cs = MIN_CELL end
            if cs > MAX_CELL then cs = MAX_CELL end
            if cs == lastCell then return end
            lastCell = cs
            inst.cellSize = cs
            local w, hh = inst._FrameSizeForCell(cs)
            f:SetSize(w, hh)
            inst:ReflowCells()
            inst:ApplyBorderAnchors()
        end

        h:SetScript("OnMouseDown", function(self2, button)
            if button ~= "LeftButton" then return end
            anchorL = f:GetLeft()
            anchorT = f:GetTop()
            lastCell = nil
            self2:SetScript("OnUpdate", function()
                if not IsMouseButtonDown("LeftButton") then
                    self2:SetScript("OnUpdate", nil)
                    inst:Save()
                    return
                end
                apply()
            end)
            apply()
        end)
        h:SetScript("OnHide", function(self2) self2:SetScript("OnUpdate", nil) end)

        self.resizeHandle = h
        return h
    end
end)

-- ============================================================================

-- Реестр поведений по видам и характерам.
--
--   NSQC4.QuestsDfBehavior[kind][behavior] = { Tick = function(entity) ... end }
--
-- kind     — "goblin" | "dwarf" | ...
-- behavior — "aggressive" | "peaceful" | ...
--
-- Каждый вид регистрирует свои поведения в df/<Kind>/Behavior/*.lua.

-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Registry = {}
    NSQC4.QuestsDfBehavior = Registry

    -- ========================================================================
    -- Регистрация поведения.
    --   kind     — "goblin" | "dwarf"
    --   behavior — "aggressive" | "peaceful"
    --   handler  — { Tick = function(entity) ... end }
    -- ========================================================================
    function Registry:Register(kind, behavior, handler)
        if type(kind) ~= "string" or kind == "" then return false end
        if type(behavior) ~= "string" or behavior == "" then return false end
        if type(handler) ~= "table" then return false end

        if not self[kind] then self[kind] = {} end
        self[kind][behavior] = handler
        return true
    end

    function Registry:Get(kind, behavior)
        if type(kind) ~= "string" or type(behavior) ~= "string" then return nil end
        local byKind = self[kind]
        return byKind and byKind[behavior] or nil
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Complete
-- Финальный шаг работы: существо докопало клетку до нуля.
--
-- Вызывается из Tick.lua, когда Work:Tick вернул true.
-- Здесь только:
--   1. Достать профессию (yield, nextTerrain)
--   2. Записать результат (print; позже — в opts.inventory)
--   3. Сменить terrain клетки (earth по умолчанию)
--   4. Сбросить состояние в idle
--
-- Не считает total, не уменьшает hardness, не двигает, не ищет цель.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Complete = {}
    NSQC4.QuestsDfAIComplete = Complete

    local Grid    = NSQC4.QuestsDfGrid
    local Terrain = NSQC4.QuestsDfTerrain
    local State   = NSQC4.QuestsDfAIState
    local ProfApi = NSQC4.QuestsDfProfessionsApi

    -- ========================================================================
    -- Завершить работу на текущей цели существа.
    --   entity — существо с opts.ai (state="working", target, terrainId, profId)
    -- Возвращает true, если завершение обработано.
    -- ========================================================================
    function Complete:Finish(entity)
        if not entity or not entity.opts then return false end

        local ai = entity.opts.ai
        if not ai or not ai.target then return false end

        local target    = ai.target
        local terrainId = ai.terrainId
        local profId    = ai.profId

        -- 1. Профессия (может быть nil — работали без профы)
        local profDef = profId and ProfApi and ProfApi:Get(profId) or nil

        -- 2. Yield (что добыли) и nextTerrain (во что превратить клетку)
        local yield       = profDef and profDef.work and profDef.work.yields       or nil
        local nextTerrain = profDef and profDef.work and profDef.work.nextTerrain  or "earth"

        -- 3. Лог результата. Позже — в opts.inventory.
        print(string.format("|cff00ff00[NSQC4]|r %s закончил работать с %s (клетка %d)%s",
            entity.opts.name or "Существо",
            terrainId or "?",
            target,
            yield and (" → " .. yield) or ""))

        -- 4. Смена ландшафта клетки (Storage поставит hardness нового типа)
        if Terrain and Terrain.SetTerrain then
            Terrain:SetTerrain(target, nextTerrain)
        end

        -- 5. Сброс состояния в idle (существо свободно, ищет новую цель)
        if State and State.SetIdle then
            State:SetIdle(entity)
        end

        return true
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Default
-- Общий fallback-цикл поведения существа: idle → moving → working.
--
-- При idle→moving — звук по kind существа (gob.ogg / gom.ogg).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Default = {}
    NSQC4.QuestsDfAIDefault = Default

    Default.DEBUG = false

    local function dbg(...)
        if Default.DEBUG then print("|cff00ffff[DEF]|r", ...) end
    end

    local State    = NSQC4.QuestsDfAIState
    local Find     = NSQC4.QuestsDfAIFind
    local Path     = NSQC4.QuestsDfAIPath
    local Move     = NSQC4.QuestsDfAIMove
    local Work     = NSQC4.QuestsDfAIWork
    local Complete = NSQC4.QuestsDfAIComplete

    if not (State and Find and Path and Move and Work and Complete) then
        print("|cffff0000[NSQC4]|r AI/Default: не все зависимости загружены.")
        return
    end

    local function IsTargetStillSelected(entity)
        local ai = entity and entity.opts and entity.opts.ai
        if not ai or not ai.target or not ai.terrainId then return false end
        local Grid = NSQC4.QuestsDfGrid
        if not Grid then return true end
        return Grid:IsTerrainSelected(ai.target, ai.terrainId) and true or false
    end

    function Default:Tick(entity)
        if not entity or not entity.opts then return false end
        if entity.opts.sleeping then return false end

        local ai = entity.opts.ai
        if ai and ai.target and (ai.state == "moving" or ai.state == "working") then
            if not IsTargetStillSelected(entity) then
                dbg("drop target", ai.target, "terrain", ai.terrainId,
                    "state", ai.state, "entity", entity.id)

                local Animate = NSQC4.QuestsDfEntitiesAnimate
                if Animate and Animate:IsAnimating(entity.id) then
                    Animate:Stop(entity.id)
                end
                State:SetIdle(entity)
                return true
            end
        end

        if State:IsIdle(entity) then
            local pick = Find:Pick(entity)
            if not pick then return false end
            local path = Path:Find(entity.index, pick.index)
            if not path then return false end
            State:SetMoving(entity, pick.index, pick.terrainId, pick.profId, path)
            dbg("idle→moving", entity.id, "->", pick.index, pick.terrainId)

            local Sound = NSQC4.QuestsDfAISound
            if Sound then
                Sound:PlayKind(entity.kind or entity.opts.kind)
            end
            return true
        end

        if State:IsMoving(entity) then
            local r = Move:Advance(entity)
            if r == "arrived" then
                State:SetWorking(entity)
                dbg("moving→working", entity.id)
                return true
            elseif r == "moved" or r == "wait" then
                return true
            else
                State:SetIdle(entity)
                return false
            end
        end

        if State:IsWorking(entity) then
            local done = Work:Tick(entity)
            if done == true then
                Complete:Finish(entity)
                dbg("work done", entity.id)
                return true
            elseif done == false then
                return true
            else
                State:SetIdle(entity)
                return false
            end
        end

        State:SetIdle(entity)
        return false
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Find
-- Выбор цели для существа.
--
-- Логика:
--   1. Allowed существа по opts.kind (Goblin/Dwarf).
--   2. Выделенные клетки (nsDbc4.quests.df.selections.terrain_types).
--      Фильтр: allowed[terrainId] И не зарезервировано другим существом.
--   3. Лучшие (есть своя профессия) / просто.
--   4. Случайный из лучших, иначе из просто.
--   5. { index, terrainId, profId|nil } | nil.
--
-- known_skills — таблица [имя] = уровень (1..100). Для выбора цели
-- достаточно самого факта «есть def с этим terrain», уровень не нужен.
--
-- HasProfForSelected(entity) — проверка: есть ли у существа профессия
-- под хотя бы одну из выделенных клеток. Используется в Ticker, чтобы
-- дать приоритет существам с профессией.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Find = {}
    NSQC4.QuestsDfAIFind = Find

    Find.DEBUG = false

    local function dbg(...)
        if Find.DEBUG then print("|cff00ffff[FIND]|r", ...) end
    end

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r AI/Find: Grid не загружен.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    local function GetAllowed(kind)
        if kind == "goblin" then
            local g = NSQC4.QuestsDfGoblin
            return g and g.Allowed or nil
        elseif kind == "dwarf" then
            local d = NSQC4.QuestsDfDwarf
            return d and d.Allowed or nil
        end
        return nil
    end

    local function GetSelectedCells()
        local result = {}
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.selections = nsDbc4.quests.df.selections or {}
        local s = nsDbc4.quests.df.selections

        local byType = s.terrain_types or {}
        for terrainId, list in pairs(byType) do
            for index in pairs(list) do
                result[#result + 1] = { index = index, terrainId = terrainId }
            end
        end
        return result
    end

    -- ========================================================================
    -- Есть ли у существа def профессии под данный terrain?
    -- known_skills — hash [имя] = уровень, поэтому pairs.
    -- ========================================================================
    local function FindProfForTerrain(entity, terrainId)
        local ProfApi = NSQC4.QuestsDfProfessionsApi
        if not ProfApi then return nil end

        local opts = entity.opts
        if not opts or type(opts.known_skills) ~= "table" then return nil end

        for name in pairs(opts.known_skills) do
            local def = ProfApi:GetByName(name)
            if def and def.terrain == terrainId then
                return def
            end
        end
        return nil
    end

    -- ========================================================================
    -- Есть ли у существа профессия под хотя бы одну ВЫДЕЛЕННУЮ клетку?
    -- (учитываем и Allowed, иначе смысла нет)
    -- ========================================================================
    function Find:HasProfForSelected(entity)
        if not entity or not entity.opts then return false end

        local kind = entity.kind or entity.opts.kind
        local allowed = GetAllowed(kind)
        if not allowed then return false end

        local selected = GetSelectedCells()
        if #selected == 0 then return false end

        for _, cell in ipairs(selected) do
            if allowed[cell.terrainId] then
                if FindProfForTerrain(entity, cell.terrainId) then
                    return true
                end
            end
        end
        return false
    end

    function Find:Pick(entity)
        if not entity or not entity.opts then return nil end

        local kind = entity.kind or entity.opts.kind
        local allowed = GetAllowed(kind)
        if not allowed then
            dbg("no allowed for", tostring(kind))
            return nil
        end

        local selected = GetSelectedCells()
        if #selected == 0 then
            dbg("no selected cells")
            return nil
        end

        -- 1. Фильтр: allowed + не зарезервировано другим
        local candidates = {}
        for _, cell in ipairs(selected) do
            if allowed[cell.terrainId] then
                if not Grid:IsReserved(cell.index, entity.id) then
                    candidates[#candidates + 1] = cell
                else
                    dbg("skip reserved", cell.index, "terrain", cell.terrainId,
                        "by", tostring(Grid:GetReservedBy(cell.index)))
                end
            end
        end
        if #candidates == 0 then
            dbg("no candidates (all reserved or not allowed)")
            return nil
        end

        -- 2. Лучшие (с профой) / просто
        local best, rest = {}, {}
        for _, cell in ipairs(candidates) do
            local prof = FindProfForTerrain(entity, cell.terrainId)
            if prof then
                best[#best + 1] = { index = cell.index, terrainId = cell.terrainId, profId = prof.id }
            else
                rest[#rest + 1] = { index = cell.index, terrainId = cell.terrainId }
            end
        end

        local pool = (#best > 0) and best or rest
        if #pool == 0 then return nil end
        local pick = pool[math.random(#pool)]
        dbg("pick", pick.index, pick.terrainId, pick.profId or "-")
        return pick
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Move
-- Передвижение существа по маршруту. Скорость — по абсолютной выносливости.
--
-- stepTime = 100 − 99 × (vitality / 100)
--   vitality 0/1     → ~100–99 сек/клетку
--   vitality 100     → 1 сек/клетку
--
-- При каждом шаге — шанс прокачки живучести (intellect / 100).
--
-- Отладка: /run NSQC4.QuestsDfAIMove.DEBUG = true
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Move = {}
    NSQC4.QuestsDfAIMove = Move

    Move.DEBUG = false

    local function dbg(...)
        if Move.DEBUG then print("|cff00ffff[MOVE]|r", ...) end
    end

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r AI/Move: Grid не загружен.")
        return
    end

    -- ========================================================================
    -- Время на один шаг (секунды) по АБСОЛЮТНОЙ выносливости (0..100).
    -- max_vitality не используется.
    -- ========================================================================
    function Move:StepTime(entity)
        if not entity or not entity.opts then return 100 end
        local opts = entity.opts

        local vit = tonumber(opts.vitality) or 0
        local pct = vit / 100
        if pct < 0 then pct = 0 elseif pct > 1 then pct = 1 end

        local t = 100 - 99 * pct

        dbg(string.format("step %s vit=%s pct=%.3f stepTime=%.2fs",
            tostring(entity.id), tostring(vit), pct, t))
        return t
    end

    -- ========================================================================
    -- Один шаг по маршруту.
    -- ========================================================================
    function Move:Advance(entity)
        if not entity or not entity.opts then return nil end

        local ai = entity.opts.ai
        if not ai or not ai.path or not ai.pathIdx then return nil end

        local path = ai.path
        local idx  = ai.pathIdx

        if idx >= #path then
            return "arrived"
        end

        local now = GetTime()
        if ai.nextStepAt and now < ai.nextStepAt then
            dbg(string.format("wait %s now=%.2f nextStepAt=%.2f left=%.2fs",
                tostring(entity.id), now, ai.nextStepAt, ai.nextStepAt - now))
            return "wait"
        end

        local nextIndex = path[idx + 1]
        if not nextIndex then return nil end

        local isTarget = (idx + 1 == #path)
        if not isTarget and not Grid:IsPassable(nextIndex) then
            dbg(string.format("blocked %s next=%d", tostring(entity.id), nextIndex))
            ai.path = nil
            ai.pathIdx = nil
            return nil
        end

        local fromIndex = entity.index
        local stepTime  = self:StepTime(entity)

        -- 1. Данные
        Grid:SetEntity(fromIndex, nil)
        entity.index = nextIndex
        Grid:SetEntity(nextIndex, entity)

        -- 2. Анимация
        local Animate = NSQC4.QuestsDfEntitiesAnimate
        if Animate and fromIndex and nextIndex then
            Animate:Start(entity.id, {
                icon     = entity.opts.icon or entity.opts.texture,
                anchor   = "TOPLEFT",
                expanded = entity.opts.expanded,
                alpha    = entity.opts.alpha,
            }, fromIndex, nextIndex, stepTime)
        end

        -- 3. Путь и таймер
        ai.pathIdx    = idx + 1
        ai.nextStepAt = now + stepTime

        -- 4. Прокачка живучести при шаге
        local Stats = NSQC4.QuestsDfAIStats
        if Stats and Stats.TryUpVitality then
            Stats:TryUpVitality(entity)
        end

        dbg(string.format("move %s %d -> %d  stepTime=%.2fs",
            tostring(entity.id), fromIndex or 0, nextIndex, stepTime))

        if ai.pathIdx >= #path then
            return "arrived"
        end
        return "moved"
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Path
-- Поиск кратчайшего маршрута по сетке 10×10 (BFS, 4-связность).
--
-- Правила:
--   - старт — текущая клетка существа (entity.index)
--   - цель  — клетка target (index)
--   - проходимость: Grid:IsPassable(index) (нет entity на клетке)
--   - старт считается проходимым, даже если на нём стоит само существо
--   - цель считается проходимой, даже если на ней стоит само существо
--
-- Возвращает:
--   { {index, index, …, target}, ... } — массив индексов от старта до цели,
--   включая старт и цель. Если старт == цель → { index }.
--   nil — если пути нет.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Path = {}
    NSQC4.QuestsDfAIPath = Path

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r AI/Path: Grid не загружен.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    -- ========================================================================
    -- Восстановить путь по prev[].
    -- ========================================================================
    local function Reconstruct(prev, startIdx, targetIdx)
        local path = {}
        local cur = targetIdx
        while cur ~= nil do
            path[#path + 1] = cur
            if cur == startIdx then break end
            cur = prev[cur]
        end

        -- Развернуть (сейчас от цели к старту).
        local n = #path
        for i = 1, math.floor(n / 2) do
            path[i], path[n - i + 1] = path[n - i + 1], path[i]
        end
        return path
    end

    -- ========================================================================
    -- BFS: кратчайший путь от startIndex до targetIndex.
    --   selfIndex — индекс клетки, на которой стоит существо
    --               (её считаем проходимой даже если entity есть).
    -- ========================================================================
    function Path:Find(startIndex, targetIndex)
        startIndex  = tonumber(startIndex)  or 0
        targetIndex = tonumber(targetIndex) or 0

        if startIndex < 1 or startIndex > TOTAL then return nil end
        if targetIndex < 1 or targetIndex > TOTAL then return nil end

        if startIndex == targetIndex then
            return { startIndex }
        end

        local prev    = {}
        local visited = {}
        local queue   = { startIndex }
        local head    = 1

        visited[startIndex] = true
        prev[startIndex]    = nil

        while head <= #queue do
            local cur = queue[head]
            head = head + 1

            for _, neighbor in ipairs(Grid:Neighbors(cur)) do
                if not visited[neighbor] then
                    -- Проходимость: цель всегда проходима,
                    -- стартовая клетка уже в visited.
                    local passable = Grid:IsPassable(neighbor)
                    if neighbor == targetIndex then passable = true end

                    if passable then
                        visited[neighbor] = true
                        prev[neighbor]    = cur
                        queue[#queue + 1] = neighbor

                        if neighbor == targetIndex then
                            return Reconstruct(prev, startIndex, targetIndex)
                        end
                    end
                end
            end
        end

        return nil
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / State
-- Единственное место, где живёт и меняется entity.opts.ai.
--
-- Резерв клетки:
--   SetMoving(entity, target, …)  → Grid:Reserve(target, entity.id)
--   SetIdle(entity)               → Grid:Release(old.target, entity.id)
--   Clear(entity)                 → Grid:Release(old.target, entity.id)
--   SetWorking(entity)            → ничего (резерв уже стоит с SetMoving)
--
-- Формат opts.ai:
--   { state, target, terrainId, profId, path, pathIdx, nextStepAt, work }
--
-- API:
--   Get, Clear, SetIdle, SetMoving, SetWorking,
--   IsIdle, IsMoving, IsWorking
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local State = {}
    NSQC4.QuestsDfAIState = State

    State.DEBUG = false

    local function dbg(...)
        if State.DEBUG then print("|cff00ffff[STATE]|r", ...) end
    end

    local function GetGrid()
        return NSQC4.QuestsDfGrid
    end

    -- ========================================================================
    -- Получить opts.ai (создать «idle» при первом обращении).
    -- ========================================================================
    function State:Get(entity)
        if not entity or not entity.opts then return nil end
        local ai = entity.opts.ai
        if not ai then
            ai = { state = "idle" }
            entity.opts.ai = ai
        end
        if not ai.state then ai.state = "idle" end
        return ai
    end

    -- ========================================================================
    -- Сбросить всё (с освобождением резерва).
    -- ========================================================================
    function State:Clear(entity)
        if not entity or not entity.opts then return end
        local ai = entity.opts.ai
        if ai and ai.target then
            local Grid = GetGrid()
            if Grid and Grid.Release then
                Grid:Release(ai.target, entity.id)
            end
            dbg("Clear: release", ai.target, tostring(entity.id))
        end
        entity.opts.ai = nil
    end

    -- ========================================================================
    -- idle: цель/путь/работа сброшены, резерв снят.
    -- ========================================================================
    function State:SetIdle(entity)
        if not entity or not entity.opts then return false end
        local old = entity.opts.ai
        if old and old.target then
            local Grid = GetGrid()
            if Grid and Grid.Release then
                Grid:Release(old.target, entity.id)
            end
            dbg("SetIdle: release", old.target, tostring(entity.id))
        end
        entity.opts.ai = { state = "idle" }
        return true
    end

    -- ========================================================================
    -- moving: цель + маршрут + резерв клетки.
    -- ========================================================================
    function State:SetMoving(entity, target, terrainId, profId, path)
        if not entity or not entity.opts then return false end
        if not target or not path or #path == 0 then return false end

        -- Освободить старый резерв, если был
        local old = entity.opts.ai
        if old and old.target and old.target ~= target then
            local Grid = GetGrid()
            if Grid and Grid.Release then
                Grid:Release(old.target, entity.id)
            end
        end

        entity.opts.ai = {
            state      = "moving",
            target     = target,
            terrainId  = terrainId,
            profId     = profId,
            path       = path,
            pathIdx    = 1,
            nextStepAt = GetTime(),
            work       = nil,
        }

        local Grid = GetGrid()
        if Grid and Grid.Reserve then
            Grid:Reserve(target, entity.id)
        end
        dbg("SetMoving", tostring(entity.id), "->", target, tostring(terrainId), tostring(profId))
        return true
    end

    -- ========================================================================
    -- working: дошли до цели, путь не нужен. Резерв остаётся.
    -- ========================================================================
    function State:SetWorking(entity)
        if not entity or not entity.opts then return false end
        local ai = self:Get(entity)
        if not ai.target then return false end

        ai.state      = "working"
        ai.path       = nil
        ai.pathIdx    = nil
        ai.nextStepAt = nil
        ai.work       = ai.work or {}
        dbg("SetWorking", tostring(entity.id), "at", ai.target)
        return true
    end

    -- ========================================================================
    -- Проверки
    -- ========================================================================
    function State:IsIdle(entity)
        local ai = self:Get(entity)
        return ai and ai.state == "idle"
    end

    function State:IsMoving(entity)
        local ai = self:Get(entity)
        return ai and ai.state == "moving"
    end

    function State:IsWorking(entity)
        local ai = self:Get(entity)
        return ai and ai.state == "working"
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Stats
-- Прокачка характеристик существ + краткая отладка.
--
-- Шанс за бросок: int из 10000 (целочисленный рандом).
--   int = 4   → 4/10000  = 0.04%  (1 из 2500)
--   int = 50  → 50/10000 = 0.5%   (1 из 200)
--   int = 100 → 100/10000 = 1%    (1 из 100)
--
-- Отладка:
--   /ns_stats on      — включить сводку раз в 60 сек
--   /ns_stats off     — выключить
--   /ns_stats dump    — напечатать сейчас
--   /ns_stats reset   — сбросить счётчики
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Stats = {}
    NSQC4.QuestsDfAIStats = Stats

    Stats.DEBUG = false

    local dbg = {
        calls        = 0,   -- сколько раз вызвали TryUpVitality
        earlyExitMax = 0,   -- вышли: max_vitality <= 0 или vit >= max
        earlyExitInt = 0,   -- вышли: intellect <= 0
        rolls        = 0,   -- сколько раз реально дошли до math.random()
        successes    = 0,   -- сколько раз повысили
        sumChance    = 0,   -- сумма шансов (доли) для среднего
        minRand      = nil, -- минимальный встреченный math.random()
        maxRand      = nil, -- максимальный встреченный math.random()
        lastRand     = nil, -- последнее значение math.random()
        startedAt    = GetTime(),
    }
    Stats._dbg = dbg

    local function TrackRand(r)
        if dbg.minRand == nil or r < dbg.minRand then dbg.minRand = r end
        if dbg.maxRand == nil or r > dbg.maxRand then dbg.maxRand = r end
        dbg.lastRand = r
    end

    -- ========================================================================
    -- Краткая сводка
    -- ========================================================================
    local function PrintStats(tag)
        local elapsed = GetTime() - dbg.startedAt
        local avgChance = (dbg.rolls > 0) and (dbg.sumChance / dbg.rolls) or 0
        local expected  = avgChance * dbg.rolls

        print(string.format(
            "|cff00ffff[STATS %s]|r t=%.0fs calls=%d rolls=%d succ=%d | " ..
            "chance=%.4f%% expected=%.3f | rand min=%s max=%s | earlyMax=%d earlyInt=%d",
            tag or "", elapsed,
            dbg.calls, dbg.rolls, dbg.successes,
            avgChance * 100,
            expected,
            dbg.minRand and string.format("%.6f", dbg.minRand) or "nil",
            dbg.maxRand and string.format("%.6f", dbg.maxRand) or "nil",
            dbg.earlyExitMax, dbg.earlyExitInt))
    end

    function Stats:ResetStats()
        dbg.calls        = 0
        dbg.earlyExitMax = 0
        dbg.earlyExitInt = 0
        dbg.rolls        = 0
        dbg.successes    = 0
        dbg.sumChance    = 0
        dbg.minRand      = nil
        dbg.maxRand      = nil
        dbg.lastRand     = nil
        dbg.startedAt    = GetTime()
        print("|cff00ffff[STATS]|r счётчики сброшены.")
    end

    function Stats:DumpStats()
        PrintStats("dump")
    end

    -- ========================================================================
    -- Попытка повысить живучесть (vitality).
    -- Шанс: math.random(1, 10000) <= int  →  int из 10000.
    --   int=4   → 0.04%  (1 из 2500)
    --   int=50  → 0.5%   (1 из 200)
    --   int=100 → 1%     (1 из 100)
    -- ========================================================================
    function Stats:TryUpVitality(entity)
        dbg.calls = dbg.calls + 1

        if not entity or not entity.opts then return false end
        local opts = entity.opts

        local vit    = tonumber(opts.vitality)     or 0
        local maxVit = tonumber(opts.max_vitality) or 0
        if maxVit <= 0 or vit >= maxVit then
            dbg.earlyExitMax = dbg.earlyExitMax + 1
            return false
        end

        local int = tonumber(opts.intellect) or 0
        if int <= 0 then
            dbg.earlyExitInt = dbg.earlyExitInt + 1
            return false
        end

        -- Шанс в долях (для сводки) и целочисленный бросок
        local chance = int / 10000
        if chance > 1 then chance = 1 end
        dbg.sumChance = dbg.sumChance + chance
        dbg.rolls     = dbg.rolls + 1

        local roll = math.random(1, 10000)
        -- для min/max rand берём долю, чтобы видеть «сетку» math.random()
        TrackRand(math.random())

        if roll <= int then
            opts.vitality = vit + 1
            dbg.successes = dbg.successes + 1
            if Stats.DEBUG then
                print(string.format(
                    "|cff00ff00[STATS]|r +1 vitality у %s: %d → %d (roll=%d <= int=%d)",
                    tostring(opts.name or entity.id), vit, opts.vitality, roll, int))
            end
            return true
        end
        return false
    end

    -- ========================================================================
    -- Периодический вывод (раз в 60 сек), только если DEBUG включён.
    -- ========================================================================
    local ticker = CreateFrame("Frame")
    local acc = 0
    ticker:SetScript("OnUpdate", function(_, elapsed)
        if not Stats.DEBUG then return end
        acc = acc + elapsed
        if acc < 60 then return end
        acc = acc - 60
        PrintStats("60s")
    end)
    ticker:Show()

    -- ========================================================================
    -- Слэш-команда: /ns_stats [on|off|dump|reset]
    -- ========================================================================
    SLASH_NSSTATS1 = "/ns_stats"
    SlashCmdList["NSSTATS"] = function(msg)
        local cmd = (msg or ""):lower():match("^(%S*)")
        if cmd == "" or cmd == "dump" then
            Stats:DumpStats()
        elseif cmd == "on" then
            Stats.DEBUG = true
            print("|cff00ffff[STATS]|r DEBUG = true (сводка раз в 60 сек)")
        elseif cmd == "off" then
            Stats.DEBUG = false
            print("|cff00ffff[STATS]|r DEBUG = false")
        elseif cmd == "reset" then
            Stats:ResetStats()
        else
            print("|cff00ffff[STATS]|r /ns_stats on | off | dump | reset")
        end
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Tick
-- Диспетчер поведения существ.
--
-- Вызывается из Ticker:Tick() через AI:TickEntity(entity).
-- Смотрит kind + behavior, ищет Behavior[kind][behavior]:Tick.
-- Если нет — Default:Tick(entity).
--
-- Сам ничего не делает: ни поиска, ни пути, ни работы.
-- Только выбирает, чьё поведение выполнять.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local AI = NSQC4.QuestsDfAI
    if not AI then
        print("|cffff0000[NSQC4]|r AI/Tick: AI/Init.lua не загружен.")
        return
    end

    local Default = NSQC4.QuestsDfAIDefault
    if not Default then
        print("|cffff0000[NSQC4]|r AI/Tick: AI/Default.lua не загружен.")
        return
    end

    -- Переопределяем заглушку из AI/Init.lua.
    function AI:TickEntity(entity)
        if not entity or not entity.opts then return false end

        local kind     = entity.kind or entity.opts.kind
        local behavior = entity.opts.behavior or "peaceful"

        -- Прямой поиск: Behavior[kind][behavior]:Tick
        local Registry = NSQC4.QuestsDfBehavior
        local byKind   = Registry and kind and Registry[kind]
        local handler  = byKind and byKind[behavior]

        if handler and handler.Tick then
            return handler.Tick(entity)
        end

        -- Fallback
        return Default:Tick(entity)
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Ticker
-- Один глобальный OnUpdate. Раз в INTERVAL секунд проходит по всем
-- клеткам Grid и для каждого существа зовёт AI:TickEntity(entity).
--
-- ВАЖНО: чтобы существа с подходящей профессией не уступали клетки
-- существам без профессии, мы делаем ДВА ПРОХОДА за тик:
--   1. Сначала тикают существа, у которых есть профессия под выделенную
--      клетку (они резервируют "свои" клетки).
--   2. Затем тикают все остальные (они берут то, что осталось).
--
-- ФИКС: frame:Show() в EnsureFrame — без Show OnUpdate на скрытом фрейме
-- не тикает.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local AI = NSQC4.QuestsDfAI
    if not AI then
        print("|cffff0000[NSQC4]|r AI/Ticker: AI/Init.lua не загружен.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r AI/Ticker: Grid не загружен.")
        return
    end

    local Ticker = {}
    NSQC4.QuestsDfAITicker = Ticker

    local frame       = nil
    local accumulator = 0

    local function EnsureFrame()
        if frame then return frame end
        frame = CreateFrame("Frame")
        frame:SetScript("OnUpdate", function(_, elapsed)
            accumulator = accumulator + elapsed
            if accumulator < AI.INTERVAL then return end
            accumulator = accumulator - AI.INTERVAL
            Ticker:Tick()
        end)
        frame:Show()   -- ФИКС: без Show OnUpdate не тикает
        return frame
    end

    function Ticker:Tick()
        local Find = NSQC4.QuestsDfAIFind
        local withProf, withoutProf = {}, {}

        -- 1. Разделяем существ на две группы
        for i = 1, TOTAL do
            local entity = Grid:GetEntity(i)
            if entity and entity.opts then
                local hasProf = false
                if Find and Find.HasProfForSelected then
                    hasProf = Find:HasProfForSelected(entity)
                end
                if hasProf then
                    withProf[#withProf + 1] = entity
                else
                    withoutProf[#withoutProf + 1] = entity
                end
            end
        end

        -- 2. Сначала тикают "профи" — они забирают свои клетки
        for _, entity in ipairs(withProf) do
            AI:TickEntity(entity)
        end

        -- 3. Потом все остальные — берут что осталось
        for _, entity in ipairs(withoutProf) do
            AI:TickEntity(entity)
        end
    end

    function Ticker:Start()
        EnsureFrame()
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / AI / Work
-- Один клик работы существа по клетке-цели.
--
-- При каждом клике — шанс прокачки живучести (intellect / 100).
-- Звук работы: если мышь на клетке существа (cell._hovered) — играем
-- звук по ai.terrainId каждый тик.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Work = {}
    NSQC4.QuestsDfAIWork = Work

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r AI/Work: Grid не загружен.")
        return
    end

    local function EnsureWork(entity, ai)
        if ai.work and ai.work.total then return ai.work end

        local ProfApi = NSQC4.QuestsDfProfessionsApi
        local Resolve = NSQC4.QuestsDfProfessionsResolve

        local hardness = Grid:GetHardness(ai.target)
        if not hardness or hardness <= 0 then
            ai.work = { done = 0, total = 1, startHardness = 1, stepDamage = 1 }
            return ai.work
        end

        local total
        local profDef = ai.profId and ProfApi and ProfApi:Get(ai.profId) or nil

        if profDef and profDef.clicks and Resolve then
            local pp = Resolve:ProfPct(entity, ai.terrainId)
            local sp = Resolve:StrPct(entity)
            total = profDef.clicks(hardness, pp, sp)
        else
            total = math.floor(hardness + 0.5)
        end

        if not total or total < 1 then total = 1 end

        ai.work = {
            done          = 0,
            total         = total,
            startHardness = hardness,
            stepDamage    = hardness / total,
        }
        return ai.work
    end

    function Work:Tick(entity)
        if not entity or not entity.opts then return nil end

        local ai = entity.opts.ai
        if not ai or ai.state ~= "working" then return nil end
        if not ai.target then return nil end

        local w = EnsureWork(entity, ai)
        if not w then return nil end

        -- Один клик
        w.done = w.done + 1

        local hardness = Grid:GetHardness(ai.target) or 0
        if hardness > 0 then
            local newHardness = hardness - w.stepDamage
            if newHardness < 0 then newHardness = 0 end
            Grid:SetHardness(ai.target, newHardness)
        end

        -- Прокачка живучести при клике по объекту
        local Stats = NSQC4.QuestsDfAIStats
        if Stats and Stats.TryUpVitality then
            Stats:TryUpVitality(entity)
        end

        -- Звук работы: только если мышь на клетке существа
        local QuestsField = NSQC4.QuestsField
        local cell = QuestsField and QuestsField.cells and QuestsField.cells[entity.index]
        if cell and cell._hovered then
            local Sound = NSQC4.QuestsDfAISound
            if Sound then Sound:PlayTerrain(ai.terrainId) end
        end

        if w.done >= w.total then
            Grid:SetHardness(ai.target, 0)
            return true
        end

        return false
    end

    function Work:Reset(entity)
        if not entity or not entity.opts then return end
        local ai = entity.opts.ai
        if ai then ai.work = nil end
    end

end)

-- NSQC4 / Quests / df / Actions / Terrain / Earth / LeftClickEntities
-- Действие ЛКМ по существу в клетке с "earth".
-- Вызывается из LeftClickSelf.lua, когда существо НЕ спит.
--
-- Разворачивает/сворачивает существо. Если идёт анимация движения —
-- Animate:SetExpanded (мгновенно меняет размер temp), иначе Spawn
-- (перерисовывает anchor в клетке).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local A = NSQC4.QuestsDfTerrainActions
    if not A then
        print("|cffff0000[NSQC4]|r Earth/LeftClickEntities: Init.lua не загружен.")
        return
    end

    A.Types["earth"] = A.Types["earth"] or {}

    A.Types["earth"].LeftClickEntities = function(index)
        index = tonumber(index) or 0
        if index < 1 then return false end

        local Grid = NSQC4.QuestsDfGrid
        if not Grid then return false end

        local entity = Grid:GetEntity(index)
        if not entity or not entity.opts then return false end

        if entity.opts.sleeping then return false end

        entity.opts.expanded = not entity.opts.expanded

        local Animate = NSQC4.QuestsDfEntitiesAnimate
        if Animate and Animate:IsAnimating(entity.id) then
            Animate:SetExpanded(entity.id, entity.opts.expanded)
        else
            local Entities = NSQC4.QuestsDfEntities
            if Entities then
                Entities:Spawn(entity.id, index, entity.opts)
            end
        end

        print(string.format("|cff00ff00[NSQC4]|r %s: %s",
            entity.opts.name or "Существо",
            entity.opts.expanded and "развёрнут" or "свёрнут"))

        return true
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Actions / Terrain / Earth / LeftClickSelf
-- ЛКМ по ландшафту "earth" (свой клик).
--   - существо есть (не спит) → LeftClickEntities
--   - существа нет / спит     → toggle выделения ландшафта
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local A = NSQC4.QuestsDfTerrainActions
    if not A then
        print("|cffff0000[NSQC4]|r Earth/LeftClickSelf: Init.lua не загружен.")
        return
    end

    local QuestsField = NSQC4.QuestsField
    if not QuestsField then return end

    local Grid = NSQC4.QuestsDfGrid

    A.Types["earth"] = A.Types["earth"] or {}

    A.Types["earth"].LeftClickSelf = function(index)
        index = tonumber(index) or 0
        if index < 1 then return false end

        QuestsField:SetCellOnMouseClick(index, "LeftButton", function(btn, button, idx)
            local entity   = Grid and Grid:GetEntity(idx)
            local sleeping = entity and entity.opts and entity.opts.sleeping

            if entity and not sleeping then
                local T = NSQC4.QuestsDfTerrainActions
                          and NSQC4.QuestsDfTerrainActions.Types
                          and NSQC4.QuestsDfTerrainActions.Types["earth"]
                if T and T.LeftClickEntities then
                    return T.LeftClickEntities(idx)
                end
                return false
            end

            local cell = QuestsField.cells and QuestsField.cells[idx]
            if not Grid then return false end

            local isSelected = Grid:IsTerrainSelected(idx)

            if isSelected then
                Grid:SetTerrainSelected(idx, "earth", false)
                if cell and cell.texture then
                    cell.texture:SetVertexColor(1, 1, 1, 1)
                end
                print("|cffff8800[NSQC4]|r Снято выделение earth в клетке " .. idx)
            else
                Grid:SetTerrainSelected(idx, "earth", true)
                if cell and cell.texture then
                    cell.texture:SetVertexColor(0.5, 0.5, 0.5, 1)
                end
                print("|cff00ff00[NSQC4]|r Выделено earth в клетке " .. idx)
            end

            return true
        end)

        return true
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Actions / Terrain / Init
-- Каркас «объекта ландшафта» для регистрации кликов.
--
-- Namespace: NSQC4.QuestsDfTerrainActions
--
-- Подпапки (по типу ландшафта), каждая с четырьмя файлами:
--   Earth/  { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
--   Grass/  { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
--   Stone/  { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
--   Tree/   { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
--   Pool/   { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
--
-- Каждый тип регистрирует себя в реестре:
--   TerrainActions.Types[<id>] = {
--       Check, LeftClickSelf, LeftClickEntities, RightClickEntities
--   }
--
-- Точка входа: TerrainUIApi:RenderCellFromStorage(index) зовёт
--   local T = TerrainActions.Types[terrainId]
--   if T and T.Check and T.Check(index) then
--       T.LeftClickSelf(index)
--       T.RightClickEntities(index)
--   end
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Actions = {}
    NSQC4.QuestsDfTerrainActions = Actions

    -- Реестр типов: id → { Check, LeftClickSelf, LeftClickEntities, RightClickEntities }
    Actions.Types = {}

    -- ========================================================================
    -- Регистрация типа (вызывается из подпапки)
    --   id      — "earth" | "grass" | "stone" | "tree" | "pool"
    --   handler — { Check=fn, LeftClickSelf=fn, LeftClickEntities=fn, RightClickEntities=fn }
    -- ========================================================================
    function Actions:RegisterTypeHandler(id, handler)
        if type(id) ~= "string" or id == "" then return end
        if type(handler) ~= "table" then return end
        self.Types[id] = handler
    end

    function Actions:GetTypeHandler(id)
        return self.Types[id]
    end

    -- ========================================================================
    -- Проверка готовности (ленивая, для диагностики)
    -- ========================================================================
    function Actions:Init()
        local count = 0
        for id, handler in pairs(self.Types) do
            if handler.Check
               and handler.LeftClickSelf
               and handler.LeftClickEntities
               and handler.RightClickEntities then
                count = count + 1
            else
                print("|cffffaa00[NSQC4]|r TerrainActions: тип '" .. id .. "' зарегистрирован неполно.")
            end
        end

        if count == 0 then
            print("|cffffaa00[NSQC4]|r TerrainActions: ни один тип ландшафта не зарегистрирован.")
            return false
        end

        return true
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Dwarf / Spawner
-- Генерация дворфа. Генерирует УНИКАЛЬНОЕ имя (проверка byName).
-- [3] — отброс, [12] — бог.
--
-- known_skills[name] = уровень (1..prof_max), где prof_max берётся
-- из stats.prof категории существа (Dwarf/Stats.lua).
--
-- Профессии выбираются УНИКАЛЬНЫЕ по имени. Количество = skills_max.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local IMG_PATH = "Interface\\AddOns\\NSQC4\\Media\\Images\\"
    local Dwarf = NSQC4.QuestsDfDwarf
    if not Dwarf then return end

    local WEIGHTS = {
        [3] = 300, [4] = 180, [5] = 150, [6] = 120, [7] = 100,
        [8] = 80, [9] = 40, [10] = 20, [11] = 9, [12] = 1
    }

    local function GetRandomLength()
        local totalWeight = 0
        for _, w in pairs(WEIGHTS) do totalWeight = totalWeight + w end
        local roll = math.random(totalWeight)
        local current = 0
        for len = 3, 12 do
            current = current + (WEIGHTS[len] or 0)
            if roll <= current then return len end
        end
        return 3
    end

    local function ShuffleArray(array)
        local result = {}
        for i = 1, #array do result[i] = array[i] end
        for i = #result, 2, -1 do
            local j = math.random(i)
            result[i], result[j] = result[j], result[i]
        end
        return result
    end

    local function RollBehavior()
        return (math.random() < 0.5) and "aggressive" or "peaceful"
    end

    -- ========================================================================
    -- Уникальное имя
    -- ========================================================================
    local function PickUniqueName(names, category)
        local Entities = NSQC4.QuestsDfEntities
        if not Entities then return nil end
        local byName = Entities:EnsureByName()
        local list = names[category]
        if not list or #list == 0 then return nil end

        for _ = 1, 50 do
            local name = list[math.random(#list)]
            if not byName[name] then return name end
        end
        local base = list[math.random(#list)]
        for i = 2, 999 do
            local name = base .. "#" .. i
            if not byName[name] then return name end
        end
        return nil
    end

    function Dwarf:Spawn(index)
        index = tonumber(index) or 0
        if index < 1 or index > 100 then return nil end

        local Entities = NSQC4.QuestsDfEntities
        if not Entities then return nil end

        local names = NSQC4.QuestsDfDwarfNames
        if not names then return nil end

        local category = GetRandomLength()
        while not names[category] or #names[category] == 0 do
            category = category + 1
            if category > 12 then category = 3 end
        end

        local name = PickUniqueName(names, category)
        if not name then
            print("|cffff0000[NSQC4]|r Dwarf:Spawn: не удалось подобрать уникальное имя")
            return nil
        end

        local descTable = NSQC4.QuestsDfDwarfDescriptions or {}
        local desc = descTable[category] or descTable[99] or "Описание отсутствует."

        local statsTable = NSQC4.QuestsDfDwarfStats
        local stats = (statsTable and statsTable[category]) or {}

        local function rollStat(range, defaultMin)
            if not range or #range < 2 then return defaultMin, defaultMin end
            return math.random(range[1], range[2]), range[2]
        end

        local skills_max, skills_max_cap = rollStat(stats.skills_max, 1)
        local skills_cap, skills_cap_max = rollStat(stats.skills_cap, 1)
        local strength, strength_max       = rollStat(stats.strength, 1)
        local agility, agility_max         = rollStat(stats.agility, 1)
        local vitality, vitality_max       = rollStat(stats.vitality, 1)
        local intellect, intellect_max     = rollStat(stats.intellect, 1)

        local lineage, lineage_max = 0, 1
        if stats.lineage then lineage, lineage_max = rollStat(stats.lineage, 0) end

        -- ====================================================================
        -- Стартовое число профессий = skills_max (текущее число умений).
        -- known_skills[name] = уровень (1..prof_max).
        -- Берем из shuffled по одной, проверяя уникальность имени.
        -- ====================================================================
        local known_skills = {}
        local allProf = NSQC4.QuestsDfProfessions
        if allProf and #allProf > 0 then
            local shuffled = ShuffleArray(allProf)

            local profMin = (stats.prof and stats.prof[1]) or 1
            local profMax = (stats.prof and stats.prof[2]) or 1
            if profMin > profMax then profMin, profMax = profMax, profMin end

            local need = skills_max
            local added = 0
            local seen = {}

            for i = 1, #shuffled do
                if added >= need then break end
                local def = shuffled[i]
                if def and def.name and not seen[def.name] then
                    seen[def.name] = true
                    known_skills[def.name] = math.random(profMin, profMax)
                    added = added + 1
                end
            end
        end

        local statuses = {
            [3] = "Отброс", [4] = "Преступник", [5] = "Мигрант", [6] = "Гражданин",
            [7] = "Новобранец", [8] = "Воин", [9] = "Герой", [10] = "Царь",
            [11] = "Полубог", [12] = "Бог"
        }
        local title = statuses[category] or "Неизвестный"

        local opts = {
            name = name, title = title, description = desc,
            texture = IMG_PATH .. "gom", icon = IMG_PATH .. "gom",
            color = { 0.60, 0.60, 1.00 }, kind = "dwarf",
            expanded = false,
            alpha = 1.0,
            behavior = RollBehavior(),

            skills_max = skills_max, skills_cap = skills_cap,
            strength = strength, agility = agility, vitality = vitality, intellect = intellect,
            lineage = lineage,

            max_skills_max = skills_max_cap, max_skills_cap = skills_cap_max,
            max_strength = strength_max, max_agility = agility_max,
            max_vitality = vitality_max, max_intellect = intellect_max, max_lineage = lineage_max,

            known_skills = known_skills,
        }

        local entity = Entities:Spawn(nil, index, opts)
        if entity then
            local skillParts = {}
            for pname, lvl in pairs(known_skills) do
                skillParts[#skillParts + 1] = pname .. ":" .. tostring(lvl)
            end
            table.sort(skillParts)
            local skillsStr = table.concat(skillParts, ", ")
            print(string.format("|cff00ff00[NSQC4]|r Спавн дворфа: %s [%s] | %s | УМ: %d | Профессии: %s",
                name, title, opts.behavior, skills_max, skillsStr ~= "" and skillsStr or "нет"))
        end
        return entity
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Dwarf / Tooltip
-- Тултип дворфа: регистрация билдера в реестре Entities.
--
-- Регистрируется один раз при загрузке файла:
--   Entities:RegisterTooltip("dwarf", buildFn)
--
-- buildFn(opts) → function(btn) — вызывается при каждом рендере клетки
-- с дворфом (TerrainUIApi:RenderCellFromStorage → ResolveTooltip → 
-- QuestsField:SetCellTooltip). Сам код тултипа — здесь.
--
-- Общий сборщик AddLine'ов — NSQC4.QuestsDfEntitiesTooltip.Build(opts)
-- (df/Entities/Tooltip.lua). Здесь только привязка kind → Build.
-- Если у дворфа появится уникальная строка тултипа — её место здесь,
-- до или после Build, через отдельный AddLine.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Entities = NSQC4.QuestsDfEntities
    if not Entities then
        print("|cffff0000[NSQC4]|r Dwarf/Tooltip: Entities не загружен.")
        return
    end

    if not Entities.RegisterTooltip then
        print("|cffff0000[NSQC4]|r Dwarf/Tooltip: Entities:RegisterTooltip не найден (Шаг 4a не применён).")
        return
    end

    -- ========================================================================
    -- Код тултипа дворфа.
    --   opts — данные существа (name/title/description/статы/known_skills)
    --   → возвращает function(btn), которая пишет AddLine'ы.
    --
    -- Сейчас — общий Build. При желании можно добавить дворф-специфичные
    -- строки до/после, например:
    --   GameTooltip:AddLine("Борода: " .. (opts.beard or "нет"), 0.8, 0.6, 0.4)
    -- ========================================================================
    local function BuildDwarfTooltip(opts)
        if not opts then return nil end

        local Tooltip = NSQC4.QuestsDfEntitiesTooltip
        if Tooltip and Tooltip.Build then
            return Tooltip.Build(opts)
        end

        -- Резерв: если общий Build недоступен — минимальный тултип
        return function(btn)
            GameTooltip:AddLine(opts.name or "Безымянный", 1, 1, 0)
            if opts.title then
                GameTooltip:AddLine("[" .. opts.title .. "]", 1, 0.82, 0)
            end
            if opts.description and opts.description ~= "Описание отсутствует." then
                GameTooltip:AddLine(opts.description, 1, 1, 1, true)
            end
        end
    end

    -- ========================================================================
    -- Регистрация в реестре Entities (kind → билдер).
    -- Вызывается один раз при загрузке файла.
    -- TerrainUIApi:ResolveTooltip найдёт его через Entities:GetTooltip("dwarf").
    -- ========================================================================
    Entities:RegisterTooltip("dwarf", BuildDwarfTooltip)

    print("|cff00ffff[DWF-DEBUG]|r Dwarf Tooltip зарегистрирован.")
end)

-- ============================================================================
-- NSQC4 / Quests / df / Entities / Init
-- Точка входа подмодуля живых существ.
--
-- Модель byName:
--   nsDbc4.quests.df.entities.byName[name] = entity
--   grid[i].entityName = name
--
-- Реестр тултипов:
--   Entities:RegisterTooltip(kind, buildFn) — kind → function(opts) → function(btn)
--   Entities:GetTooltip(kind)               — вернуть билдер (или nil)
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then
        print("|cffff0000[NSQC4]|r Entities: ядро поля не загружено.")
        return
    end

    local Entities = {}
    Entities.__index = Entities

    NSQC4.QuestsDfEntities = Entities
    NSQC4.QuestsDfModules = NSQC4.QuestsDfModules or {}
    NSQC4.QuestsDfModules["entities"] = NSQC4.QuestsDfModules["entities"] or {
        label   = "Живые существа",
        order   = 100,
        enabled = true,
    }
    NSQC4.QuestsDfModules["entities"].api = Entities

    -- ========================================================================
    -- Реестр тултипов
    -- ========================================================================
    local tooltipBuilders = {}

    function Entities:RegisterTooltip(kind, buildFn)
        if type(kind) ~= "string" or kind == "" then return false end
        if type(buildFn) ~= "function" then return false end
        tooltipBuilders[kind] = buildFn
        return true
    end

    function Entities:GetTooltip(kind)
        if type(kind) ~= "string" or kind == "" then return nil end
        return tooltipBuilders[kind]
    end

    -- ========================================================================
    -- byName: таблица существ по имени
    -- ========================================================================
    local function EnsureByName()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.entities = nsDbc4.quests.df.entities or {}
        nsDbc4.quests.df.entities.byName = nsDbc4.quests.df.entities.byName or {}
        return nsDbc4.quests.df.entities.byName
    end

    function Entities:EnsureByName() return EnsureByName() end

    -- ========================================================================
    -- Проверка готовности (Storage/UI/Logic)
    -- ========================================================================
    local function Ready()
        return Entities.EnsureStorage ~= nil
           and Entities.Spawn        ~= nil
           and Entities.Render       ~= nil
    end

    function Entities:Init()
        if not Ready() then
            return false
        end

        self:EnsureStorage()
        if self:IsInitialized() then
            return false
        end

        self:SetInitialized(true)
        return true
    end

    NSQC4.QuestsDfModules["entities"].init = function()
        return NSQC4.QuestsDfEntities:Init()
    end

    -- ========================================================================
    -- Слэш-команды
    -- ========================================================================
    SLASH_NSDF_ENTITIES1 = "/ns_df_entities"
    SlashCmdList["NSDF_ENTITIES"] = function(msg)
        local cmd = (msg or ""):lower():match("^(%S*)")
        local E = NSQC4.QuestsDfEntities

        if cmd == "" or cmd == "info" then
            print("|cff00ff00[NSQC4]|r df/entities — живые существа")
            if not Ready() then
                print("|cffffaa00  Storage/UI/Logic ещё не загружены.|r")
                return
            end
            local st = E:EnsureStorage()
            local byName = EnsureByName()
            local n = 0
            for _ in pairs(byName) do n = n + 1 end
            print("  initialized:", E:IsInitialized())
            print("  существ в byName:", n)
            print("  в Grid:", E:Count())
            print("|cff808080Команды: /ns_df_entities init | clear|r")

        elseif cmd == "init" then
            local ok = E:Init()
            print("|cff00ff00[NSQC4]|r entities:", ok and "инициализированы." or "уже были инициализированы (или Storage не готов).")

        elseif cmd == "clear" then
            if not Ready() then
                print("|cffffaa00[NSQC4]|r entities: Storage ещё не загружен.")
                return
            end
            E:Clear()
            print("|cffff8800[NSQC4]|r entities: все существа убраны.")

        else
            print("|cffff0000[NSQC4]|r entities: неизвестная команда '" .. cmd .. "'")
        end
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Entities / Storage
-- Хранилище живых существ через byName:
--   nsDbc4.quests.df.entities.byName[name] = entity
--   grid[i].entityName = name
--
-- Entities:Spawn(id, index, opts) — opts.name должен быть свободен в byName.
-- Если занят — Spawn возвращает nil (Spawner должен был сгенерировать
-- уникальное имя сам).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Entities = NSQC4.QuestsDfEntities
    if not Entities then return end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then return end

    local function ByName()
        return Entities:EnsureByName()
    end

    local function RenderCell(index)
        local TerrainApi = NSQC4.QuestsDfTerrainUIApi
        if TerrainApi and TerrainApi.RenderCellFromStorage then
            TerrainApi:RenderCellFromStorage(index)
        end
    end

    local function RenderAll()
        local TerrainApi = NSQC4.QuestsDfTerrainUIApi
        if TerrainApi and TerrainApi.RenderAll then
            TerrainApi:RenderAll()
        end
    end

    local function StopAnimation(id)
        local Animate = NSQC4.QuestsDfEntitiesAnimate
        if Animate and id and Animate.Stop then
            Animate:Stop(id)
        end
    end

    -- ========================================================================
    -- Флаг инициализации / авто-id
    -- ========================================================================
    local function EnsureTable()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.entities = nsDbc4.quests.df.entities or {}

        local s = nsDbc4.quests.df.entities
        s.nextId = s.nextId or 1
        s.initialized = s.initialized and true or false
        s.byName = s.byName or {}
        return s
    end

    function Entities:EnsureStorage() return EnsureTable() end
    function Entities:IsInitialized() return EnsureTable().initialized and true or false end
    function Entities:SetInitialized(v) EnsureTable().initialized = v and true or false end

    function Entities:EnsureUniqueId()
        local s = EnsureTable()
        local id = "e" .. s.nextId
        s.nextId = s.nextId + 1
        return id
    end

    -- ========================================================================
    -- Доступ
    -- ========================================================================
    function Entities:GetAt(index)
        return Grid:GetEntity(index)
    end

    function Entities:Get(id)
        if not id then return nil end
        for _, e in pairs(ByName()) do
            if e.id == id then return e end
        end
        return nil
    end

    function Entities:GetByName(name)
        if type(name) ~= "string" or name == "" then return nil end
        return ByName()[name]
    end

    function Entities:GetAll()
        local list = {}
        for _, e in pairs(ByName()) do
            list[#list + 1] = e
        end
        return list
    end

    function Entities:Count()
        local n = 0
        for _ in pairs(ByName()) do n = n + 1 end
        return n
    end

    -- ========================================================================
    -- Создание / перемещение
    -- ========================================================================
    function Entities:Spawn(id, index, opts)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return nil end
        if type(opts) ~= "table" then return nil end
        if type(opts.name) ~= "string" or opts.name == "" then return nil end

        local name = opts.name
        local byName = ByName()

        -- Коллизия по имени — Spawner должен был сгенерировать уникальное
        local existing = byName[name]
        if existing then
            -- Разрешаем только если это тот же id (перемещение / обновление)
            if id and existing.id ~= id then
                print("|cffff0000[NSQC4]|r Entities:Spawn: имя '" .. name .. "' уже занято (" .. tostring(existing.id) .. ")")
                return nil
            end
        end

        id = id or existing and existing.id or self:EnsureUniqueId()
        if type(id) ~= "string" then id = tostring(id) end

        if existing then
            StopAnimation(id)
            local oldIndex = existing.index
            if oldIndex and oldIndex ~= index then
                Grid:SetEntityName(oldIndex, nil)
                RenderCell(oldIndex)
            end
            existing.index = index
            existing.opts  = opts
            existing.kind  = opts.kind or existing.kind
            byName[name]   = existing
            Grid:SetEntityName(index, name)
            RenderCell(index)
            return existing
        end

        local entity = {
            id    = id,
            index = index,
            opts  = opts,
            kind  = opts.kind or "generic",
        }

        byName[name] = entity
        Grid:SetEntityName(index, name)
        RenderCell(index)
        return entity
    end

    -- ========================================================================
    -- Удаление
    -- ========================================================================
    function Entities:Despawn(id)
        StopAnimation(id)

        local byName = ByName()
        for name, e in pairs(byName) do
            if e.id == id then
                local idx = e.index
                byName[name] = nil
                if idx then
                    Grid:SetEntityName(idx, nil)
                    if Grid.ClearSelection then
                        Grid:ClearSelection("entity", id)
                    end
                    RenderCell(idx)
                end
                return true
            end
        end
        return false
    end

    function Entities:Move(id, newIndex)
        local entity = self:Get(id)
        if not entity then return false end
        return self:Spawn(entity.id, newIndex, entity.opts) ~= nil
    end

    function Entities:Clear()
        local Animate = NSQC4.QuestsDfEntitiesAnimate
        local byName = ByName()

        for name, e in pairs(byName) do
            if Animate and e.id then Animate:Stop(e.id) end
            if Grid.ClearSelection then
                Grid:ClearSelection("entity", e.id)
            end
            if e.index then
                Grid:SetEntityName(e.index, nil)
            end
            byName[name] = nil
        end

        for i = 1, TOTAL do
            RenderCell(i)
        end
    end

    -- ========================================================================
    -- Выделение (по id — как было)
    -- ========================================================================
    function Entities:ToggleSelect(id)
        local entity = self:Get(id)
        if not entity then return false end

        local isSel = Grid:IsSelected("entity", id)

        if isSel then
            Grid:ClearSelection("entity", id)
        else
            Grid:SetSelected("entity", id, true)
        end

        if entity.index then RenderCell(entity.index) end
        return true
    end

    function Entities:IsSelected(id)
        return Grid:IsSelected("entity", id)
    end

    function Entities:ClearSelection()
        Grid:ClearAllSelections("entity")
        RenderAll()
    end

    -- ========================================================================
    -- Рендер
    -- ========================================================================
    function Entities:RenderCell(index)
        RenderCell(index)
    end

    function Entities:Render()
        RenderAll()
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Entities / Tooltip
-- Сборка GameTooltip. opts._landscape = { index, terrainId }.
-- В OnEnter hardness читается заново из Grid и Terrain/Data — тултип всегда
-- показывает актуальное состояние.
--
-- known_skills — таблица [имя] = уровень (1..100).
--
-- Порядок строк:
--   имя / титул / описание
--   статы
--   ПРОФЕССИИ        ← выше, чтобы не обрезалось снизу
--   ЛАНДШАФТ         ← последним (может обрезаться — не критично)
--
-- Отладка:
--   /run NSQC4.QuestsDfTooltip__DEBUG = true
--   /run NSQC4.QuestsDfTooltip__DEBUG_SKILLS = true
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Tooltip = {}
    NSQC4.QuestsDfEntitiesTooltip = Tooltip

    local function dbgSkills(...)
        if NSQC4.QuestsDfTooltip__DEBUG_SKILLS then
            print("|cffffaa00[TT-SKILLS]|r", ...)
        end
    end

    local function GetColorByPercent(val, maxVal)
        if not val or not maxVal or maxVal == 0 then return 0.62, 0.62, 0.62 end
        local percent = val / maxVal
        if percent <= 0.05 then return 0.62, 0.62, 0.62
        elseif percent <= 0.15 then return 1.00, 1.00, 1.00
        elseif percent <= 0.30 then return 0.12, 1.00, 0.00
        elseif percent <= 0.50 then return 0.00, 0.44, 0.87
        elseif percent <= 0.70 then return 0.64, 0.21, 0.93
        elseif percent <= 0.90 then return 1.00, 0.50, 0.00
        else return 0.90, 0.80, 0.50
        end
    end
    Tooltip.GetColorByPercent = GetColorByPercent

    local function LandscapeLine(index, terrainId)
        local Grid = NSQC4.QuestsDfGrid
        local Data = NSQC4.QuestsDfTerrainData
        if not index or not Grid or not Data then return nil end

        local curH = Grid:GetHardness(index)
        local def = terrainId and Data:Get(terrainId) or nil
        local maxH = def and def.hardness or nil
        if not curH or not maxH or maxH <= 0 then return nil end

        local r, g, b = GetColorByPercent(curH, maxH)
        local curR = math.floor(curH + 0.5)
        local maxR = math.floor(maxH + 0.5)
        local pct  = (curH / maxH) * 100
        return string.format("Ландшафт: %s  %d / %d (%.0f%%)",
            tostring(terrainId or "?"), curR, maxR, pct), r, g, b
    end

    -- ========================================================================
    -- Строка с профессиями: "Шахтер: 42, Топорщик: 7, ..."
    -- known_skills — hash [имя] = уровень.
    -- Если в таблице есть и числовые ключи (старый формат) — они тоже
    -- попадут, но в виде "1: Шахтер" (индекс: имя). Чтобы этого не было,
    -- сущность надо пересоздать (/ns_df restart).
    -- ========================================================================
    local function BuildSkillsLine(known_skills)
        if type(known_skills) ~= "table" then
            dbgSkills("BuildSkillsLine: known_skills is not table, type=",
                      type(known_skills))
            return nil
        end

        local parts = {}
        local count = 0

        for name, lvl in pairs(known_skills) do
            count = count + 1
            parts[#parts + 1] = tostring(name) .. ": " .. tostring(lvl)
            dbgSkills(string.format("  [%s] = %s (type name=%s, type lvl=%s)",
                tostring(name), tostring(lvl), type(name), type(lvl)))
        end

        dbgSkills(string.format("BuildSkillsLine: count=%d", count))

        if count == 0 then
            dbgSkills("BuildSkillsLine: empty → nil")
            return nil
        end

        table.sort(parts)
        local line = table.concat(parts, ", ")
        dbgSkills("BuildSkillsLine: line = " .. line)
        return line
    end

    function Tooltip.Build(opts)
        if not opts then return nil end

        return function(btn)
            -- ================================================================
            -- ОБЩАЯ ОТЛАДКА
            -- ================================================================
            if NSQC4.QuestsDfTooltip__DEBUG then
                print(string.format("[TT] name=%s title=%s kind=%s",
                    tostring(opts.name), tostring(opts.title), tostring(opts.kind)))
                local land = opts._landscape
                if land then
                    print(string.format("[TT] landscape index=%s terrain=%s",
                        tostring(land.index), tostring(land.terrainId)))
                else
                    print("[TT] landscape = nil")
                end
                print(string.format("[TT] known_skills type=%s",
                    type(opts.known_skills)))
                if type(opts.known_skills) == "table" then
                    local n = 0
                    for k, v in pairs(opts.known_skills) do
                        n = n + 1
                        print(string.format("[TT]   skills[%s] = %s (%s)",
                            tostring(k), tostring(v), type(v)))
                    end
                    print("[TT] known_skills count = " .. n)
                end
            end

            -- ================================================================
            -- ИМЯ / ТИТУЛ / ОПИСАНИЕ
            -- ================================================================
            GameTooltip:AddLine(opts.name or "Безымянный", 1, 1, 0)

            if opts.title then
                GameTooltip:AddLine("[" .. opts.title .. "]", 1, 0.82, 0)
            end

            if opts.description and opts.description ~= "Описание отсутствует." then
                GameTooltip:AddLine(opts.description, 1, 1, 1, true)
            end

            GameTooltip:AddLine(" ")

            -- ================================================================
            -- СТАТЫ
            -- ================================================================
            if opts.skills_max and opts.skills_cap then
                local r, g, b = GetColorByPercent(opts.skills_cap, opts.max_skills_cap)
                GameTooltip:AddLine(string.format("Умения: %d / %d",
                    opts.skills_max, opts.max_skills_cap or opts.skills_cap), r, g, b)
            end

            if opts.strength then
                local r, g, b = GetColorByPercent(opts.strength, opts.max_strength)
                GameTooltip:AddLine(string.format("Сила: %d / %d",
                    opts.strength, opts.max_strength or opts.strength), r, g, b)
            end

            if opts.agility then
                local r, g, b = GetColorByPercent(opts.agility, opts.max_agility)
                GameTooltip:AddLine(string.format("Ловкость: %d / %d",
                    opts.agility, opts.max_agility or opts.agility), r, g, b)
            end

            if opts.vitality then
                local r, g, b = GetColorByPercent(opts.vitality, opts.max_vitality)
                GameTooltip:AddLine(string.format("Живучесть: %d / %d",
                    opts.vitality, opts.max_vitality or opts.vitality), r, g, b)
            end

            if opts.intellect then
                local r, g, b = GetColorByPercent(opts.intellect, opts.max_intellect)
                GameTooltip:AddLine(string.format("Интеллект: %d / %d",
                    opts.intellect, opts.max_intellect or opts.intellect), r, g, b)
            end

            if opts.lineage ~= nil then
                local lineageStr = (opts.lineage == 1)
                    and "|cffffd100Аристократ|r"
                    or  "|cff808080Безродный|r"
                GameTooltip:AddLine("Происхождение: " .. lineageStr, 1, 1, 1)
            end

            -- ================================================================
            -- ПРОФЕССИИ (выше ландшафта — чтобы не обрезалось снизу)
            -- ================================================================
            local skillsLine = BuildSkillsLine(opts.known_skills)
            if skillsLine then
                GameTooltip:AddLine(" ")
                GameTooltip:AddLine("Профессии:", 1, 0.82, 0.0)
                GameTooltip:AddLine(skillsLine, 0.8, 1.0, 0.8, true)
            else
                dbgSkills("Профессии: строка НЕ выведена (BuildSkillsLine = nil)")
            end

            -- ================================================================
            -- ЛАНДШАФТ (последним — может обрезаться, это не критично)
            -- ================================================================
            local land = opts._landscape
            if land and land.index then
                local line, r, g, b = LandscapeLine(land.index, land.terrainId)
                if line then
                    GameTooltip:AddLine(" ")
                    GameTooltip:AddLine(line, r, g, b)
                else
                    dbgSkills("LandscapeLine вернул nil (index=" ..
                        tostring(land.index) .. ", terrain=" ..
                        tostring(land.terrainId) .. ")")
                end
            end
        end
    end

end)

 FILE: df/Entities/UI/Animate.lua

-- ============================================================================
-- NSQC4 / Quests / df / Entities / UI / Animate
-- Плавное передвижение иконки существа между клетками.
--
-- temp — на кнопке toIndex, слой ARTWORK/7 (слой существа).
-- Anchor в fromIndex прячется на время анимации.
--
-- ФИКС: при завершении (Tick) и в Stop перерисовываются ОБЕ клетки —
-- fromIndex и toIndex. Иначе клики/тултип остаются на fromIndex, где
-- существа уже нет (entity.index уже в toIndex с начала шага).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Animate = {}
    NSQC4.QuestsDfEntitiesAnimate = Animate

    Animate.DEBUG = false

    local function dbg(...)
        if Animate.DEBUG then print("|cff00ffff[ANIM]|r", ...) end
    end

    local QuestsField = NSQC4.QuestsField
    if not QuestsField then
        print("|cffff0000[NSQC4]|r Animate: поле не загружено.")
        return
    end

    local active = {}
    local frame  = nil

    local function CellGeom(index)
        local f = QuestsField.frame
        local btn = QuestsField.cells and QuestsField.cells[index]
        if not f or not btn then return nil end
        local fL, fB = f:GetLeft() or 0, f:GetBottom() or 0
        local bL, bB = btn:GetLeft() or 0, btn:GetBottom() or 0
        local w, h   = btn:GetWidth() or 0, btn:GetHeight() or 0
        return (bL - fL), (bB - fB), w, h
    end

    local function LayoutFor(anchorPos, expanded, left, bottom, cellW, cellH)
        if expanded then
            return "CENTER", left + cellW / 2, bottom + cellH / 2, cellW, cellH
        end
        local w, h = cellW / 3, cellH / 3
        local point, px, py
        if anchorPos == "TOPLEFT" then
            point, px, py = "TOPLEFT", left, bottom + cellH
        elseif anchorPos == "TOPCENTER" then
            point, px, py = "TOP", left + cellW / 2, bottom + cellH
        elseif anchorPos == "TOPRIGHT" then
            point, px, py = "TOPRIGHT", left + cellW, bottom + cellH
        elseif anchorPos == "MIDLEFT" then
            point, px, py = "LEFT", left, bottom + cellH / 2
        elseif anchorPos == "CENTER" then
            point, px, py = "CENTER", left + cellW / 2, bottom + cellH / 2
        elseif anchorPos == "MIDRIGHT" then
            point, px, py = "RIGHT", left + cellW, bottom + cellH / 2
        elseif anchorPos == "BOTLEFT" then
            point, px, py = "BOTTOMLEFT", left, bottom
        elseif anchorPos == "BOTCENTER" then
            point, px, py = "BOTTOM", left + cellW / 2, bottom
        elseif anchorPos == "BOTRIGHT" then
            point, px, py = "BOTTOMRIGHT", left + cellW, bottom
        else
            point, px, py = "TOPLEFT", left, bottom + cellH
        end
        return point, px, py, w, h
    end

    local function EnsureFrame()
        if frame then return frame end
        frame = CreateFrame("Frame")
        frame:SetScript("OnUpdate", function() Animate:Tick() end)
        frame:Show()
        dbg("EnsureFrame: created and shown")
        return frame
    end

    local function RenderCell(index)
        if not index then return end
        local TerrainApi = NSQC4.QuestsDfTerrainUIApi
        if TerrainApi and TerrainApi.RenderCellFromStorage then
            TerrainApi:RenderCellFromStorage(index)
        end
    end

    -- Перерисовать обе клетки — старую и новую
    local function RenderBoth(fromIndex, toIndex)
        if fromIndex and toIndex and fromIndex == toIndex then
            RenderCell(toIndex)
            return
        end
        RenderCell(fromIndex)
        RenderCell(toIndex)
    end

    local function HostFor(toIndex)
        local btn = QuestsField.cells and QuestsField.cells[toIndex]
        return btn or QuestsField.frame
    end

    function Animate:Start(id, opts, fromIndex, toIndex, duration)
        if not id or not opts or not opts.icon or not toIndex then
            dbg("Start: bad args", tostring(id), tostring(opts and opts.icon), tostring(toIndex))
            return false
        end
        if not duration or duration <= 0 then duration = 1 end

        if active[id] then
            if active[id].temp then active[id].temp:Hide() end
            active[id] = nil
        end

        local fromL, fromB, fromW, fromH = CellGeom(fromIndex)
        local toL,   toB,   toW,   toH   = CellGeom(toIndex)
        if not fromL or not toL then
            dbg("Start: CellGeom nil", tostring(fromIndex), tostring(toIndex))
            return false
        end

        local anchorPos = opts.anchor or "TOPLEFT"
        local expanded  = opts.expanded and true or false

        local fromPoint, fromX, fromY, fromSizeW, fromSizeH =
            LayoutFor(anchorPos, expanded, fromL, fromB, fromW, fromH)
        local toPoint, toX, toY, toSizeW, toSizeH =
            LayoutFor(anchorPos, expanded, toL, toB, toW, toH)

        local host = HostFor(toIndex)
        local temp = host:CreateTexture(nil, "ARTWORK", nil, 7)
        temp:SetDrawLayer("ARTWORK", 7)
        temp:SetTexture(opts.icon)
        temp:SetTexCoord(0, 1, 0, 1)
        temp:SetVertexColor(1, 1, 1, 1)
        temp:SetAlpha(opts.alpha or 1)
        temp:SetSize(fromSizeW, fromSizeH)
        temp:SetPoint(fromPoint, QuestsField.frame, "BOTTOMLEFT", fromX, fromY)
        temp:Show()

        active[id] = {
            temp      = temp,
            icon      = opts.icon,
            alpha     = opts.alpha or 1,
            expanded  = expanded,
            anchorPos = anchorPos,
            point     = fromPoint,
            fromX = fromX, fromY = fromY,
            toX   = toX,   toY   = toY,
            sizeW = fromSizeW, sizeH = fromSizeH,
            toSizeW = toSizeW, toSizeH = toSizeH,
            duration  = duration,
            startedAt = GetTime(),
            fromIndex = fromIndex,
            toIndex   = toIndex,
        }

        local cell = QuestsField.cells and QuestsField.cells[fromIndex]
        if cell and cell.anchors and cell.anchors[anchorPos] then
            cell.anchors[anchorPos]:Hide()
        end

        EnsureFrame()
        dbg("Start", tostring(id), fromIndex, "->", toIndex, "dur", duration)
        return true
    end

    function Animate:SetExpanded(id, expanded)
        local a = active[id]
        if not a then return false end
        expanded = expanded and true or false
        if a.expanded == expanded then return true end

        local fromL, fromB, fromW, fromH = CellGeom(a.fromIndex)
        local toL,   toB,   toW,   toH   = CellGeom(a.toIndex)
        if not fromL or not toL then return false end

        local fromPoint, fromX, fromY, fromSizeW, fromSizeH =
            LayoutFor(a.anchorPos, expanded, fromL, fromB, fromW, fromH)
        local toPoint, toX, toY, toSizeW, toSizeH =
            LayoutFor(a.anchorPos, expanded, toL, toB, toW, toH)

        a.expanded = expanded
        a.point = fromPoint
        a.fromX, a.fromY = fromX, fromY
        a.toX,   a.toY   = toX,   toY
        a.sizeW, a.sizeH = fromSizeW, fromSizeH
        a.toSizeW, a.toSizeH = toSizeW, toSizeH

        if a.temp then a.temp:SetSize(fromSizeW, fromSizeH) end

        local t = (GetTime() - a.startedAt) / a.duration
        if t > 1 then t = 1 end
        if a.temp then
            local x = a.fromX + (a.toX - a.fromX) * t
            local y = a.fromY + (a.toY - a.fromY) * t
            a.temp:ClearAllPoints()
            a.temp:SetPoint(a.point, QuestsField.frame, "BOTTOMLEFT", x, y)
        end
        dbg("SetExpanded", tostring(id), expanded)
        return true
    end

    -- ========================================================================
    -- Stop: обрываем анимацию. Перерисовываем ОБЕ клетки —
    -- fromIndex (снять клик/тултип с пустой) и toIndex (поставить на новую).
    -- ========================================================================
    function Animate:Stop(id)
        local a = active[id]
        if not a then return false end
        local fromIndex, toIndex = a.fromIndex, a.toIndex
        if a.temp then
            a.temp:Hide()
            a.temp:ClearAllPoints()
            a.temp:SetTexture(nil)
        end
        active[id] = nil
        RenderBoth(fromIndex, toIndex)
        dbg("Stop", tostring(id), fromIndex, "->", toIndex)
        return true
    end

    function Animate:IsAnimating(id)
        return id and active[id] ~= nil
    end

    -- ========================================================================
    -- Tick: при завершении — тоже обе клетки.
    -- ========================================================================
    function Animate:Tick()
        local now = GetTime()
        for id, a in pairs(active) do
            local t = (now - a.startedAt) / a.duration
            if t >= 1 then
                local fromIndex, toIndex = a.fromIndex, a.toIndex
                if a.temp then
                    a.temp:Hide()
                    a.temp:ClearAllPoints()
                    a.temp:SetTexture(nil)
                end
                active[id] = nil
                RenderBoth(fromIndex, toIndex)
                dbg("Tick done", tostring(id), fromIndex, "->", toIndex)
            else
                local x = a.fromX + (a.toX - a.fromX) * t
                local y = a.fromY + (a.toY - a.fromY) * t
                local w = a.sizeW + (a.toSizeW - a.sizeW) * t
                local h = a.sizeH + (a.toSizeH - a.sizeH) * t
                if a.temp then
                    a.temp:ClearAllPoints()
                    a.temp:SetPoint(a.point, QuestsField.frame, "BOTTOMLEFT", x, y)
                    a.temp:SetSize(w, h)
                end
            end
        end
    end

    function Animate:Dump()
        local n = 0
        for id, a in pairs(active) do
            n = n + 1
            print(string.format("[ANIM] %s: %s→%s  t=%.2f  temp=%s",
                tostring(id), tostring(a.fromIndex), tostring(a.toIndex),
                (GetTime() - a.startedAt) / a.duration,
                a.temp and (a.temp:IsShown() and "shown" or "hidden") or "nil"))
        end
        print("[ANIM] active:", n,
              "frame:", frame and (frame:IsShown() and "shown" or "hidden") or "nil")
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Entities / UI / Api
-- Визуал существ: ТОЛЬКО иконка. Клики и тултип НЕ вешает.
--
-- Клики существа вешает ландшафт (TerrainActions) через Mouse.
-- Тултип существа вешает TerrainUIApi (владелец клетки) через
-- QuestsField:SetCellTooltip / ClearCellTooltip (Field/UI/API/Tooltip.lua).
--
-- Пока идёт анимация (Animate:IsAnimating(id)), Render пропускается —
-- иконку ведёт Animate, а не Api. Иначе телепорт перебьёт анимацию.
--
-- Контракт opts:
--   icon    = "path"            — иконка (обязательна для Render)
--   anchor  = "TOPLEFT"|...     — позиция (по умолчанию "TOPLEFT")
--   expanded= true|false        — на всю клетку / 1/3
--   alpha   = 0..1              — прозрачность иконки
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsField
    if not QuestsField then
        print("|cffff0000[NSQC4]|r Entities/UI/Api: ядро поля не загружено.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    -- ========================================================================
    -- Namespace API
    -- ========================================================================
    local UIApi = {}
    NSQC4.QuestsDfEntitiesUIApi = UIApi

    -- ========================================================================
    -- Константы
    -- ========================================================================
    local DEFAULT_ANCHOR = "TOPLEFT"
    local DEFAULT_ALPHA  = 1.0

    local function Valid(index)
        index = tonumber(index) or 0
        return (index >= 1 and index <= TOTAL) and index or nil
    end

    -- ========================================================================
    -- Идёт ли анимация у существа в клетке index?
    -- Смотрим entity в Grid: если у него есть id и Animate:IsAnimating(id)
    -- → true. Тогда Render не трогает иконку.
    -- ========================================================================
    local function IsAnimatingAt(index)
        local Animate = NSQC4.QuestsDfEntitiesAnimate
        if not Animate or not Animate.IsAnimating then return false end

        local Grid = NSQC4.QuestsDfGrid
        if not Grid then return false end

        local entity = Grid:GetEntity(index)
        if not entity or not entity.id then return false end

        return Animate:IsAnimating(entity.id)
    end

    -- ========================================================================
    -- RENDER — ТОЛЬКО визуал иконки существа в клетке.
    --   opts == nil → очистить иконку (якоря).
    --   Клики и тултип НЕ трогает.
    --   Если идёт анимация — ничего не делает (иконку ведёт Animate).
    -- ========================================================================
    function UIApi:Render(index, opts)
        index = Valid(index)
        if not index then return false end

        -- Анимация в приоритете: не трогаем иконку
        if IsAnimatingAt(index) then
            return true
        end

        if not opts then
            QuestsField:ClearAnchorTextures(index)
            return true
        end

        local anchor   = opts.anchor   or DEFAULT_ANCHOR
        local expanded = opts.expanded and true or false
        local alpha    = opts.alpha    or DEFAULT_ALPHA

        if opts.icon then
            QuestsField:SetAnchorExpanded(index, anchor, opts.icon, expanded)
            local cell = QuestsField.cells and QuestsField.cells[index]
            local tex  = cell and cell.anchors and cell.anchors[anchor]
            if tex then tex:SetAlpha(alpha) end
        else
            QuestsField:ClearAnchorTextures(index)
        end

        return true
    end

    -- ========================================================================
    -- RENDER CELL FROM STORAGE — иконка по данным из Grid:GetEntity(index)
    -- ========================================================================
    function UIApi:RenderCellFromStorage(index)
        index = Valid(index)
        if not index then return false end

        local Grid = NSQC4.QuestsDfGrid
        if not Grid then return false end

        local entity = Grid:GetEntity(index)
        if not entity or not entity.opts then
            self:Render(index, nil)
            return true
        end

        local opts = entity.opts

        self:Render(index, {
            icon     = opts.icon or opts.texture,
            anchor   = opts.anchor,
            expanded = opts.expanded,
            alpha    = opts.alpha,
        })

        return true
    end

    -- ========================================================================
    -- RENDER ALL
    -- ========================================================================
    function UIApi:RenderAll()
        for i = 1, TOTAL do
            self:RenderCellFromStorage(i)
        end
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Goblin / Spawner
-- Генерация гоблина. Генерирует УНИКАЛЬНОЕ имя (проверка byName).
-- [1] — бог, [10] — отброс.
--
-- known_skills[name] = уровень (1..prof_max), где prof_max берётся
-- из stats.prof категории существа (Goblin/Stats.lua).
--
-- Профессии выбираются УНИКАЛЬНЫЕ по имени. Количество = skills_max.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local IMG_PATH = "Interface\\AddOns\\NSQC4\\Media\\Images\\"
    local Goblin = NSQC4.QuestsDfGoblin
    if not Goblin then return end

    local WEIGHTS = {
        [1] = 1, [2] = 9, [3] = 20, [4] = 40, [5] = 80,
        [6] = 100, [7] = 120, [8] = 150, [9] = 180, [10] = 300
    }

    local function GetRandomLength()
        local totalWeight = 0
        for _, w in pairs(WEIGHTS) do totalWeight = totalWeight + w end
        local roll = math.random(totalWeight)
        local current = 0
        for len = 1, 10 do
            current = current + (WEIGHTS[len] or 0)
            if roll <= current then return len end
        end
        return 10
    end

    local function ShuffleArray(array)
        local result = {}
        for i = 1, #array do result[i] = array[i] end
        for i = #result, 2, -1 do
            local j = math.random(i)
            result[i], result[j] = result[j], result[i]
        end
        return result
    end

    local function RollBehavior()
        return (math.random() < 0.5) and "aggressive" or "peaceful"
    end

    -- ========================================================================
    -- Уникальное имя: пробуем случайное, пока не свободно в byName.
    -- ========================================================================
    local function PickUniqueName(names, category)
        local Entities = NSQC4.QuestsDfEntities
        if not Entities then return nil end
        local byName = Entities:EnsureByName()
        local list = names[category]
        if not list or #list == 0 then return nil end

        for _ = 1, 50 do
            local name = list[math.random(#list)]
            if not byName[name] then return name end
        end
        -- fallback: суффикс
        local base = list[math.random(#list)]
        for i = 2, 999 do
            local name = base .. "#" .. i
            if not byName[name] then return name end
        end
        return nil
    end

    function Goblin:Spawn(index)
        index = tonumber(index) or 0
        if index < 1 or index > 100 then return nil end

        local Entities = NSQC4.QuestsDfEntities
        if not Entities then return nil end

        local names = NSQC4.QuestsDfGoblinNames
        if not names then return nil end

        local category = GetRandomLength()
        local name = PickUniqueName(names, category)
        if not name then
            print("|cffff0000[NSQC4]|r Goblin:Spawn: не удалось подобрать уникальное имя")
            return nil
        end

        local descTable = NSQC4.QuestsDfGoblinDescriptions or {}
        local desc = descTable[category] or descTable[99] or "Описание отсутствует."

        local statsTable = NSQC4.QuestsDfGoblinStats
        local stats = (statsTable and statsTable[category]) or {}

        local function rollStat(range, defaultMin)
            if not range or #range < 2 then return defaultMin, defaultMin end
            return math.random(range[1], range[2]), range[2]
        end

        local skills_max, skills_max_cap = rollStat(stats.skills_max, 1)
        local skills_cap, skills_cap_max = rollStat(stats.skills_cap, 1)
        local strength, strength_max       = rollStat(stats.strength, 1)
        local agility, agility_max         = rollStat(stats.agility, 1)
        local vitality, vitality_max       = rollStat(stats.vitality, 1)
        local intellect, intellect_max     = rollStat(stats.intellect, 1)

        local lineage, lineage_max = 0, 1
        if stats.lineage then lineage, lineage_max = rollStat(stats.lineage, 0) end

        -- ====================================================================
        -- Стартовое число профессий = skills_max (текущее число умений).
        -- known_skills[name] = уровень (1..prof_max).
        -- Берем из shuffled по одной, проверяя уникальность имени.
        -- ====================================================================
        local known_skills = {}
        local allProf = NSQC4.QuestsDfProfessions
        if allProf and #allProf > 0 then
            local shuffled = ShuffleArray(allProf)

            local profMin = (stats.prof and stats.prof[1]) or 1
            local profMax = (stats.prof and stats.prof[2]) or 1
            if profMin > profMax then profMin, profMax = profMax, profMin end

            local need = skills_max
            local added = 0
            local seen = {}

            for i = 1, #shuffled do
                if added >= need then break end
                local def = shuffled[i]
                if def and def.name and not seen[def.name] then
                    seen[def.name] = true
                    known_skills[def.name] = math.random(profMin, profMax)
                    added = added + 1
                end
            end
        end

        local statuses = {
            [1] = "Бог", [2] = "Полубог", [3] = "Царь", [4] = "Герой", [5] = "Воин",
            [6] = "Новобранец", [7] = "Гражданин", [8] = "Мигрант", [9] = "Преступник", [10] = "Отброс"
        }
        local title = statuses[category] or "Неизвестный"

        local opts = {
            name = name, title = title, description = desc,
            texture = IMG_PATH .. "gob", icon = IMG_PATH .. "gob",
            color = { 1.00, 1.00, 0.00 }, kind = "goblin",
            expanded = false,
            alpha = 1.0,
            behavior = RollBehavior(),

            skills_max = skills_max, skills_cap = skills_cap,
            strength = strength, agility = agility, vitality = vitality, intellect = intellect,
            lineage = lineage,

            max_skills_max = skills_max_cap, max_skills_cap = skills_cap_max,
            max_strength = strength_max, max_agility = agility_max,
            max_vitality = vitality_max, max_intellect = intellect_max, max_lineage = lineage_max,

            known_skills = known_skills,
        }

        local entity = Entities:Spawn(nil, index, opts)
        if entity then
            local skillParts = {}
            for pname, lvl in pairs(known_skills) do
                skillParts[#skillParts + 1] = pname .. ":" .. tostring(lvl)
            end
            table.sort(skillParts)
            local skillsStr = table.concat(skillParts, ", ")
            print(string.format("|cff00ff00[NSQC4]|r Спавн гоблина: %s [%s] | %s | УМ: %d | Профессии: %s",
                name, title, opts.behavior, skills_max, skillsStr ~= "" and skillsStr or "нет"))
        end
        return entity
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Goblin / Tooltip
-- Тултип гоблина: регистрация билдера в реестре Entities.
--
-- Регистрируется один раз при загрузке файла:
--   Entities:RegisterTooltip("goblin", buildFn)
--
-- buildFn(opts) → function(btn) — вызывается при каждом рендере клетки
-- с гоблином (TerrainUIApi:RenderCellFromStorage → ResolveTooltip →
-- QuestsField:SetCellTooltip). Сам код тултипа — здесь.
--
-- Общий сборщик AddLine'ов — NSQC4.QuestsDfEntitiesTooltip.Build(opts)
-- (df/Entities/Tooltip.lua). Здесь только привязка kind → Build.
-- Если у гоблина появится уникальная строка тултипа — её место здесь,
-- до или после Build, через отдельный AddLine.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Entities = NSQC4.QuestsDfEntities
    if not Entities then
        print("|cffff0000[NSQC4]|r Goblin/Tooltip: Entities не загружен.")
        return
    end

    if not Entities.RegisterTooltip then
        print("|cffff0000[NSQC4]|r Goblin/Tooltip: Entities:RegisterTooltip не найден (Шаг 4a не применён).")
        return
    end

    -- ========================================================================
    -- Код тултипа гоблина.
    --   opts — данные существа (name/title/description/статы/known_skills)
    --   → возвращает function(btn), которая пишет AddLine'ы.
    --
    -- Сейчас — общий Build. При желании можно добавить гоблин-специфичные
    -- строки до/после, например:
    --   GameTooltip:AddLine("Жадность: " .. (opts.greed or "умеренная"), 0.6, 1, 0.2)
    -- ========================================================================
    local function BuildGoblinTooltip(opts)
        if not opts then return nil end

        local Tooltip = NSQC4.QuestsDfEntitiesTooltip
        if Tooltip and Tooltip.Build then
            return Tooltip.Build(opts)
        end

        -- Резерв: если общий Build недоступен — минимальный тултип
        return function(btn)
            GameTooltip:AddLine(opts.name or "Безымянный", 1, 1, 0)
            if opts.title then
                GameTooltip:AddLine("[" .. opts.title .. "]", 1, 0.82, 0)
            end
            if opts.description and opts.description ~= "Описание отсутствует." then
                GameTooltip:AddLine(opts.description, 1, 1, 1, true)
            end
        end
    end

    -- ========================================================================
    -- Регистрация в реестре Entities (kind → билдер).
    -- Вызывается один раз при загрузке файла.
    -- TerrainUIApi:ResolveTooltip найдёт его через Entities:GetTooltip("goblin").
    -- ========================================================================
    Entities:RegisterTooltip("goblin", BuildGoblinTooltip)

    print("|cff00ffff[GOB-DEBUG]|r Goblin Tooltip зарегистрирован.")
end)

    -- ========================================================================
    -- Поставить текущее HP клетки.
    --   index  — 1..100
    --   value  — число (может быть 0), nil = сбросить
    -- ========================================================================
    function Grid:SetHardness(index, value)
        local c = self:GetCell(index)
        if not c then return false end
        c.hardness = value
        return true
    end

    -- ========================================================================

    -- Сбросить HP клетки в nil.
    -- ========================================================================

-- ============================================================================
-- NSQC4 / Quests / df / Grid / Neighbors
-- Соседи клетки на сетке 10×10 (4-связность).
-- Только индексы. Ничего не знает про проходимость, существ, ландшафт.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then
        print("|cffff0000[NSQC4]|r Grid/Neighbors: Grid не загружен.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local COLS, ROWS = CONST.COLS, CONST.ROWS
    local TOTAL = COLS * ROWS

    -- ========================================================================
    -- Индекс → col, row (0..COLS-1, 0..ROWS-1).
    -- ========================================================================
    local function IndexToColRow(index)
        local i = index - 1
        return i % COLS, math.floor(i / COLS)
    end

    -- ========================================================================
    -- col, row → индекс (1..TOTAL).
    -- ========================================================================
    local function ColRowToIndex(col, row)
        return row * COLS + col + 1
    end

    -- ========================================================================
    -- Соседи клетки (4-связность). Возвращает массив индексов.
    -- ========================================================================
    function Grid:Neighbors(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return {} end

        local col, row = IndexToColRow(index)
        local result = {}

        -- Верх
        if row > 0 then
            result[#result + 1] = ColRowToIndex(col, row - 1)
        end
        -- Низ
        if row < ROWS - 1 then
            result[#result + 1] = ColRowToIndex(col, row + 1)
        end
        -- Лево
        if col > 0 then
            result[#result + 1] = ColRowToIndex(col - 1, row)
        end
        -- Право
        if col < COLS - 1 then
            result[#result + 1] = ColRowToIndex(col + 1, row)
        end

        return result
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Grid
-- Единое хранилище клеток 10×10.
--
-- Модель byName:
--   nsDbc4.quests.df.entities.byName[name] = entity  — таблица существа
--   grid[i].entityName = name                        — ссылка на существо
--
-- Grid:GetEntity(index) резолвит по byName → возвращает entity (как раньше).
-- Снаружи ничего не меняется: Find, Actions, UIApi, Animate продолжают
-- получать entity через Grid:GetEntity(index).
--
-- SetTerrainSelected(index, terrainId, true) снимает ВСЕ другие типы
-- в этой клетке — на клетке всегда один выделенный terrain.
-- IsTerrainSelected(index, terrainId) — проверка конкретного типа.
--
-- Reserve / Release / IsReserved — резерв клетки под существо.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Grid = {}
    NSQC4.QuestsDfGrid = Grid

    Grid.DEBUG = false  -- /run NSQC4.QuestsDfGrid.DEBUG = true

    local function dbg(...)
        if Grid.DEBUG then print("|cff00ffff[GRID]|r", ...) end
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    -- ========================================================================
    -- Хранилище клеток
    -- ========================================================================
    local function EnsureGrid()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.grid = nsDbc4.quests.df.grid or {}
        local g = nsDbc4.quests.df.grid
        for i = 1, TOTAL do
            if not g[i] then
                g[i] = { terrain = nil, entityName = nil, hardness = nil }
            end
        end
        return g
    end

    -- ========================================================================
    -- Реестр существ по имени
    -- ========================================================================
    local function EnsureByName()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.entities = nsDbc4.quests.df.entities or {}
        nsDbc4.quests.df.entities.byName = nsDbc4.quests.df.entities.byName or {}
        return nsDbc4.quests.df.entities.byName
    end

    local function EnsureSelections()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.selections = nsDbc4.quests.df.selections or {}
        local s = nsDbc4.quests.df.selections
        s.terrain_types = s.terrain_types or {}
        s.entity   = s.entity   or {}
        s.reserved = s.reserved or {}
        return s
    end

    -- ========================================================================
    -- Базовое API
    -- ========================================================================
    function Grid:GetCell(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return nil end
        return EnsureGrid()[index]
    end

    function Grid:ClearCell(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return end
        EnsureGrid()[index] = { terrain = nil, entityName = nil, hardness = nil }
    end

    function Grid:ClearAll()
        local g = EnsureGrid()
        for i = 1, TOTAL do
            g[i] = { terrain = nil, entityName = nil, hardness = nil }
        end
        local s = EnsureSelections()
        s.terrain_types = {}
        s.entity = {}
        s.reserved = {}
        EnsureByName()  -- гарантируем, что таблица есть; чистит Entities:Clear
    end

    -- ========================================================================
    -- Террейн
    -- ========================================================================
    function Grid:SetTerrain(index, terrainId)
        local c = self:GetCell(index)
        if c then c.terrain = terrainId end
    end

    function Grid:GetTerrain(index)
        local c = self:GetCell(index)
        return c and c.terrain
    end

    -- ========================================================================
    -- Существо в клетке
    --   SetEntityName(index, name) — записать имя-ссылку
    --   GetEntityName(index)       — имя (строка)
    --   GetEntity(index)           — entity из byName (таблица, как раньше)
    --   SetEntity(index, entity)   — записать имя из entity (для совместимости)
    -- ========================================================================
    function Grid:SetEntityName(index, name)
        local c = self:GetCell(index)
        if c then c.entityName = name end
    end

    function Grid:GetEntityName(index)
        local c = self:GetCell(index)
        return c and c.entityName
    end

    function Grid:GetEntity(index)
        local c = self:GetCell(index)
        if not c or not c.entityName then return nil end
        return EnsureByName()[c.entityName]
    end

    -- Совместимость: старый вызов Grid:SetEntity(index, entity)
    function Grid:SetEntity(index, entity)
        local c = self:GetCell(index)
        if not c then return end
        if entity and entity.opts and entity.opts.name then
            c.entityName = entity.opts.name
        else
            c.entityName = nil
        end
    end

    -- ========================================================================
    -- Выделение
    -- ========================================================================
    function Grid:SetTerrainSelected(index, terrainId, isSelected)
        local s = EnsureSelections()

        if isSelected then
            for tid, list in pairs(s.terrain_types) do
                if tid ~= terrainId and list[index] then
                    list[index] = nil
                end
            end
            if not s.terrain_types[terrainId] then s.terrain_types[terrainId] = {} end
            s.terrain_types[terrainId][index] = true
        else
            if s.terrain_types[terrainId] then
                s.terrain_types[terrainId][index] = nil
            end
        end
        dbg("SetTerrainSelected", index, terrainId, isSelected and "ON" or "OFF")
    end

    function Grid:IsTerrainSelected(index, terrainId)
        local s = EnsureSelections()
        if terrainId then
            local list = s.terrain_types[terrainId]
            return (list and list[index]) and true or false
        end
        for tid, list in pairs(s.terrain_types) do
            if list[index] then return true, tid end
        end
        return false
    end

    function Grid:SetSelected(layer, id, isSelected)
        local s = EnsureSelections()
        if not s[layer] then s[layer] = {} end
        if isSelected then s[layer][id] = true else s[layer][id] = nil end
    end

    function Grid:IsSelected(layer, id)
        local s = EnsureSelections()
        return s[layer] and s[layer][id] == true
    end

    function Grid:ClearSelection(layer, id)
        if layer == "terrain" then
            local s = EnsureSelections()
            for terrainId, list in pairs(s.terrain_types) do
                if list[id] then list[id] = nil end
            end
        else
            self:SetSelected(layer, id, false)
        end
    end

    function Grid:ClearAllSelections(layer)
        local s = EnsureSelections()
        if layer == "terrain" then
            s.terrain_types = {}
        elseif layer == "reserved" then
            s.reserved = {}
        elseif layer then
            s[layer] = {}
        else
            s.terrain_types = {}
            s.entity = {}
            s.reserved = {}
        end
    end

    -- ========================================================================
    -- Резерв (по entityName или id — как передали)
    -- ========================================================================
    function Grid:Reserve(index, who)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end
        local s = EnsureSelections()
        s.reserved = s.reserved or {}
        s.reserved[index] = who
        dbg("Reserve", index, tostring(who))
        return true
    end

    function Grid:Release(index, who)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end
        local s = EnsureSelections()
        s.reserved = s.reserved or {}
        if not who or s.reserved[index] == who then
            s.reserved[index] = nil
        end
        dbg("Release", index, tostring(who))
        return true
    end

    function Grid:IsReserved(index, who)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end
        local s = EnsureSelections()
        s.reserved = s.reserved or {}
        local r = s.reserved[index]
        if not r then return false end
        if who and r == who then return false end
        return true
    end

    function Grid:GetReservedBy(index)
        local s = EnsureSelections()
        s.reserved = s.reserved or {}
        return s.reserved[tonumber(index) or 0]
    end

    -- ========================================================================
    -- Проходимость
    -- ========================================================================
    function Grid:IsPassable(index)
        local c = self:GetCell(index)
        if not c then return false end
        return not c.entityName
    end

    -- ========================================================================
    -- Отладка
    -- ========================================================================
    function Grid:DumpSelections()
        local s = EnsureSelections()
        local total = 0
        for tid, list in pairs(s.terrain_types) do
            local cnt = 0
            local cells = {}
            for idx in pairs(list) do
                cnt = cnt + 1
                cells[#cells + 1] = tostring(idx)
            end
            total = total + cnt
            print(string.format("[GRID] terrain=%s cells=%d  [%s]", tid, cnt, table.concat(cells, ",")))
        end
        print("[GRID] total selected terrain cells:", total)

        local rcnt = 0
        local rcells = {}
        for idx, who in pairs(s.reserved) do
            rcnt = rcnt + 1
            rcells[#rcells + 1] = tostring(idx) .. "=" .. tostring(who)
        end
        print(string.format("[GRID] reserved=%d  [%s]", rcnt, table.concat(rcells, " ")))
    end

    print("|cff00ffff[GRID-DEBUG]|r Единый реестр клеток инициализирован (byName).")

end)

    local function EnsureCreatures()
        local Entities = NSQC4.QuestsDfEntities
        local Grid     = NSQC4.QuestsDfGrid
        if not Entities or not Grid then return end

        local byName = Entities:EnsureByName()
        local hasGoblin, hasDwarf = false, false
        for _, e in pairs(byName) do
            if e.opts then
                local k = e.opts.kind or e.kind
                if k == "goblin" then hasGoblin = true end
                if k == "dwarf"  then hasDwarf  = true end
            end
        end

        -- Свободные клетки
        local free = {}
        for i = 1, 100 do
            if not Grid:GetEntityName(i) then
                free[#free + 1] = i
            end
        end

        local function pickFree()
            if #free == 0 then return nil end
            local p = math.random(#free)
            return table.remove(free, p)
        end

        local Goblin = NSQC4.QuestsDfGoblin
        local Dwarf  = NSQC4.QuestsDfDwarf

        if not hasGoblin and Goblin and Goblin.Spawn then
            local idx = pickFree()
            if idx then Goblin:Spawn(idx) end
        end
        if not hasDwarf and Dwarf and Dwarf.Spawn then
            local idx = pickFree()
            if idx then Dwarf:Spawn(idx) end
        end
    end

        local types = Terrain:GetTypes() or {}
        local typeIds = {}
        for id in pairs(types) do typeIds[#typeIds + 1] = id end
        table.sort(typeIds)
        if #typeIds == 0 then
            print("|cffff0000[GEN]|r ОШИБКА: нет зарегистрированных типов — генерация невозможна")
            return false
        end
        print("|cffff0000[GEN]|r Зарегистрированные типы: " .. table.concat(typeIds, ", "))

        Terrain:Clear()
        if NSQC4.QuestsDfEntities then NSQC4.QuestsDfEntities:Clear() end

        local pool = { "grass", "stone", "tree", "earth" }
        for i = 1, 100 do
            Terrain:SetTerrain(i, pool[math.random(1, #pool)])
        end
        if Terrain.Render then Terrain:Render() end

-- ============================================================================
-- NSQC4 / Quests / df / Professions / Digger
-- Профессия «Диггер». Копает землю (earth).
--
-- Профессия — самостоятельная единица. Знает:
--   - к каким объектам применяется (terrain)
--   - каким параметром существа ускоряется (stat)
--   - как считать клики под конкретный объект (clicks)
--   - что даёт по завершении (work.yields)
--
-- Формула кликов — ВНУТРИ профессии. Другие профессии могут считать иначе.
-- Позже сюда добавим скорость прокачки (зависит от интеллекта и т.п.).
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local P = NSQC4.QuestsDfProfessionsApi
    if not P then
        print("|cffff0000[NSQC4]|r Digger: Professions/Init.lua не загружен.")
        return
    end

    -- Веса вклада в снижение кликов (в сумме 0.999).
    local PROF_WEIGHT = 0.666
    local STR_WEIGHT  = 0.333

    -- ========================================================================
    -- Формула кликов Диггера.
    --   hardness — базовое HP объекта (из Terrain/Data)
    --   prof_pct — 0..1 (насколько прокачана профессия)
    --   str_pct  — 0..1 (сила существа / максимум)
    -- Возвращает целое ≥ 1.
    -- ========================================================================
    local function clicks(hardness, prof_pct, str_pct)
        hardness = tonumber(hardness) or 0
        if hardness <= 0 then return 1 end

        prof_pct = tonumber(prof_pct) or 0
        str_pct  = tonumber(str_pct)  or 0
        if prof_pct < 0 then prof_pct = 0 elseif prof_pct > 1 then prof_pct = 1 end
        if str_pct  < 0 then str_pct  = 0 elseif str_pct  > 1 then str_pct  = 1 end

        local factor = 1 - PROF_WEIGHT * prof_pct - STR_WEIGHT * str_pct
        if factor < 0 then factor = 0 end

        local n = math.floor(hardness * factor + 0.5)
        if n < 1 then n = 1 end
        return n
    end

    -- ========================================================================
    -- Регистрация профессии.
    -- ========================================================================
    P:Register("digger", {
        name    = "Диггер",
        desc    = "Копает землю, добывает глину.",
        terrain = "earth",          -- к какому объекту применяется
        stat    = "strength",       -- какой параметр ускоряет
        clicks  = clicks,           -- формула кликов (внутри профессии)
        work    = { verb = "копает", duration = 1, yields = "clay" },
    })

end)

-- ============================================================================
-- NSQC4 / Quests / df / Professions / Init
-- API реестра профессий.
--
-- Ничего не знает о конкретных профессиях. Только:
--   - хранит зарегистрированные def по id
--   - отдаёт их по запросу (по id, по имени, все, по terrain)
--
-- Каждая профессия — отдельный файл (df/Professions/<Prof>.lua),
-- который при загрузке зовёт Register(id, def).
--
-- Контракт def:
--   name    = "Диггер"                    — русское имя (для known_skills)
--   desc    = "..."                       — описание
--   terrain = "earth"                     — к какому объекту применяется
--   stat    = "strength"                  — какой параметр существа ускоряет
--   clicks  = function(hardness, prof_pct, str_pct) -> int
--                                          — формула кликов (внутри профессии)
--   work    = { verb, duration, yields }  — что делает / что даёт
--
-- API (namespace: NSQC4.QuestsDfProfessionsApi):
--   Register(id, def)             — зарегистрировать
--   Get(id)                       — def | nil
--   GetByName(name)               — def | nil (по русскому имени)
--   GetByTerrain(terrainId)       — первый def с этим terrain | nil
--   GetAll()                      — массив def
--   Count()                       — число
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local P = {}
    NSQC4.QuestsDfProfessionsApi = P

    local byId  = {}
    local order = {}

    -- ========================================================================
    -- Регистрация профессии.
    --   id  — "digger" | "stonecutter" | ...
    --   def — см. контракт в шапке файла
    -- ========================================================================
    function P:Register(id, def)
        if type(id) ~= "string" or id == "" then return false end
        if type(def) ~= "table" then return false end

        if not byId[id] then
            order[#order + 1] = id
        end

        def.id = id
        byId[id] = def
        return true
    end

    -- ========================================================================
    -- Получить по id.
    -- ========================================================================
    function P:Get(id)
        if type(id) ~= "string" or id == "" then return nil end
        return byId[id]
    end

    -- ========================================================================
    -- Получить по имени (opts.known_skills хранит имена).
    -- ========================================================================
    function P:GetByName(name)
        if type(name) ~= "string" or name == "" then return nil end
        for _, id in ipairs(order) do
            local def = byId[id]
            if def and def.name == name then return def end
        end
        return nil
    end

    -- ========================================================================
    -- Получить первую профессию, применимую к данному terrain.
    -- ========================================================================
    function P:GetByTerrain(terrainId)
        if type(terrainId) ~= "string" or terrainId == "" then return nil end
        for _, id in ipairs(order) do
            local def = byId[id]
            if def and def.terrain == terrainId then return def end
        end
        return nil
    end

    -- ========================================================================
    -- Все зарегистрированные (в порядке регистрации).
    -- ========================================================================
    function P:GetAll()
        local list = {}
        for _, id in ipairs(order) do
            if byId[id] then list[#list + 1] = byId[id] end
        end
        return list
    end

    -- ========================================================================
    -- Сколько зарегистрировано.
    -- ========================================================================
    function P:Count()
        return #order
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Professions / Resolve
-- Перевод данных существа в числа, которые ждёт формула профессии.
--
-- Формула любой профессии принимает:
--   clicks(hardness, prof_pct, str_pct)
--
-- Resolve готовит два числа:
--   prof_pct — насколько прокачана «своя» профессия для данного terrain (0..1)
--   str_pct  — сила существа относительно максимума (0..1)
--
-- Откуда берёт:
--   prof_pct:
--     opts.known_skills — таблица [имя] = уровень (1..100)
--     → ProfessionsApi:GetByName(name) → def.terrain == terrainId
--     → уровень / 100 = 0..1
--   str_pct:
--     opts.strength / opts.max_strength (обрезать до 0..1)
--
-- Ничего не знает про Grid, движение, работу, клики.
-- Только entity + terrainId → два числа.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Resolve = {}
    NSQC4.QuestsDfProfessionsResolve = Resolve

    -- ========================================================================
    -- prof_pct: 0..1.
    --   0 — нет подходящей профессии для этого terrain (или уровень 0).
    -- ========================================================================
    function Resolve:ProfPct(entity, terrainId)
        if not entity or not entity.opts then return 0 end
        if type(terrainId) ~= "string" or terrainId == "" then return 0 end

        local opts = entity.opts
        local ProfApi = NSQC4.QuestsDfProfessionsApi
        if not ProfApi or type(opts.known_skills) ~= "table" then return 0 end

        -- Ищем «свою» профессию: имя → def → def.terrain == terrainId.
        -- Уровень (значение) — 1..100, переводим в 0..1.
        for name, lvl in pairs(opts.known_skills) do
            local def = ProfApi:GetByName(name)
            if def and def.terrain == terrainId then
                local pct = (tonumber(lvl) or 0) / 100
                if pct < 0 then pct = 0 elseif pct > 1 then pct = 1 end
                return pct
            end
        end

        return 0
    end

    -- ========================================================================
    -- str_pct: 0..1.
    --   0 — силы нет или max_strength не задан.
    -- ========================================================================
    function Resolve:StrPct(entity)
        if not entity or not entity.opts then return 0 end
        local opts = entity.opts

        local str = tonumber(opts.strength) or 0
        local max = tonumber(opts.max_strength) or 0
        if max <= 0 then return 0 end

        local pct = str / max
        if pct < 0 then pct = 0 elseif pct > 1 then pct = 1 end
        return pct
    end

    -- ========================================================================
    -- Удобный фасад: оба числа сразу.
    --   → prof_pct, str_pct
    -- ========================================================================
    function Resolve:Both(entity, terrainId)
        return self:ProfPct(entity, terrainId), self:StrPct(entity)
    end

end)

    --   prof_pct — 0..1 (насколько прокачана профессия)
    --   str_pct  — 0..1 (сила существа / максимум)
    -- Возвращает целое ≥ 1.
    -- ========================================================================
    local function clicks(hardness, prof_pct, str_pct)
        hardness = tonumber(hardness) or 0
        if hardness <= 0 then return 1 end

        prof_pct = tonumber(prof_pct) or 0
        str_pct  = tonumber(str_pct)  or 0
        if prof_pct < 0 then prof_pct = 0 elseif prof_pct > 1 then prof_pct = 1 end
        if str_pct  < 0 then str_pct  = 0 elseif str_pct  > 1 then str_pct  = 1 end

        local factor = 1 - PROF_WEIGHT * prof_pct - STR_WEIGHT * str_pct
        if factor < 0 then factor = 0 end

        local n = math.floor(hardness * factor + 0.5)
        if n < 1 then n = 1 end
        return n
    end

    -- ========================================================================
    -- Регистрация профессии.
    -- ========================================================================
    P:Register("stonecutter", {
        name    = "Камнерез",
        desc    = "Обрабатывает камень для построек.",
        terrain = "stone",          -- к какому объекту применяется
        stat    = "strength",       -- какой параметр ускоряет
        clicks  = clicks,           -- формула кликов (внутри профессии)
        work    = { verb = "режет", duration = 1, yields = "stone_block" },
    })

end)

-- ============================================================================
-- NSQC4 / Quests / df / Terrain / Data / Init
-- Реестр базовых характеристик типов ландшафта.
--
-- Каждый тип — отдельный файл в df/Terrain/Data/:
--   Earth.lua, Grass.lua, Stone.lua, Tree.lua, Pool.lua
--
-- Файл типа зовёт:
--   NSQC4.QuestsDfTerrainData:Register("earth", {
--       hardness = 1000,
--       texture  = "Interface\\AddOns\\NSQC4\\Media\\Images\\00z",
--       textSize = 9,
--       color    = { 0.85, 0.85, 0.85 },
--   })
--
-- API:
--   Data:Register(id, def)  — зарегистрировать
--   Data:Get(id)            — def | nil
--   Data:GetAll()           — таблица id → def
--   Data:Count()            — число
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Data = {}
    NSQC4.QuestsDfTerrainData = Data

    local byId = {}
    local order = {}

    function Data:Register(id, def)
        if type(id) ~= "string" or id == "" then return false end
        if type(def) ~= "table" then return false end

        if not byId[id] then
            order[#order + 1] = id
        end

        def.id = id
        byId[id] = def
        return true
    end

    function Data:Get(id)
        if type(id) ~= "string" or id == "" then return nil end
        return byId[id]
    end

    function Data:GetAll()
        return byId
    end

    function Data:Count()
        return #order
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Terrain / Init
-- Ландшафт на поле 10×10. Точка входа подмодуля.
--
-- Контракт (реализуется в Storage/UI/Logic):
--   Terrain:EnsureStorage()                  -- nsDbc4.quests.df.terrain
--   Terrain:IsInitialized() / SetInitialized(v)
--   Terrain:GetAt(index)                     -- terrainId в ячейке (или nil)
--   Terrain:SetAt(index, terrainId, opts?)   -- задать ландшафт
--   Terrain:ClearAt(index)                   -- убрать ландшафт из ячейки
--   Terrain:Clear()                          -- сбросить карту
--   Terrain:GetTypes()                       -- список известных типов
--   Terrain:RegisterType(id, opts)           -- объявить тип (напр. "grass")
--   Terrain:Render() / RenderCell(index)     -- перерисовать
--
-- Правило: работа с полем — только через API QuestsField
-- (SetCellTexture/SetCellText/SetAnchorTexture/SetCellOnClick/ClearCell).
-- Прямой доступ к btn.* и nsDbc4.quests.field запрещён.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsFieldClass
    if not QuestsField then
        print("|cffff0000[NSQC4]|r Terrain: ядро поля не загружено.")
        return
    end

    -- ========================================================================
    -- Namespace подмодуля
    -- ========================================================================
    local Terrain = {}
    Terrain.__index = Terrain

    NSQC4.QuestsDfTerrain = Terrain
    NSQC4.QuestsDfModules = NSQC4.QuestsDfModules or {}
    NSQC4.QuestsDfModules["terrain"] = NSQC4.QuestsDfModules["terrain"] or {
        label   = "Ландшафт",
        order   = 200,
        enabled = true,
    }
    NSQC4.QuestsDfModules["terrain"].api = Terrain

    -- ========================================================================
    -- Ленивая проверка готовности Storage/UI/Logic
    -- ========================================================================
    local function Ready()
        return Terrain.EnsureStorage ~= nil
           and Terrain.SetAt       ~= nil
           and Terrain.Render      ~= nil
    end

    -- ========================================================================
    -- Инициализация подмодуля (идемпотентно)
    -- ========================================================================
    function Terrain:Init()
        if not Ready() then
            return false
        end

        self:EnsureStorage()
        if self:IsInitialized() then
            return false
        end

        self:SetInitialized(true)
        return true
    end

    -- ========================================================================
    -- Регистрация в df/Init.lua
    -- ========================================================================
    NSQC4.QuestsDfModules["terrain"].init = function()
        return NSQC4.QuestsDfTerrain:Init()
    end

    -- ========================================================================
    -- Слэш-команды
    -- ========================================================================
    SLASH_NSDF_TERRAIN1 = "/ns_df_terrain"
    SlashCmdList["NSDF_TERRAIN"] = function(msg)
        local cmd = (msg or ""):lower():match("^(%S*)")
        local T = NSQC4.QuestsDfTerrain

        if cmd == "" or cmd == "info" then
            print("|cff00ff00[NSQC4]|r df/terrain — ландшафт")
            if not Ready() then
                print("|cffffaa00  Storage/UI/Logic ещё не загружены.|r")
                return
            end
            local st = T:EnsureStorage()
            local n = 0
            if st and st.map then
                for _ in pairs(st.map) do n = n + 1 end
            end
            print("  initialized:", T:IsInitialized())
            print("  закрашенных клеток:", n)
            local types = T:GetTypes() or {}
            local tcount = 0
            for _ in pairs(types) do tcount = tcount + 1 end
            print("  объявленных типов:", tcount)
            print("|cff808080Команды: /ns_df_terrain init | clear|r")

        elseif cmd == "init" then
            local ok = T:Init()
            print("|cff00ff00[NSQC4]|r terrain:",
                  ok and "инициализирован."
                     or  "уже был инициализирован (или Storage не готов).")

        elseif cmd == "clear" then
            if not Ready() then
                print("|cffffaa00[NSQC4]|r terrain: Storage ещё не загружен.")
                return
            end
            T:Clear()
            print("|cffff8800[NSQC4]|r terrain: карта очищена.")

        else
            print("|cffff0000[NSQC4]|r terrain: неизвестная команда '" .. cmd .. "'")
        end
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Terrain / Logic / Init
-- Менеджер генерации и слэш-команд для ландшафта.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Terrain = NSQC4.QuestsDfTerrain
    if not Terrain then return end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local COLS, ROWS = CONST.COLS, CONST.ROWS
    local TOTAL = COLS * ROWS

    -- ========================================================================
    -- УСТАНОВКА ТИПА
    -- ========================================================================
    function Terrain:SetTerrain(index, terrainId, opts)
        -- Просто вызываем SetAt, типы уже зарегистрированы в Storage
        return self:SetAt(index, terrainId, opts)
    end

    function Terrain:ClearTerrain(index)
        return self:ClearAt(index)
    end

    -- ========================================================================
    -- ГЕНЕРАЦИЯ (Пятна и случайное заполнение)
    -- ========================================================================
    function Terrain:SeedDemo(kind, count)
        kind = kind or "grass"
        count = math.min(tonumber(count) or 20, TOTAL)

        local free = {}
        for i = 1, TOTAL do
            if not self:GetAt(i) then
                free[#free + 1] = i
            end
        end

        if #free == 0 then return 0 end

        local made = 0
        for _ = 1, count do
            if #free == 0 then break end
            local pick = math.random(#free)
            local idx = table.remove(free, pick)
            self:SetTerrain(idx, kind)
            made = made + 1
        end

        if self.Render then self:Render() end
        return made
    end

    function Terrain:SeedPatch(kind, centerIndex, radius)
        kind = kind or "grass"
        centerIndex = tonumber(centerIndex) or 0
        radius = math.max(0, math.floor(tonumber(radius) or 1))

        if centerIndex < 1 or centerIndex > TOTAL then return 0 end

        local function RC(index)
            local i = index - 1
            return i % COLS, math.floor(i / COLS)
        end

        local cCol, cRow = RC(centerIndex)
        local made = 0

        for row = cRow - radius, cRow + radius do
            for col = cCol - radius, cCol + radius do
                if col >= 0 and col < COLS and row >= 0 and row < ROWS then
                    local idx = row * COLS + col + 1
                    self:SetTerrain(idx, kind)
                    made = made + 1
                end
            end
        end

        if self.Render then self:Render() end
        return made
    end

    -- ========================================================================
    -- СЛЭШ-КОМАНДЫ
    -- ========================================================================
    SLASH_NSDF_TERRAIN_LOGIC1 = "/ns_df_terr"
    SlashCmdList["NSDF_TERRAIN_LOGIC"] = function(msg)
        local T = NSQC4.QuestsDfTerrain
        if not T then return end

        local cmd, a1, a2, a3 = (msg or ""):match("^(%S*)%s*(%S*)%s*(%S*)%s*(%S*)")
        cmd = (cmd or ""):lower()

        if cmd == "" or cmd == "help" then
            print("|cff00ff00[NSQC4]|r df/terrain — логика")
            print("  /ns_df_terr types                — список типов")
            print("  /ns_df_terr set <index> <id>     — поставить тип")
            print("  /ns_df_terr clear <index>        — снять тип")
            print("  /ns_df_terr seed <id> [count]    — залить случайные")
            print("  /ns_df_terr patch <id> <center> [radius] — пятно")

        elseif cmd == "types" then
            local types = T:GetTypes() or {}
            local names = {}
            for id in pairs(types) do names[#names + 1] = id end
            table.sort(names)
            print("|cff00ff00[NSQC4]|r terrain types:", #names)
            for _, id in ipairs(names) do
                local o = types[id]
                print(string.format("  %s → %s", id, tostring(o and o.texture)))
            end

        elseif cmd == "set" then
            local index = tonumber(a1)
            local id = a2
            if index and id then T:SetTerrain(index, id) end

        elseif cmd == "clear" then
            local index = tonumber(a1)
            if index then T:ClearTerrain(index) end

        elseif cmd == "seed" then
            local id = a1 ~= "" and a1 or "grass"
            local n = tonumber(a2) or 20
            T:SeedDemo(id, n)

        elseif cmd == "patch" then
            local id = a1 ~= "" and a1 or "grass"
            local center = tonumber(a2)
            local radius = tonumber(a3) or 1
            if center then T:SeedPatch(id, center, radius) end
        end
    end

end)

 FILE: df/Terrain/Storage.lua

-- ============================================================================
-- NSQC4 / Quests / df / Terrain / Storage
-- Хранилище ландшафта: адаптер к общему Grid + точка входа рендера.
--
-- Terrain:Clear() теперь ещё и чистит Grid:ClearAllSelections("terrain"),
-- чтобы после generate/restart не оставались старые выделения
-- (grass[56], stone[57] и т.п.). Иначе существа «залипают» на них.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local Terrain = NSQC4.QuestsDfTerrain
    if not Terrain then return end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    local Grid = NSQC4.QuestsDfGrid
    if not Grid then return end

    local function UIApi()
        return NSQC4.QuestsDfTerrainUIApi
    end

    local function TerrainData()
        return NSQC4.QuestsDfTerrainData
    end

    local function RenderCell(index)
        local api = UIApi()
        if api and api.RenderCellFromStorage then
            api:RenderCellFromStorage(index)
        end
    end

    local function RenderAll()
        local api = UIApi()
        if api and api.RenderAll then
            api:RenderAll()
        end
    end

    local function ClearCellClicks(index)
        local QuestsField = NSQC4.QuestsField
        if not QuestsField then return end
        if QuestsField.ClearCellMouse then
            QuestsField:ClearCellMouse(index, "LeftButton")
            QuestsField:ClearCellMouse(index, "RightButton")
        end
    end

    local function EnsureTable()
        nsDbc4 = nsDbc4 or {}
        nsDbc4.quests = nsDbc4.quests or {}
        nsDbc4.quests.df = nsDbc4.quests.df or {}
        nsDbc4.quests.df.terrain = nsDbc4.quests.df.terrain or {}

        local s = nsDbc4.quests.df.terrain
        s.types = s.types or {}
        s.initialized = s.initialized and true or false
        return s
    end

    function Terrain:EnsureStorage() return EnsureTable() end
    function Terrain:IsInitialized() return EnsureTable().initialized and true or false end
    function Terrain:SetInitialized(v) EnsureTable().initialized = v and true or false end

    -- ========================================================================
    -- Реестр типов
    -- ========================================================================
    function Terrain:RegisterType(id, opts)
        if type(id) ~= "string" or id == "" then return nil end
        local s = EnsureTable()
        s.types[id] = opts or {}
        return s.types[id]
    end

    function Terrain:GetType(id)
        local s = EnsureTable()
        return s.types[id]
    end

    function Terrain:GetTypes()
        return EnsureTable().types
    end

    -- ========================================================================
    -- Карта
    -- ========================================================================
    function Terrain:GetAt(index)
        return Grid:GetTerrain(index)
    end

    function Terrain:SetAt(index, terrainId, opts)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end
        if type(terrainId) ~= "string" or terrainId == "" then return false end

        local s = EnsureTable()
        if opts and not s.types[terrainId] then
            s.types[terrainId] = opts
        end

        Grid:SetTerrain(index, terrainId)

        local Data = TerrainData()
        local def = Data and Data:Get(terrainId)
        if def and def.hardness then
            Grid:SetHardness(index, def.hardness)
        else
            Grid:ClearHardness(index)
        end

        RenderCell(index)
        return true
    end

    function Terrain:ClearAt(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end

        local current = Grid:GetTerrain(index)
        if not current then return false end

        if self:IsSelected(index) then
            self:ToggleSelect(index)
        end

        Grid:SetTerrain(index, nil)
        Grid:ClearHardness(index)
        ClearCellClicks(index)
        RenderCell(index)
        return true
    end

    function Terrain:Clear()
        for i = 1, TOTAL do
            if self:IsSelected(i) then
                self:ToggleSelect(i)
            end
            Grid:SetTerrain(i, nil)
            Grid:ClearHardness(i)
            ClearCellClicks(i)
        end
        RenderAll()

        -- Чистим старые выделения, чтобы после generate/restart
        -- существа не «залипали» на чужих terrain в клетке.
        if Grid.ClearAllSelections then
            Grid:ClearAllSelections("terrain")
        end
    end

    function Terrain:Count()
        local count = 0
        for i = 1, TOTAL do
            if Grid:GetTerrain(i) then count = count + 1 end
        end
        return count
    end

    -- ========================================================================
    -- Выделение
    -- ========================================================================
    function Terrain:ToggleSelect(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end

        local terrainId = Grid:GetTerrain(index)
        if not terrainId then return false end

        local isSelected = Grid:IsTerrainSelected(index)

        if isSelected then
            Grid:SetTerrainSelected(index, terrainId, false)
        else
            Grid:SetTerrainSelected(index, terrainId, true)
        end

        RenderCell(index)
        return true
    end

    function Terrain:IsSelected(index)
        return Grid:IsTerrainSelected(index)
    end

    function Terrain:ClearSelection()
        Grid:ClearAllSelections("terrain")
        RenderAll()
    end

    -- ========================================================================
    -- Рендер-вход
    -- ========================================================================
    function Terrain:RenderCell(index)
        RenderCell(index)
    end

    function Terrain:Render()
        RenderAll()
    end

end)

-- ============================================================================
-- NSQC4 / Quests / df / Terrain / UI / Api
-- Ландшафт — владелец клетки.
--
-- opts._landscape для тултипа: { index, terrainId, hardness, maxHardness }.
-- hardness в тултипе читается заново (Tooltip.Build → Grid:GetHardness),
-- поэтому здесь важно передать index и terrainId.
-- ============================================================================

NSQC4.RegisterModule("quests", function()

    local QuestsField = NSQC4.QuestsField
    if not QuestsField then
        print("|cffff0000[NSQC4]|r Terrain/UI/Api: ядро поля не загружено.")
        return
    end

    local CONST = NSQC4.QuestsFieldConst
    if not CONST then return end
    local TOTAL = CONST.COLS * CONST.ROWS

    local UIApi = {}
    NSQC4.QuestsDfTerrainUIApi = UIApi

    UIApi.DEBUG = false

    local function dbg(...)
        if UIApi.DEBUG then print("|cff00ffff[TUIA]|r", ...) end
    end

    local DEFAULTS = {
        text     = nil,
        color    = { 0.85, 0.85, 0.85 },
        textSize = 9,
        anchors  = {},
    }

    local function MergeOpts(opts)
        opts = opts or {}
        return {
            texture  = opts.texture,
            text     = opts.text ~= nil and opts.text or DEFAULTS.text,
            color    = opts.color    or DEFAULTS.color,
            textSize = opts.textSize or DEFAULTS.textSize,
            anchors  = opts.anchors  or DEFAULTS.anchors,
        }
    end

    local function ApplyTint(index, rgb)
        local cell = QuestsField.cells and QuestsField.cells[index]
        if not cell or not cell.texture then return end
        if rgb then
            cell.texture:SetVertexColor(rgb[1], rgb[2], rgb[3], rgb[4] or 1)
        else
            cell.texture:SetVertexColor(1, 1, 1, 1)
        end
    end

    local function ResolveTooltip(entity)
        if not entity or not entity.opts then return nil end
        local Entities = NSQC4.QuestsDfEntities
        local kind = entity.kind or entity.opts.kind
        if Entities and Entities.GetTooltip and kind then
            local builder = Entities:GetTooltip(kind)
            if builder then
                local ok, fn = pcall(builder, entity.opts)
                if ok and type(fn) == "function" then return fn end
            end
        end
        return nil
    end

    function UIApi:Render(index, opts)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end

        if not opts then
            QuestsField:SetCellTexture(index, nil)
            QuestsField:ClearCellText(index)
            QuestsField:ClearAnchorTextures(index)
            ApplyTint(index, nil)
            return true
        end

        local v = MergeOpts(opts)

        QuestsField:SetCellTexture(index, v.texture)

        if v.text and v.text ~= "" then
            local c = v.color or { 1, 1, 1 }
            QuestsField:SetCellText(index, v.text, nil, c[1], c[2], c[3], v.textSize)
        else
            QuestsField:ClearCellText(index)
        end

        QuestsField:ClearAnchorTextures(index)
        if v.anchors then
            for pos, tex in pairs(v.anchors) do
                if tex then
                    QuestsField:SetAnchorTexture(index, pos, tex)
                end
            end
        end

        ApplyTint(index, opts.highlight)
        return true
    end

    function UIApi:RenderCellFromStorage(index)
        index = tonumber(index) or 0
        if index < 1 or index > TOTAL then return false end

        local Terrain = NSQC4.QuestsDfTerrain
        if not Terrain then return false end

        local Grid = NSQC4.QuestsDfGrid
        local Data = NSQC4.QuestsDfTerrainData

        local terrainId = Terrain:GetAt(index)

        if not terrainId then
            self:Render(index, nil)
        else
            local data = Data and Data:Get(terrainId) or {}
            local logic = Terrain:GetType(terrainId) or {}

            local selected = false
            if Grid and Grid.IsTerrainSelected then
                selected = Grid:IsTerrainSelected(index, terrainId) and true or false
            end
            dbg("cell", index, "terrain", terrainId, "selected", selected and 1 or 0)

            self:Render(index, {
                texture   = data.texture   or logic.texture,
                text      = data.text      or logic.text,
                color     = data.color     or logic.color,
                textSize  = data.textSize  or logic.textSize,
                anchors   = data.anchors   or logic.anchors,
                highlight = selected and { 1.0, 0.8, 0.0, 1.0 } or nil,
            })
        end

        local EntitiesApi = NSQC4.QuestsDfEntitiesUIApi
        local entity = Grid and Grid:GetEntity(index)

        if entity and entity.opts and EntitiesApi and EntitiesApi.Render then
            EntitiesApi:Render(index, {
                icon     = entity.opts.icon or entity.opts.texture,
                anchor   = entity.opts.anchor,
                expanded = entity.opts.expanded,
                alpha    = entity.opts.alpha,
            })
        elseif EntitiesApi and EntitiesApi.Render then
            EntitiesApi:Render(index, nil)
        end

        -- Тултип: отдаём index и terrainId, hardness читается в OnEnter заново
        if entity and entity.opts and not entity.opts.sleeping then
            entity.opts._landscape = {
                index     = index,
                terrainId = terrainId,
            }

            local tooltipFn = ResolveTooltip(entity)
            if tooltipFn then
                QuestsField:SetCellTooltip(index, tooltipFn)
            else
                QuestsField:ClearCellTooltip(index)
            end
        else
            QuestsField:ClearCellTooltip(index)
        end

        local Actions = NSQC4.QuestsDfTerrainActions
        local T = Actions and Actions.Types and terrainId and Actions.Types[terrainId]

        if T and T.Check and T.Check(index) then
            if T.LeftClickSelf then T.LeftClickSelf(index) end
            if T.RightClickEntities then T.RightClickEntities(index) end
        else
            QuestsField:SetCellOnMouseClick(index, "LeftButton",  nil)
            QuestsField:SetCellOnMouseClick(index, "RightButton", nil)
        end

        return true
    end

    function UIApi:RenderAll()
        for i = 1, TOTAL do
            self:RenderCellFromStorage(i)
        end
    end

end)
