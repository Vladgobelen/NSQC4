NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local A = {}
    NSQC4.Nowim.Aliases = A

    C.db.aliases = C.db.aliases or {}

    -- Получить "главный" ник. Если это твинк, вернет ник владельца.
    function A.GetMainName(nick)
        if not nick or nick == "" then return nil end
        local lowerNick = nick:lower()
        
        -- Проверяем, не является ли ник основным
        for mainName, alts in pairs(C.db.aliases) do
            if mainName:lower() == lowerNick then
                return mainName
            end
            -- Проверяем твинков
            for _, altName in ipairs(alts) do
                if altName:lower() == lowerNick then
                    return mainName -- Возвращаем основного (например, "ВасяПупкин")
                end
            end
        end
        
        return nick -- Если не найден в связках, возвращаем как есть
    end

    -- Добавить твинка к основному нику
    function A.AddAlt(mainName, altName)
        if not mainName or not altName then return end
        C.db.aliases[mainName] = C.db.aliases[mainName] or {}
        
        -- Проверяем, нет ли уже такого твинка
        for _, name in ipairs(C.db.aliases[mainName]) do
            if name:lower() == altName:lower() then return end
        end
        
        table.insert(C.db.aliases[mainName], altName)
    end
end)

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local S = NSQC4.Nowim.Storage
    local N = NSQC4.Nowim.Network
    local A = NSQC4.Nowim.Aliases

    local E = {}
    NSQC4.Nowim.ChatEngine = E

    C.db.unread = C.db.unread or {}
    C.db.openTabs = C.db.openTabs or {}
    C.db.savedScroll = C.db.savedScroll or {}

    function E.OpenDialog(key)
        if not key or key == "" then return end
        
        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }

        local isOpen = false
        for _, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then isOpen = true; break end
        end

        if not isOpen then
            table.insert(C.db.openTabs, key)
            if C.db.settings.autoScroll then
                C.db.savedScroll[key] = 999999 
            else
                C.db.savedScroll[key] = 0
            end
        end

        C.db.activeTab = key
        C.db.unread[key] = 0

        E.UpdateUI()
    end

    function E.CloseTab(key)
        for i, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then
                table.remove(C.db.openTabs, i)
                break
            end
        end

        if C.db.activeTab == key then
            C.db.activeTab = C.db.openTabs[#C.db.openTabs]
        end

        E.UpdateUI()
    end

    function E.SendMessage(text)
        local target = C.db.activeTab
        if not target then return end
        N.QueueMessage(target, text)
    end

    function E.OnIncomingMessage(sender, targetKey, text)
        if C.db.activeTab ~= targetKey then
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
        end
        E.UpdateUI()
    end

    function E.UpdateUI()
        local UI = NSQC4.Nowim.UI
        if not UI then return end

        -- ДОБАВЛЕНА ПРОВЕРКА НА NIL
        if UI.Tabs and UI.Tabs.Refresh then 
            UI.Tabs.Refresh() 
        end
        
        if UI.LeftPanel and UI.LeftPanel.Refresh then 
            UI.LeftPanel.Refresh() 
        end
        
        if C.db.activeTab and UI.ChatView and UI.ChatView.Refresh then
            UI.ChatView.Refresh()
        end
    end

    SLASH_NOWIM_CHAT1 = "/nowim"
    SlashCmdList["NOWIM_CHAT"] = function(msg)
        msg = (msg or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if msg ~= "" then
            E.OpenDialog(msg)
        end
        if NSQC4.Nowim.UI and NSQC4.Nowim.UI.MainFrame then
            NSQC4.Nowim.UI.MainFrame:ShowFrame()
        end
    end

end)

NSQC4.RegisterModule("nowim", function()

    local S = NSQC4.Nowim.Storage
    local C = NSQC4.Nowim.Config
    local myName = UnitName("player")

    local frame = CreateFrame("Frame")
    frame:RegisterEvent("CHAT_MSG_GUILD")
    frame:RegisterEvent("CHAT_MSG_OFFICER")
    frame:RegisterEvent("CHAT_MSG_RAID")
    frame:RegisterEvent("CHAT_MSG_RAID_LEADER")
    frame:RegisterEvent("CHAT_MSG_WHISPER")
    
    frame:SetScript("OnEvent", function(self, event, text, sender)
        if not text or text == "" then return end
        if text:sub(1, 5) == "NSMSG" then return end 

        local E = NSQC4.Nowim.ChatEngine
        if not E then return end

        local chatType = event:match("^CHAT_MSG_(.+)$")
        local targetKey = chatType
        
        -- Анти-дубль
        if sender == myName and NSQC4.Nowim._lastSentText == text and (GetTime() - (NSQC4.Nowim._lastSentTime or 0)) < 2 then
            return
        end

        -- Убираем первый ник (ретранслятор)
        local cleanText = text:gsub("^" .. sender .. ":%s*", "")

        -- Парсим AOClick
        local formattedText = cleanText
        if NSQC4.AOClick_ReplaceAll then
            formattedText = NSQC4.AOClick_ReplaceAll(cleanText)
        end

        if chatType == "WHISPER" then
            targetKey = sender
            E.OpenDialog(targetKey) 
        end

        -- Сохраняем
        S.AddMessage(targetKey, formattedText, false, nil)
        
        -- Обновляем UI
        if C.db.activeTab == targetKey then
            E.UpdateUI()
        else
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
            
            -- Безопасный вызов ChatView.Refresh, если он есть
            local UI = NSQC4.Nowim.UI
            if UI and UI.ChatView and UI.ChatView.Refresh then
                UI.ChatView.Refresh()
            end
            
            E.UpdateUI()
        end
    end)
end)

    local function ApplyBinding()
        local Nowim = NSQC4.Nowim
        if not Nowim or not Nowim.Config then return end
        local enabled = Nowim.Config.db.settings.interceptEnter

        if enabled then
            SetOverrideBindingClick(enterCatcher, true, "ENTER", "NSQC4_NowimEnterCatcher")
            SetOverrideBindingClick(enterCatcher, true, "NUMPADENTER", "NSQC4_NowimEnterCatcher")
        else
            ClearOverrideBindings(enterCatcher)
        end
    end

    NSQC4.Nowim = NSQC4.Nowim or {}
    NSQC4.Nowim.ApplyEnterBinding = ApplyBinding

    -- Ждём, когда всё загрузится, и вешаем бинд
    local f = CreateFrame("Frame")
    f:RegisterEvent("PLAYER_LOGIN")
    f:SetScript("OnEvent", function(self)
        self:UnregisterAllEvents()

            ApplyBinding()

    -- (перехват должен работать и при закрытом окне, если настройка включена)
    local function HookMainFrame()
        local UI = NSQC4.Nowim and NSQC4.Nowim.UI
        local mainFrame = UI and UI.MainFrame
        if not mainFrame or mainFrame._nowimEnterHooked then return end
        mainFrame._nowimEnterHooked = true

        local orig_Hide = mainFrame.Hide
        mainFrame.Hide = function(self)

            -- Бинд НЕ снимаем: если interceptEnter включён, Enter должен
            -- открывать окно даже когда оно скрыто.
            ApplyBinding()
        end

        local orig_Show = mainFrame.ShowFrame
        mainFrame.ShowFrame = function(self)
            if orig_Show then orig_Show(self) else self:Show() end
            ApplyBinding()
        end
    end

    if not HookMainFrame() then
        local g = CreateFrame("Frame")
        g:RegisterEvent("PLAYER_LOGIN")
        g:SetScript("OnEvent", function(self)
            if HookMainFrame() then self:UnregisterAllEvents() end
        end)
    end

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local S = NSQC4.Nowim.Storage

    local N = {}
    NSQC4.Nowim.Network = N

    local sendQueue = {}
    local isSending = false

    local function ProcessQueue()
        if isSending or #sendQueue == 0 then return end
        isSending = true

        local msgData = table.remove(sendQueue, 1)
        local target = msgData.target
        local text = U.TrimToBytes(msgData.text, C.MAX_BYTES)
        
        local isChannel = C.db.channels[target] ~= nil

        if target == UnitName("player") then
            S.AddMessage(target, text, true, UnitName("player"))
            N.UpdateUI(target)
        elseif isChannel then
            local chatType = C.db.channels[target].type
            
            -- Запоминаем, что мы только что отправили это в канал,
            -- чтобы ChatHook проигнорировал входящее эхо
            NSQC4.Nowim._lastSentText = text
            NSQC4.Nowim._lastSentTime = GetTime()
            NSQC4.Nowim._lastSentTarget = target
            
            if chatType == "GUILD" or chatType == "RAID" then
                SendChatMessage(text, chatType)
            end
            
            S.AddMessage(target, text, true, UnitName("player"))
        else
            SendAddonMessage(C.ADDON_PREFIX, string.format("FROM:%s|TEXT:%s", UnitName("player"), text), "WHISPER", target)
            S.AddMessage(target, text, true, UnitName("player"))
        end

        local timer = CreateFrame("Frame")
        local elapsed = 0
        timer:SetScript("OnUpdate", function(self, dt)
            elapsed = elapsed + dt
            if elapsed >= C.SEND_DELAY then
                self:SetScript("OnUpdate", nil)
                isSending = false
                ProcessQueue()
            end
        end)
    end

    function N.QueueMessage(target, text)
        if not target or not text or text == "" then return end
        table.insert(sendQueue, { target = target, text = text })
        ProcessQueue()
    end

    -- ========================================================================
    -- ПРИЁМ АДДОН-СООБЩЕНИЙ
    -- ========================================================================
    if type(RegisterAddonMessagePrefix) == "function" then
        RegisterAddonMessagePrefix(C.ADDON_PREFIX)
    end

    if NSQC4.ChatHandler and type(NSQC4.ChatHandler.Register) == "function" then
        NSQC4.ChatHandler:Register("ADDON:" .. string.lower(C.ADDON_PREFIX), {
            func = function(words, sender, text, channel, prefix)
                -- (без изменений, оставляем как было)
                if text:sub(1, 5) == "CHAN:" then
                    local chatType, fromNick, msgText = text:match("^CHAN:([^|]+)|FROM:([^|]+)|TEXT:(.+)$")
                    if chatType and msgText then
                        S.AddMessage(chatType, msgText, false, fromNick)
                        N.UpdateUI(chatType)
                    end
                else
                    local fromNick, msgText = text:match("^FROM:([^|]+)|TEXT:(.+)$")
                    if fromNick and msgText then
                        S.AddMessage(fromNick, msgText, false, fromNick)
                        N.UpdateUI(fromNick)
                        if C.db.settings.playSound and PlaySoundFile then
                            PlaySoundFile("Interface\\AddOns\\NSQC4\\Media\\Sounds\\m.ogg")
                        end
                    end
                end
            end,
            stopOnMatch = true,
        })
    end

    function N.UpdateUI(target)
        if NSQC4.Nowim.UI and NSQC4.Nowim.UI.ActiveTarget == target then
            if NSQC4.Nowim.UI.ChatView then NSQC4.Nowim.UI.ChatView.Refresh() end
        end
    end
end)

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local A = NSQC4.Nowim.Aliases

    local S = {}
    NSQC4.Nowim.Storage = S

    local function ResolveKey(nick)
        if not nick then return nil end
        if C.db.channels[nick] then return nick end 
        return A.GetMainName(nick)
    end

    function S.AddMessage(target, text, isMine, realSender)
        local key = ResolveKey(target)
        if not key or not text then return end
        
        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }
        local dialog = C.db.dialogs[key]
        
        -- Если realSender nil, значит имя уже есть в тексте (ретранслятор)
        local charTag = realSender or (isMine and UnitName("player") or nil)
        
        table.insert(dialog.messages, {
            text = text,
            isMine = isMine,
            time = U.GetTimeString(),
            char = charTag 
        })
        
        while #dialog.messages > C.MAX_HISTORY do
            table.remove(dialog.messages, 1)
        end
    end

    function S.GetMessages(key)
        local resolvedKey = ResolveKey(key)
        if not resolvedKey or not C.db.dialogs[resolvedKey] then return nil end
        return C.db.dialogs[resolvedKey].messages
    end

    function S.ClearHistory(key)
        local resolvedKey = ResolveKey(key)
        if resolvedKey and C.db.dialogs[resolvedKey] then
            C.db.dialogs[resolvedKey].messages = {}
        end
    end

    function S.GetContacts()
        local list = {}
        
        -- Добавляем ЛС (диалоги, которых нет в channels)
        for key in pairs(C.db.dialogs) do
            if not C.db.channels[key] then
                table.insert(list, { key = key, name = tostring(key), isChannel = false })
            end
        end
        
        -- Добавляем каналы
        for id, data in pairs(C.db.channels) do
            table.insert(list, { key = id, name = tostring(data.name or id), isChannel = true })
        end

        -- Безопасная сортировка: сравниваем только строки
        table.sort(list, function(a, b)
            local nameA = tostring(a.name or "")
            local nameB = tostring(b.name or "")
            if a.isChannel ~= b.isChannel then 
                return a.isChannel 
            end
            return nameA < nameB
        end)
        
        return list
    end

end)

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local S = NSQC4.Nowim.Storage
    local UI = NSQC4.Nowim.UI
    local V = UI.ChatView
    local myName = UnitName("player") or "Unknown"

    -- ========================================================================
    -- КОНСТАНТЫ ОФОРМЛЕНИЯ ПУЗЫРЕЙ
    -- ========================================================================
    local SHRINK_AMOUNT = 4  -- Сжатие фона (отрицательный зазор вокруг текста)
    local GAP           = 4   -- Вертикальный отступ между пузырями
    local PADDING_LEFT  = 2   -- Сдвиг чужих сообщений влево
    local PADDING_RIGHT = 2   -- Сдвиг своих сообщений вправо
    -- ========================================================================

    local measure = V.Content:CreateFontString(nil, "OVERLAY")
    measure:SetJustifyH("LEFT")
    measure:SetWordWrap(true)
    measure:Hide()

    function V.Refresh()
        local active = C.db.activeTab
        if not active then return end

        local messages = S.GetMessages(active)
        if not messages or #messages == 0 then
            V.Content:SetHeight(1)
            V.ScrollToBottom()
            return
        end

        for _, child in ipairs({V.Content:GetChildren()}) do
            if child ~= measure then
                child:Hide()
                child:SetParent(nil)
            end
        end

        local yOffset = 0
        local totalWidth = V.Scroll:GetWidth()
        if totalWidth < 50 then totalWidth = 300 end
        
        local maxBubbleWidth = totalWidth - 20
        V.Content:SetWidth(totalWidth)

        local fontPath = "Fonts\\FRIZQT__.TTF"
        local fontSize = C.db.settings.fontSize or 12

        measure:SetFont(fontPath, fontSize)
        measure:SetWordWrap(true)
        measure:SetNonSpaceWrap(true)

        for _, msg in ipairs(messages) do
            local timeStr = C.COLOR_TIME .. "[" .. msg.time .. "]|r "
            local senderColor = msg.isMine and C.COLOR_MY_MSG or C.COLOR_OTHER_MSG
            local senderPrefix = ""
            if msg.char then
                local senderName = msg.isMine and myName or msg.char
                senderPrefix = senderColor .. senderName .. ":|r "
            elseif msg.isMine then
                senderPrefix = senderColor .. myName .. ":|r "
            end

            local fullText = string.format("%s%s%s", timeStr, senderPrefix, msg.text)

            local maxTextW = maxBubbleWidth - (SHRINK_AMOUNT * 2)
            
            measure:SetWidth(maxTextW)
            measure:SetText(fullText)

            local measuredW = measure:GetStringWidth() or 100
            local finalTextW = math.max(40, measuredW)
            if finalTextW > maxTextW then finalTextW = maxTextW end

            measure:SetWidth(finalTextW)
            measure:SetText(fullText)
            local textHeight = measure:GetStringHeight() or (fontSize + 4)

            local bubble = CreateFrame("Frame", nil, V.Content)
            bubble:SetSize(finalTextW, textHeight)

            local bg = bubble:CreateTexture(nil, "BACKGROUND")
            bg:SetTexture("Interface\\Buttons\\WHITE8x8")

            if msg.isMine then
                bg:SetVertexColor(0.1, 0.3, 0.1, 0.9)
                
                -- СВОИ: Привязка к правому верхнему углу пузыря
                -- Сжимаем фон: отступаем влево (-SHRINK) и вниз (-SHRINK)
                bg:SetPoint("TOPRIGHT", bubble, "TOPRIGHT", -SHRINK_AMOUNT, -SHRINK_AMOUNT)
                bg:SetPoint("BOTTOMLEFT", bubble, "BOTTOMLEFT", SHRINK_AMOUNT, SHRINK_AMOUNT)
                
                bubble:SetPoint("TOPRIGHT", V.Content, "TOPRIGHT", -PADDING_RIGHT, -yOffset)
            else
                bg:SetVertexColor(0.2, 0.2, 0.2, 0.9)
                
                -- ЧУЖИЕ: Привязка к левому верхнему углу пузыря
                -- Сжимаем фон: отступаем вправо (+SHRINK) и вниз (-SHRINK)
                bg:SetPoint("TOPLEFT", bubble, "TOPLEFT", SHRINK_AMOUNT, -SHRINK_AMOUNT)
                bg:SetPoint("BOTTOMRIGHT", bubble, "BOTTOMRIGHT", -SHRINK_AMOUNT, SHRINK_AMOUNT)
                
                bubble:SetPoint("TOPLEFT", V.Content, "TOPLEFT", PADDING_LEFT, -yOffset)
            end

            local textStr = bubble:CreateFontString(nil, "OVERLAY")
            textStr:SetFont(fontPath, fontSize)
            textStr:SetJustifyH("LEFT")
            textStr:SetWordWrap(true)
            textStr:SetNonSpaceWrap(true)
            
            textStr:SetPoint("TOPLEFT", bubble, "TOPLEFT", 0, 0)
            textStr:SetWidth(finalTextW)
            textStr:SetText(fullText)

            yOffset = yOffset + textHeight + GAP
        end

        V.Content:SetHeight(math.max(1, yOffset))
        V.Scroll:UpdateScrollChildRect()
        V.ScrollToBottom()
    end
end)

NSQC4.RegisterModule("nowim", function()
    local UI = NSQC4.Nowim.UI
    local V = UI.ChatView

    -- ========================================================================
    -- КОНСТАНТА: Отступ сверху и снизу от краев окна.
    -- 0 = вплотную, 2 = небольшой зазор.
    -- ========================================================================
    local VERTICAL_PADDING = 2

    function V.OnResize()
        local scroll = V.Scroll
        local scrollBar = V.ScrollBar
        
        scroll:ClearAllPoints()
        -- Слева 0 (вплотную). Сверху отступаем на VERTICAL_PADDING.
        scroll:SetPoint("TOPLEFT", 0, -VERTICAL_PADDING)
        
        -- Справа -16 (под скроллбар). Снизу отступаем на VERTICAL_PADDING.

    divider:SetPoint("TOPLEFT", 0, 0)
    divider:SetPoint("TOPRIGHT", 0, 0)
    divider:SetHeight(1)
    divider:Hide()

    -- Без шаблона UIPanelScrollFrameTemplate: не создаём лишние стрелки и скроллбар
    local scroll = CreateFrame("ScrollFrame", "NSQC4_NowimInputScroll", container)
    scroll:SetPoint("TOPLEFT", 0, -5)
    scroll:SetPoint("BOTTOMRIGHT", 0, 0)
    I.Scroll = scroll

    local editBox = CreateFrame("EditBox", nil, scroll)
    editBox:SetMultiLine(true)
    editBox:SetAutoFocus(false)
    editBox:SetFontObject("ChatFontNormal")
    editBox:SetWidth(10) 
    editBox:SetHeight(20) 
    editBox:SetTextInsets(4, 4, 4, 4)
    scroll:SetScrollChild(editBox)
    I.EditBox = editBox

    local counter = container:CreateFontString(nil, "OVERLAY", "GameFontNormalSmall")
    counter:SetPoint("BOTTOMRIGHT", -5, 5)
    counter:SetTextColor(0.5, 0.5, 0.5)
    I.Counter = counter

    local function UpdateCounter()
        local text = editBox:GetText() or ""
        local bytes = #text
        counter:SetText(bytes .. "/" .. C.MAX_BYTES)
        if bytes > C.MAX_BYTES then counter:SetTextColor(1, 0, 0) else counter:SetTextColor(0.5, 0.5, 0.5) end
    end

    local function UpdateInputHeight()
        local text = editBox:GetText() or ""
        local lines = 1
        for _ in text:gmatch("\n") do lines = lines + 1 end
        local newH = math.min(5, lines) * 16 + 10
        container:SetHeight(newH)
        if UI.ChatView and UI.ChatView.OnResize then UI.ChatView.OnResize() end
    end

    editBox:SetScript("OnEscapePressed", function(self) self:ClearFocus() end)
    
    editBox:SetScript("OnTextChanged", function(self)
        UpdateCounter()
        UpdateInputHeight()
    end)

    editBox:SetScript("OnEnterPressed", function(self)
        if IsShiftKeyDown() then
            self:Insert("\n")
        else
            local text = self:GetText()
            if text and text:match("%S") then
                for line in text:gmatch("[^\r\n]+") do
                    line = line:gsub("^%s+", ""):gsub("%s+$", "")
                    if line ~= "" then
                        E.SendMessage(U.TrimToBytes(line, C.MAX_BYTES))
                    end
                end
            end
            self:SetText("")
            self:ClearFocus()
        end
    end)

    -- Управление видимостью
    editBox:SetScript("OnEditFocusGained", function()
        divider:Show()
        editBox:SetAlpha(1) -- ВОЗВРАЩАЕМ ВИДИМОСТЬ
    end)

    editBox:SetScript("OnEditFocusLost", function()
        divider:Hide()
        editBox:SetAlpha(0) -- СКРЫВАЕМ
    end)

    -- Публичная функция открытия (для перехвата Enter)
    function I.Open(text)
        if not container:IsShown() then container:Show() end
        editBox:SetAlpha(1) -- ВОЗВРАЩАЕМ ВИДИМОСТЬ ПРИНУДИТЕЛЬНО
        divider:Show()
        editBox:SetFocus()
        if text and text ~= "" then
            editBox:SetText(text)
        end
    end

    function I.OnResize()
        local leftOpen = UI.LeftPanel and UI.LeftPanel.Frame:IsShown()
        local rightOpen = UI.RightPanel and UI.RightPanel.Frame:IsShown()

        local leftPadding = leftOpen and 145 or 5
        local rightPadding = rightOpen and -41 or -25

        container:ClearAllPoints()
        container:SetPoint("BOTTOMLEFT", leftPadding, 5)
        container:SetPoint("BOTTOMRIGHT", rightPadding, 5)

        local w = container:GetWidth()
        if w and w > 20 then editBox:SetWidth(w - 20) end
    end

    container:SetScript("OnSizeChanged", function()
        local w = container:GetWidth()
        if w and w > 20 then editBox:SetWidth(w - 20) end
    end)
end)

-- ============================================================================
-- NSQC4 / nowim / UI / ResizeGrip
-- Ручка изменения размера окна в правом нижнем углу.
-- ============================================================================

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local UI = NSQC4.Nowim.UI
    local mainFrame = UI.MainFrame

    if not mainFrame then return end
    if mainFrame.ResizeGrip then return end -- защита от повторного создания

    local MIN_W = C.MIN_W or 200
    local MIN_H = C.MIN_H or 150

    local grip = CreateFrame("Button", "NSQC4_NowimResizeGrip", mainFrame)
    grip:SetSize(16, 16)
    grip:SetPoint("BOTTOMRIGHT", -2, 2)
    grip:SetFrameLevel(mainFrame:GetFrameLevel() + 20)
    grip:EnableMouse(true)
    -- ВАЖНО: НЕ регистрируем drag — таскаем вручную, иначе конфликт с OnMouseDown/Up

    -- Визуал: текстурка диагональных полосок
    local tex = grip:CreateTexture(nil, "ARTWORK")
    tex:SetAllPoints(grip)
    tex:SetTexture("Interface\\ChatFrame\\UI-ChatIM-SizeGrabber-Up")
    grip.tex = tex

    local isResizing = false

    local function StopResize()
        if not isResizing then return end
        isResizing = false
        grip.tex:SetVertexColor(1, 1, 1, 1)

        -- Сохраняем в конфиг
        local w, h = mainFrame:GetWidth(), mainFrame:GetHeight()
        C.db.window.w = w
        C.db.window.h = h

        -- Пересчёт внутренних элементов
        if UI.ChatView and UI.ChatView.OnResize then UI.ChatView.OnResize() end
        if UI.InputBox and UI.InputBox.OnResize then UI.InputBox.OnResize() end
        if UI.LeftPanel and UI.LeftPanel.OnMainFrameResize then UI.LeftPanel.OnMainFrameResize() end
    end

    grip:SetScript("OnMouseDown", function(self, button)
        if button ~= "LeftButton" then return end
        isResizing = true
        self.tex:SetVertexColor(0, 1, 1, 1)

        local scale = mainFrame:GetEffectiveScale()
        local cursorX, cursorY = GetCursorPosition()
        cursorX, cursorY = cursorX / scale, cursorY / scale
        self._startW = mainFrame:GetWidth()
        self._startH = mainFrame:GetHeight()
        self._startX = cursorX
        self._startY = cursorY
    end)

    -- Ловим отпускание мыши ВО ВСЁМ МИРЕ, а не только над грипом
    grip:SetScript("OnUpdate", function(self)
        if not isResizing then return end

        -- Если кнопка мыши уже отпущена — прекращаем ресайз
        if not IsMouseButtonDown("LeftButton") then
            StopResize()
            return
        end

        local scale = mainFrame:GetEffectiveScale()
        local cursorX, cursorY = GetCursorPosition()
        cursorX, cursorY = cursorX / scale, cursorY / scale

        local deltaX = cursorX - self._startX
        local deltaY = cursorY - self._startY

        local newW = math.max(MIN_W, self._startW + deltaX)
        local newH = math.max(MIN_H, self._startH - deltaY)

        mainFrame:SetSize(newW, newH)
    end)

    grip:SetScript("OnMouseUp", function(self, button)
        if button == "LeftButton" then
            StopResize()
        end
    end)

    grip:SetScript("OnHide", function()
        StopResize()
    end)

    -- Ховер-эффект
    grip:SetScript("OnEnter", function(self)
        if not isResizing then
            self.tex:SetVertexColor(0, 1, 1, 1)
        end
    end)
    grip:SetScript("OnLeave", function(self)
        if not isResizing then
            self.tex:SetVertexColor(1, 1, 1, 1)
        end
    end)

    mainFrame.ResizeGrip = grip
end)

-- NSQC4 / nowim / UI / Tabs
-- ============================================================================

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local E = NSQC4.Nowim.ChatEngine
    local UI = NSQC4.Nowim.UI

    local T = {}
    UI.Tabs = T
    local mainFrame = UI.MainFrame

    local tabContainer = CreateFrame("Frame", nil, mainFrame)
    tabContainer:SetPoint("BOTTOMLEFT", mainFrame, "TOPLEFT", 30, 0) 
    tabContainer:SetPoint("BOTTOMRIGHT", mainFrame, "TOPRIGHT", -30, 0) -- Сдвинуто влево, чтобы не мешать крестику
    tabContainer:SetHeight(24)
    tabContainer:SetFrameLevel(mainFrame:GetFrameLevel() + 10)
    
    tabContainer:SetAlpha(0.01)
    tabContainer:EnableMouse(true)

    tabContainer:SetScript("OnUpdate", function(self)
        local hasUnread = false
        for _, key in ipairs(C.db.openTabs or {}) do
            if (C.db.unread[key] or 0) > 0 then
                hasUnread = true
                break
            end
        end

        if hasUnread then
            self:SetAlpha(1)
            return
        end

        local isOver = self:IsMouseOver()
        if not isOver then
            for _, btn in pairs(T.Buttons) do
                if btn:IsMouseOver() then
                    isOver = true
                    break
                end
            end
        end

        if isOver then
            self:SetAlpha(1)
        else
            self:SetAlpha(0.01)
        end
    end)

    mainFrame.Tabs = tabContainer
    T.Buttons = {}
    local tabButtons = T.Buttons
    local TAB_WIDTH = 30
    local TAB_HEIGHT = 24

    local function GetTabLetter(key)
        if key == "GUILD" then return "Г" end
        if key == "OFFICER" then return "О" end
        if key == "RAID" then return "Р" end
        
        local firstChar = key:match("^([%z\1-\127\194-\244][\128-\191]*)") or "?"
        local upperMap = {
            ["а"]="А", ["б"]="Б", ["в"]="В", ["г"]="Г", ["д"]="Д", ["е"]="Е", ["ё"]="Ё",
            ["ж"]="Ж", ["з"]="З", ["и"]="И", ["й"]="Й", ["к"]="К", ["л"]="Л", ["м"]="М",
            ["н"]="Н", ["о"]="О", ["п"]="П", ["р"]="Р", ["с"]="С", ["т"]="Т", ["у"]="У",
            ["ф"]="Ф", ["х"]="Х", ["ц"]="Ц", ["ч"]="Ч", ["ш"]="Ш", ["щ"]="Щ", ["ъ"]="Ъ",
            ["ы"]="Ы", ["ь"]="Ь", ["э"]="Э", ["ю"]="Ю", ["я"]="Я"
        }
        if upperMap[firstChar] then return upperMap[firstChar] end
        return firstChar:upper()
    end

    -- ========================================================================
    -- Меню настроек (ПКМ по вкладке)
    -- ========================================================================
    local menuFrame

    local function ShowTabMenu(btn)
        if not menuFrame then
            menuFrame = CreateFrame("Frame", "NSQC4NowimTabMenu", UIParent, "UIDropDownMenuTemplate")
        end

        UIDropDownMenu_Initialize(menuFrame, function(self, level)
            local info = UIDropDownMenu_CreateInfo()
            
            info.text = "Проматывать непрочитанные"
            info.isNotRadio = true
            info.checked = C.db.settings.autoScroll
            info.func = function()
                C.db.settings.autoScroll = not C.db.settings.autoScroll
                if UI.ChatView then UI.ChatView.Refresh() end
            end
            UIDropDownMenu_AddButton(info, level)

            info = UIDropDownMenu_CreateInfo()
            info.text = "Перехватывать Enter"
            info.isNotRadio = true
            info.checked = C.db.settings.interceptEnter
            info.func = function()
                C.db.settings.interceptEnter = not C.db.settings.interceptEnter
            end
            UIDropDownMenu_AddButton(info, level)
        end)

        ToggleDropDownMenu(1, nil, menuFrame, btn, 0, 0)
    end

    local function CreateTabButton(key)
        local btn = CreateFrame("Button", nil, tabContainer)
        btn:SetSize(TAB_WIDTH, TAB_HEIGHT)
        btn:SetBackdrop({
            bgFile = "Interface\\Buttons\\WHITE8x8",
            edgeFile = "Interface\\Tooltips\\UI-Tooltip-Border",
            edgeSize = 8,
            insets = { left = 2, right = 2, top = 2, bottom = 0 }
        })
        btn:SetBackdropColor(0.15, 0.15, 0.15, 0.9)
        btn:SetBackdropBorderColor(0.3, 0.3, 0.3, 1)

        local text = btn:CreateFontString(nil, "OVERLAY", "GameFontNormalLarge")
        text:SetPoint("CENTER", 0, 2)
        text:SetText(GetTabLetter(key))
        text:SetTextColor(1, 1, 1)

        local close = CreateFrame("Button", nil, btn)
        close:SetSize(12, 12)
        close:SetPoint("TOPRIGHT", 2, 2)
        close:SetNormalTexture("Interface\\Buttons\\UI-GroupLoot-Pass-Up")
        close:Hide()
        
        btn:SetScript("OnEnter", function() close:Show() end)
        btn:SetScript("OnLeave", function() close:Hide() end)
        
        close:SetScript("OnClick", function(_, button)
            if button == "LeftButton" then E.CloseTab(key) end
        end)

        btn:RegisterForClicks("LeftButtonUp", "RightButtonUp")
        btn:SetScript("OnClick", function(self, button)
            if button == "RightButton" then
                ShowTabMenu(self)
            else
                E.OpenDialog(key)
            end
        end)
        
        btn:SetScript("OnEnter", function(self)
            GameTooltip:SetOwner(self, "ANCHOR_BOTTOM")
            GameTooltip:SetText(key, 1, 1, 1)
            GameTooltip:AddLine("ПКМ — настройки", 0.6, 0.6, 0.6)
            GameTooltip:Show()
        end)
        btn:SetScript("OnLeave", function()
            GameTooltip:Hide()
        end)

        tabButtons[key] = btn
        return btn
    end

    function T.Refresh()
        local openTabs = C.db.openTabs
        local active = C.db.activeTab

        for _, btn in pairs(tabButtons) do btn:Hide() end
        if not openTabs then return end

        local xPos = 0
        for _, key in ipairs(openTabs) do
            local btn = tabButtons[key] or CreateTabButton(key)
            btn:ClearAllPoints()
            btn:SetPoint("BOTTOMLEFT", tabContainer, "BOTTOMLEFT", xPos, 0) 
            btn:Show()

            local unread = C.db.unread[key] or 0

            if key == active then
                btn:SetBackdropColor(0.1, 0.1, 0.1, 1)
                btn:SetBackdropBorderColor(0.5, 0.5, 0.5, 1)
                btn:SetHeight(TAB_HEIGHT + 2)
            else
                btn:SetBackdropColor(0.15, 0.15, 0.15, 0.9)
                btn:SetBackdropBorderColor(0.3, 0.3, 0.3, 1)
                btn:SetHeight(TAB_HEIGHT)
            end
            
            if unread > 0 then
                if not btn.RedDot then
                    btn.RedDot = btn:CreateTexture(nil, "OVERLAY")
                    btn.RedDot:SetSize(6, 6)
                    btn.RedDot:SetPoint("TOPLEFT", 2, -2)
                    btn.RedDot:SetTexture("Interface\\Buttons\\WHITE8x8")
                    btn.RedDot:SetVertexColor(1, 0, 0, 1)
                end
                btn.RedDot:Show()
            elseif btn.RedDot then
                btn.RedDot:Hide()
            end

            xPos = xPos + TAB_WIDTH + 2
        end
    end
end)

-- ============================================================================
-- NSQC4 / nowim / Utils
-- Вспомогательные функции: UTF-8, время, цвета.
-- ============================================================================

NSQC4.RegisterModule("nowim", function()

    local U = {}
    NSQC4.Nowim.Utils = U

    -- Правильная обрезка строки по байтам (UTF-8)
    function U.TrimToBytes(str, maxBytes)
        if not str then return "" end
        local bytes = 0
        local result = {}
        
        -- Итерируемся по символам UTF-8
        for char in str:gmatch("[%z\1-\127\194-\244][\128-\191]*") do
            local charBytes = #char
            if bytes + charBytes > maxBytes then break end
            table.insert(result, char)
            bytes = bytes + charBytes
        end
        
        return table.concat(result)
    end

    -- Получить текущее время в формате [ЧЧ:ММ]
    function U.GetTimeString()
        return date("%H:%M")
    end

end)
