NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local A = {}
    NSQC4.Nowim.Aliases = A

    C.db.aliases = C.db.aliases or {}

    -- Получить "главный" ник. Если это твинк, вернет ник владельца.
    function A.GetMainName(nick)
        if not nick or nick == "" then return nil end
        local lowerNick = nick:lower()
        
        -- Проверяем, не является ли ник основным
        for mainName, alts in pairs(C.db.aliases) do
            if mainName:lower() == lowerNick then
                return mainName
            end
            -- Проверяем твинков
            for _, altName in ipairs(alts) do
                if altName:lower() == lowerNick then
                    return mainName -- Возвращаем основного (например, "ВасяПупкин")
                end
            end
        end
        
        return nick -- Если не найден в связках, возвращаем как есть
    end

    -- Добавить твинка к основному нику
    function A.AddAlt(mainName, altName)
        if not mainName or not altName then return end
        C.db.aliases[mainName] = C.db.aliases[mainName] or {}
        
        -- Проверяем, нет ли уже такого твинка
        for _, name in ipairs(C.db.aliases[mainName]) do
            if name:lower() == altName:lower() then return end
        end
        
        table.insert(C.db.aliases[mainName], altName)
    end
end)

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local S = NSQC4.Nowim.Storage
    local N = NSQC4.Nowim.Network
    local A = NSQC4.Nowim.Aliases

    local E = {}
    NSQC4.Nowim.ChatEngine = E

    C.db.unread = C.db.unread or {}
    C.db.openTabs = C.db.openTabs or {}
    C.db.savedScroll = C.db.savedScroll or {}

    function E.OpenDialog(key)
        if not key or key == "" then return end
        
        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }

        local isOpen = false
        for _, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then isOpen = true; break end
        end

        if not isOpen then
            table.insert(C.db.openTabs, key)
            if C.db.settings.autoScroll then
                C.db.savedScroll[key] = 999999 
            else
                C.db.savedScroll[key] = 0
            end
        end

        C.db.activeTab = key
        C.db.unread[key] = 0

        E.UpdateUI()
    end

    function E.CloseTab(key)
        for i, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then
                table.remove(C.db.openTabs, i)
                break
            end
        end

        if C.db.activeTab == key then
            C.db.activeTab = C.db.openTabs[#C.db.openTabs]
        end

        E.UpdateUI()
    end

    function E.SendMessage(text)
        local target = C.db.activeTab
        if not target then return end
        N.QueueMessage(target, text)
    end

    function E.OnIncomingMessage(sender, targetKey, text)
        if C.db.activeTab ~= targetKey then
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
        end
        E.UpdateUI()
    end

    -- ИСПРАВЛЕННАЯ ФУНКЦИЯ ОБНОВЛЕНИЯ ИНТЕРФЕЙСА
    function E.UpdateUI()
        local UI = NSQC4.Nowim.UI
        if not UI then return end

        -- 1. Обновляем вкладки
        if UI.Tabs and UI.Tabs.Refresh then 
            UI.Tabs.Refresh() 
        end
        
        -- 2. Обновляем чат (вызываем OnResize, так как он гарантирует правильные размеры перед отрисовкой)
        if UI.Chat and UI.Chat.OnResize then
            UI.Chat.OnResize()
        elseif UI.Chat and UI.Chat.Refresh then
            UI.Chat.Refresh()
        end
    end

    SLASH_NOWIM_CHAT1 = "/nowim"
    SlashCmdList["NOWIM_CHAT"] = function(msg)
        msg = (msg or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if msg ~= "" then
            E.OpenDialog(msg)
        end
        if NSQC4.Nowim.UI and NSQC4.Nowim.UI.MainFrame then
            NSQC4.Nowim.UI.MainFrame:ShowFrame()
        end
    end

end)

NSQC4.RegisterModule("nowim", function()

    local S = NSQC4.Nowim.Storage
    local C = NSQC4.Nowim.Config
    local myName = UnitName("player")

    local frame = CreateFrame("Frame")
    frame:RegisterEvent("CHAT_MSG_GUILD")
    frame:RegisterEvent("CHAT_MSG_OFFICER")
    frame:RegisterEvent("CHAT_MSG_RAID")
    frame:RegisterEvent("CHAT_MSG_RAID_LEADER")
    frame:RegisterEvent("CHAT_MSG_WHISPER")
    
    frame:SetScript("OnEvent", function(self, event, text, sender)
        if not text or text == "" then return end
        if text:sub(1, 5) == "NSMSG" then return end 

        local E = NSQC4.Nowim.ChatEngine
        if not E then return end

        local chatType = event:match("^CHAT_MSG_(.+)$")
        local targetKey = chatType
        
        -- Анти-дубль
        if sender == myName and NSQC4.Nowim._lastSentText == text and (GetTime() - (NSQC4.Nowim._lastSentTime or 0)) < 2 then
            return
        end

        -- Убираем первый ник (ретранслятор)
        local cleanText = text:gsub("^" .. sender .. ":%s*", "")

        -- Парсим AOClick
        local formattedText = cleanText
        if NSQC4.AOClick_ReplaceAll then
            formattedText = NSQC4.AOClick_ReplaceAll(cleanText)
        end

        if chatType == "WHISPER" then
            targetKey = sender
            E.OpenDialog(targetKey) 
        end

        -- Сохраняем
        S.AddMessage(targetKey, formattedText, false, nil)
        
        -- Обновляем UI
        if C.db.activeTab == targetKey then
            E.UpdateUI()
        else
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
            
            -- Безопасный вызов ChatView.Refresh, если он есть
            local UI = NSQC4.Nowim.UI
            if UI and UI.ChatView and UI.ChatView.Refresh then
                UI.ChatView.Refresh()
            end
            
            E.UpdateUI()
        end
    end)
end)

    enterCatcher:SetSize(1, 1)
    enterCatcher:SetPoint("CENTER")
    enterCatcher:Hide()

    -- Обработчик Enter
    enterCatcher:SetScript("OnClick", function()
        local Nowim = NSQC4.Nowim
        if not Nowim then return end
        local C = Nowim.Config
        local UI = Nowim.UI
        if not UI or not UI.MainFrame then return end

        local mainFrame = UI.MainFrame

        -- Если окно закрыто — открываем его на последнем чате
        if not mainFrame:IsShown() then
            if C.db.activeTab then
                -- Восстанавливаем последний чат (если он ещё в openTabs)
                local found = false
                for _, key in ipairs(C.db.openTabs or {}) do
                    if key == C.db.activeTab then found = true; break end
                end
                if found and Nowim.ChatEngine then
                    Nowim.ChatEngine.OpenDialog(C.db.activeTab)
                end
            end
            mainFrame:ShowFrame()
        end

        -- Ставим фокус в поле ввода
        if UI.InputBox and UI.InputBox.Open then
            UI.InputBox.Open()
        end
    end)

    -- Установка / снятие бинда в зависимости от настройки
    local function ApplyBinding()
        local Nowim = NSQC4.Nowim
        if not Nowim or not Nowim.Config then return end
        local enabled = Nowim.Config.db.settings.interceptEnter

        if enabled then
            SetOverrideBindingClick(enterCatcher, true, "ENTER", "NSQC4_NowimEnterCatcher")
            SetOverrideBindingClick(enterCatcher, true, "NUMPADENTER", "NSQC4_NowimEnterCatcher")
        else
            ClearOverrideBindings(enterCatcher)
        end
    end

    NSQC4.Nowim = NSQC4.Nowim or {}
    NSQC4.Nowim.ApplyEnterBinding = ApplyBinding

    -- Ждём, когда всё загрузится, и вешаем бинд
    local f = CreateFrame("Frame")
    f:RegisterEvent("PLAYER_LOGIN")
    f:SetScript("OnEvent", function(self)
        self:UnregisterAllEvents()
        -- Настройка могла быть ещё не создана — проверяем
        if NSQC4.Nowim and NSQC4.Nowim.Config and NSQC4.Nowim.Config.db then
            ApplyBinding()
        end
    end)

    -- Хук на Hide/Show окна, чтобы бинд не снимался при скрытии окна
    -- (перехват должен работать и при закрытом окне, если настройка включена)
    local function HookMainFrame()
        local UI = NSQC4.Nowim and NSQC4.Nowim.UI
        local mainFrame = UI and UI.MainFrame
        if not mainFrame or mainFrame._nowimEnterHooked then return end
        mainFrame._nowimEnterHooked = true

        local orig_Hide = mainFrame.Hide
        mainFrame.Hide = function(self)
            if orig_Hide then orig_Hide(self) else self:Hide() end
            -- Бинд НЕ снимаем: если interceptEnter включён, Enter должен
            -- открывать окно даже когда оно скрыто.
            ApplyBinding()
        end

        local orig_Show = mainFrame.ShowFrame
        mainFrame.ShowFrame = function(self)
            if orig_Show then orig_Show(self) else self:Show() end
            ApplyBinding()
        end
    end

    if not HookMainFrame() then
        local g = CreateFrame("Frame")
        g:RegisterEvent("PLAYER_LOGIN")
        g:SetScript("OnEvent", function(self)
            if HookMainFrame() then self:UnregisterAllEvents() end
        end)
    end

end)

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local S = NSQC4.Nowim.Storage

    local N = {}
    NSQC4.Nowim.Network = N

    local sendQueue = {}
    local isSending = false

    local function ProcessQueue()
        if isSending or #sendQueue == 0 then return end
        isSending = true

        local msgData = table.remove(sendQueue, 1)
        local target = msgData.target
        local text = U.TrimToBytes(msgData.text, C.MAX_BYTES)
        
        local isChannel = C.db.channels[target] ~= nil

        if target == UnitName("player") then
            S.AddMessage(target, text, true, UnitName("player"))
            N.UpdateUI(target)
        elseif isChannel then
            local chatType = C.db.channels[target].type
            
            -- Запоминаем, что мы только что отправили это в канал,
            -- чтобы ChatHook проигнорировал входящее эхо
            NSQC4.Nowim._lastSentText = text
            NSQC4.Nowim._lastSentTime = GetTime()
            NSQC4.Nowim._lastSentTarget = target
            
            if chatType == "GUILD" or chatType == "RAID" then
                SendChatMessage(text, chatType)
            end
            
            S.AddMessage(target, text, true, UnitName("player"))
        else
            SendAddonMessage(C.ADDON_PREFIX, string.format("FROM:%s|TEXT:%s", UnitName("player"), text), "WHISPER", target)
            S.AddMessage(target, text, true, UnitName("player"))
        end

        local timer = CreateFrame("Frame")
        local elapsed = 0
        timer:SetScript("OnUpdate", function(self, dt)
            elapsed = elapsed + dt
            if elapsed >= C.SEND_DELAY then
                self:SetScript("OnUpdate", nil)
                isSending = false
                ProcessQueue()
            end
        end)
    end

    function N.QueueMessage(target, text)
        if not target or not text or text == "" then return end
        table.insert(sendQueue, { target = target, text = text })
        ProcessQueue()
    end

    -- ========================================================================
    -- ПРИЁМ АДДОН-СООБЩЕНИЙ
    -- ========================================================================
    if type(RegisterAddonMessagePrefix) == "function" then
        RegisterAddonMessagePrefix(C.ADDON_PREFIX)
    end

    if NSQC4.ChatHandler and type(NSQC4.ChatHandler.Register) == "function" then
        NSQC4.ChatHandler:Register("ADDON:" .. string.lower(C.ADDON_PREFIX), {
            func = function(words, sender, text, channel, prefix)
                -- (без изменений, оставляем как было)
                if text:sub(1, 5) == "CHAN:" then
                    local chatType, fromNick, msgText = text:match("^CHAN:([^|]+)|FROM:([^|]+)|TEXT:(.+)$")
                    if chatType and msgText then
                        S.AddMessage(chatType, msgText, false, fromNick)
                        N.UpdateUI(chatType)
                    end
                else
                    local fromNick, msgText = text:match("^FROM:([^|]+)|TEXT:(.+)$")
                    if fromNick and msgText then
                        S.AddMessage(fromNick, msgText, false, fromNick)
                        N.UpdateUI(fromNick)
                        if C.db.settings.playSound and PlaySoundFile then
                            PlaySoundFile("Interface\\AddOns\\NSQC4\\Media\\Sounds\\m.ogg")
                        end
                    end
                end
            end,
            stopOnMatch = true,
        })
    end

    function N.UpdateUI(target)
        if NSQC4.Nowim.UI and NSQC4.Nowim.UI.ActiveTarget == target then
            if NSQC4.Nowim.UI.ChatView then NSQC4.Nowim.UI.ChatView.Refresh() end
        end
    end
end)

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local A = NSQC4.Nowim.Aliases

    local S = {}
    NSQC4.Nowim.Storage = S

    local function ResolveKey(nick)
        if not nick then return nil end
        if C.db.channels[nick] then return nick end 
        return A.GetMainName(nick)
    end

    function S.AddMessage(target, text, isMine, realSender)
        local key = ResolveKey(target)
        if not key or not text then return end
        
        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }
        local dialog = C.db.dialogs[key]
        
        -- Если realSender nil, значит имя уже есть в тексте (ретранслятор)
        local charTag = realSender or (isMine and UnitName("player") or nil)
        
        table.insert(dialog.messages, {
            text = text,
            isMine = isMine,
            time = U.GetTimeString(),
            char = charTag 
        })
        
        while #dialog.messages > C.MAX_HISTORY do
            table.remove(dialog.messages, 1)
        end
    end

    function S.GetMessages(key)
        local resolvedKey = ResolveKey(key)
        if not resolvedKey or not C.db.dialogs[resolvedKey] then return nil end
        return C.db.dialogs[resolvedKey].messages
    end

    function S.ClearHistory(key)
        local resolvedKey = ResolveKey(key)
        if resolvedKey and C.db.dialogs[resolvedKey] then
            C.db.dialogs[resolvedKey].messages = {}
        end
    end

    function S.GetContacts()
        local list = {}
        
        -- Добавляем ЛС (диалоги, которых нет в channels)
        for key in pairs(C.db.dialogs) do
            if not C.db.channels[key] then
                table.insert(list, { key = key, name = tostring(key), isChannel = false })
            end
        end
        
        -- Добавляем каналы
        for id, data in pairs(C.db.channels) do
            table.insert(list, { key = id, name = tostring(data.name or id), isChannel = true })
        end

        -- Безопасная сортировка: сравниваем только строки
        table.sort(list, function(a, b)
            local nameA = tostring(a.name or "")
            local nameB = tostring(b.name or "")
            if a.isChannel ~= b.isChannel then 
                return a.isChannel 
            end
            return nameA < nameB
        end)
        
        return list
    end

end)

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local E = NSQC4.Nowim.ChatEngine
    local U = NSQC4.Nowim.Utils
    local UI = NSQC4.Nowim.UI

    local f = CreateFrame("Frame")
    f:RegisterEvent("PLAYER_LOGIN")
    f:SetScript("OnEvent", function()
        if not UI.Content then return end
        local I = {}
        UI.Input = I

        local container = CreateFrame("Frame", nil, UI.Content)
        container:SetHeight(25)
        I.Container = container

        local scroll = CreateFrame("ScrollFrame", nil, container)
        scroll:SetPoint("TOPLEFT", 0, 0)
        scroll:SetPoint("BOTTOMRIGHT", 0, 0)
        
        local editBox = CreateFrame("EditBox", nil, scroll)
        editBox:SetMultiLine(true)
        editBox:SetAutoFocus(false)
        editBox:SetFontObject("ChatFontNormal")
        editBox:SetWidth(100)
        editBox:SetHeight(20)
        scroll:SetScrollChild(editBox)
        I.EditBox = editBox

        local counter = container:CreateFontString(nil, "OVERLAY", "GameFontNormalSmall")
        counter:SetPoint("BOTTOMRIGHT", -5, 5)
        counter:SetTextColor(0.5, 0.5, 0.5)

        local function UpdateInputHeight()
            local text = editBox:GetText() or ""
            local lines = 1
            for _ in text:gmatch("\n") do lines = lines + 1 end
            container:SetHeight(math.min(5, lines) * 16 + 10)
            if UI.Chat then UI.Chat.OnResize() end
        end

        editBox:SetScript("OnEscapePressed", function(self) self:ClearFocus() end)
        
        editBox:SetScript("OnTextChanged", function(self)
            local text = self:GetText() or ""
            local bytes = #text
            counter:SetText(bytes .. "/" .. C.MAX_BYTES)
            counter:SetTextColor(bytes > C.MAX_BYTES and 1 or 0.5, bytes > C.MAX_BYTES and 0 or 0.5, 0.5)
            UpdateInputHeight()
        end)

        editBox:SetScript("OnEnterPressed", function(self)
            if IsShiftKeyDown() then
                self:Insert("\n")
            else
                local text = self:GetText()
                if text and text:match("%S") then
                    for line in text:gmatch("[^\r\n]+") do
                        E.SendMessage(U.TrimToBytes(line:trim(), C.MAX_BYTES))
                    end
                end
                self:SetText("")
                self:ClearFocus()
            end
        end)

        editBox:SetScript("OnEditFocusGained", function() editBox:SetAlpha(1) end)
        editBox:SetScript("OnEditFocusLost", function() editBox:SetAlpha(0.5) end)

        function I.Open()
            container:Show()
            editBox:SetFocus()
        end

        function I.OnResize()
            container:ClearAllPoints()
            container:SetPoint("BOTTOMLEFT", 0, 2)
            container:SetPoint("BOTTOMRIGHT", -20, 2)
            editBox:SetWidth(container:GetWidth() - 10)
        end

        container:SetScript("OnSizeChanged", function() editBox:SetWidth(container:GetWidth() - 10) end)
        
        I.OnResize()
        f:UnregisterAllEvents()
    end)
end)

    container:SetHeight(25)
    I.Container = container

    local scroll = CreateFrame("ScrollFrame", nil, container)
    scroll:SetPoint("TOPLEFT", 0, 0)
    scroll:SetPoint("BOTTOMRIGHT", 0, 0)
    
    local editBox = CreateFrame("EditBox", nil, scroll)
    editBox:SetMultiLine(true)
    editBox:SetAutoFocus(false)
    editBox:SetFontObject("ChatFontNormal")
    editBox:SetWidth(100)
    editBox:SetHeight(20)
    scroll:SetScrollChild(editBox)
    I.EditBox = editBox

    local counter = container:CreateFontString(nil, "OVERLAY", "GameFontNormalSmall")
    counter:SetPoint("BOTTOMRIGHT", -5, 5)
    counter:SetTextColor(0.5, 0.5, 0.5)

    local function UpdateInputHeight()
        local text = editBox:GetText() or ""
        local lines = 1
        for _ in text:gmatch("\n") do lines = lines + 1 end
        container:SetHeight(math.min(5, lines) * 16 + 10)
        if UI.Chat then UI.Chat.OnResize() end
    end

    editBox:SetScript("OnEscapePressed", function(self) self:ClearFocus() end)
    
    editBox:SetScript("OnTextChanged", function(self)
        local text = self:GetText() or ""
        local bytes = #text
        counter:SetText(bytes .. "/" .. C.MAX_BYTES)
        counter:SetTextColor(bytes > C.MAX_BYTES and 1 or 0.5, bytes > C.MAX_BYTES and 0 or 0.5, 0.5)
        UpdateInputHeight()
    end)

    editBox:SetScript("OnEnterPressed", function(self)
        if IsShiftKeyDown() then
            self:Insert("\n")
        else
            local text = self:GetText()
            if text and text:match("%S") then
                for line in text:gmatch("[^\r\n]+") do
                    E.SendMessage(U.TrimToBytes(line:trim(), C.MAX_BYTES))
                end
            end
            self:SetText("")
            self:ClearFocus()
        end
    end)

    editBox:SetScript("OnEditFocusGained", function() editBox:SetAlpha(1) end)
    editBox:SetScript("OnEditFocusLost", function() editBox:SetAlpha(0.5) end)

    function I.Open()
        container:Show()
        editBox:SetFocus()
    end

    function I.OnResize()
        local left = (UI.LeftPanel.Frame and UI.LeftPanel.Frame:IsShown()) and K.LEFT_WIDTH or 0
        local right = (UI.RightPanel.Frame and UI.RightPanel.Frame:IsShown()) and K.RIGHT_WIDTH or 0

        container:ClearAllPoints()
        container:SetPoint("BOTTOMLEFT", left + 2, 2)
        container:SetPoint("BOTTOMRIGHT", -right - 2, 2)
        
        editBox:SetWidth(container:GetWidth() - 10)
    end

    container:SetScript("OnSizeChanged", function() editBox:SetWidth(container:GetWidth() - 10) end)
end)

-- ============================================================================
-- NSQC4 / nowim / Utils
-- Вспомогательные функции: UTF-8, время, цвета.
-- ============================================================================

NSQC4.RegisterModule("nowim", function()

    local U = {}
    NSQC4.Nowim.Utils = U

    -- Правильная обрезка строки по байтам (UTF-8)
    function U.TrimToBytes(str, maxBytes)
        if not str then return "" end
        local bytes = 0
        local result = {}
        
        -- Итерируемся по символам UTF-8
        for char in str:gmatch("[%z\1-\127\194-\244][\128-\191]*") do
            local charBytes = #char
            if bytes + charBytes > maxBytes then break end
            table.insert(result, char)
            bytes = bytes + charBytes
        end
        
        return table.concat(result)
    end

    -- Получить текущее время в формате [ЧЧ:ММ]
    function U.GetTimeString()
        return date("%H:%M")
    end

end)
