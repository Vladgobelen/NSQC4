NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local A = {}
    NSQC4.Nowim.Aliases = A

    C.db.aliases = C.db.aliases or {}

    -- Получить "главный" ник. Если это твинк, вернет ник владельца.
    function A.GetMainName(nick)
        if not nick or nick == "" then return nil end
        local lowerNick = nick:lower()
        
        -- Проверяем, не является ли ник основным
        for mainName, alts in pairs(C.db.aliases) do
            if mainName:lower() == lowerNick then
                return mainName
            end
            -- Проверяем твинков
            for _, altName in ipairs(alts) do
                if altName:lower() == lowerNick then
                    return mainName -- Возвращаем основного (например, "ВасяПупкин")
                end
            end
        end
        
        return nick -- Если не найден в связках, возвращаем как есть
    end

    -- Добавить твинка к основному нику
    function A.AddAlt(mainName, altName)
        if not mainName or not altName then return end
        C.db.aliases[mainName] = C.db.aliases[mainName] or {}
        
        -- Проверяем, нет ли уже такого твинка
        for _, name in ipairs(C.db.aliases[mainName]) do
            if name:lower() == altName:lower() then return end
        end
        
        table.insert(C.db.aliases[mainName], altName)
    end
end)

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local S = NSQC4.Nowim.Storage
    local A = {}
    NSQC4.Nowim.Archive = A

    -- Инициализация таблицы архивов
    C.db.archives = C.db.archives or {}

    -- Функция добавления сообщения в архив
    function A.AddToArchive(key, msgData)
        C.db.archives[key] = C.db.archives[key] or {}
        table.insert(C.db.archives[key], msgData)
    end

    -- Функция получения архива
    function A.GetArchive(key)
        return C.db.archives[key] or {}
    end

    -- Функция очистки архива (если понадобится)
    function A.ClearArchive(key)
        if C.db.archives[key] then
            C.db.archives[key] = {}
        end
    end

    -- Слэш-команда для просмотра архива
    SLASH_NOWIM_ARCHIVE1 = "/nowim_archive"
    SlashCmdList["NOWIM_ARCHIVE"] = function(nick)
        nick = (nick or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if nick == "" then
            print("|cffff0000[Nowim]|r Укажите ник: /nowim_archive ИмяНика")
            return
        end

        local messages = A.GetArchive(nick)
        if #messages == 0 then
            print("|cff00ff00[Nowim]|r Архив для " .. nick .. " пуст.")
            return
        end

        print("|cff00ff00[Nowim]|r Архив для " .. nick .. " (" .. #messages .. " сообщений):")
        -- Выводим последние 20 сообщений из архива в чат
        local startIdx = math.max(1, #messages - 20)
        for i = startIdx, #messages do
            local m = messages[i]
            local sender = m.char or "Неизвестно"
            print(string.format("[%s] %s: %s", m.time or "??:??", sender, m.text or ""))
        end
    end
end)

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local S = NSQC4.Nowim.Storage
    local N = NSQC4.Nowim.Network
    local A = NSQC4.Nowim.Aliases

    local E = {}
    NSQC4.Nowim.ChatEngine = E

    C.db.unread = C.db.unread or {}
    C.db.openTabs = C.db.openTabs or {}
    C.db.savedScroll = C.db.savedScroll or {}

    function E.OpenDialog(key, silent)
        if not key or key == "" then return end
        if type(key) ~= "string" then return end

        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }

        local isOpen = false
        for _, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then isOpen = true; break end
        end

        if not isOpen then
            table.insert(C.db.openTabs, key)
            if C.db.settings.autoScroll then
                C.db.savedScroll[key] = 999999
            else
                C.db.savedScroll[key] = 0
            end
        end

        if not silent then
            C.db.activeTab = key
            C.db.unread[key] = 0
        end

        E.UpdateUI()
    end

    function E.CloseTab(key)
        for i, tabKey in ipairs(C.db.openTabs) do
            if tabKey == key then
                table.remove(C.db.openTabs, i)
                break
            end
        end

        if C.db.activeTab == key then
            C.db.activeTab = C.db.openTabs[#C.db.openTabs]
        end

        E.UpdateUI()
    end

    function E.SendMessage(text)
        local target = C.db.activeTab
        if not target then return end
        N.QueueMessage(target, text)
    end

    function E.OnIncomingMessage(sender, targetKey, text)
        if C.db.activeTab ~= targetKey then
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
        end
        E.UpdateUI()
    end

    function E.UpdateUI()
        local UI = NSQC4.Nowim.UI
        if not UI then return end

        if UI.Tabs and UI.Tabs.Refresh then
            UI.Tabs.Refresh()
        end
        if UI.Chat and UI.Chat.Refresh then
            UI.Chat.Refresh()
        end
        if UI.LeftPanel and UI.LeftPanel.Refresh then
            UI.LeftPanel.Refresh()
        end
    end

    -- ========================================================================
    -- КОМАНДЫ
    -- ========================================================================
    SLASH_NOWIM_CHAT1 = "/nowim"
    SlashCmdList["NOWIM_CHAT"] = function(msg)
        msg = (msg or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if msg ~= "" then
            E.OpenDialog(msg)
        end
        if NSQC4.Nowim.UI and NSQC4.Nowim.UI.MainFrame then
            NSQC4.Nowim.UI.MainFrame:ShowFrame()
        end
    end

    SLASH_NOWIM_ARCHIVE1 = "/nowim_archive"
    SlashCmdList["NOWIM_ARCHIVE"] = function(msg)
        msg = (msg or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if msg == "" then
            print("|cffff0000[Nowim]|r Укажите ключ чата: /nowim_archive GUILD, GENERAL или ник")
            return
        end

        local key = msg:upper()
        -- Поиск по нику (если это не канал)
        if not C.db.channels[key] then key = A.GetMainName(msg) end

        local archive = C.db.archives[key]
        
        if not archive or not archive.messages or #archive.messages == 0 then
            print("|cff00ff00[Nowim]|r Архив для " .. key .. " пуст.")
            return
        end

        print("|cff00ff00[Nowim]|r Архив " .. key .. " (" .. #archive.messages .. " сообщений):")
        
        -- Выводим последние 15 сообщений в чат
        local startIdx = math.max(1, #archive.messages - 15)
        for i = startIdx, #archive.messages do
            local m = archive.messages[i]
            print(string.format("[%s] %s: %s", m.time or "??:??", m.char or "???", m.text or ""))
        end
    end

end)

    local C = NSQC4.Nowim.Config
    local myName = UnitName("player")

    local frame = CreateFrame("Frame")
    
    frame:RegisterEvent("CHAT_MSG_GUILD")
    frame:RegisterEvent("CHAT_MSG_OFFICER")
    frame:RegisterEvent("CHAT_MSG_RAID")
    frame:RegisterEvent("CHAT_MSG_RAID_LEADER")
    frame:RegisterEvent("CHAT_MSG_WHISPER")
    frame:RegisterEvent("CHAT_MSG_SAY")
    frame:RegisterEvent("CHAT_MSG_YELL")
    frame:RegisterEvent("CHAT_MSG_PARTY")
    frame:RegisterEvent("CHAT_MSG_PARTY_LEADER")
    frame:RegisterEvent("CHAT_MSG_CHANNEL")
    frame:RegisterEvent("CHAT_MSG_SYSTEM")
    frame:RegisterEvent("CHAT_MSG_RAID_WARNING")
    frame:RegisterEvent("CHAT_MSG_MONSTER_SAY")
    frame:RegisterEvent("CHAT_MSG_MONSTER_YELL")
    frame:RegisterEvent("CHAT_MSG_MONSTER_EMOTE")
    frame:RegisterEvent("CHAT_MSG_BATTLEGROUND")
    frame:RegisterEvent("CHAT_MSG_BATTLEGROUND_LEADER")
    frame:RegisterEvent("CHAT_MSG_EMOTE")
    frame:RegisterEvent("CHAT_MSG_LOOT")
    frame:RegisterEvent("CHAT_MSG_MONEY")
    frame:RegisterEvent("CHAT_MSG_SKILL")

    frame:SetScript("OnEvent", function(self, event, text, sender)
        if not text or text == "" then return end
        if text:sub(1, 5) == "NSMSG" then return end 

        local E = NSQC4.Nowim.ChatEngine
        if not E then return end

        local chatType = event:match("^CHAT_MSG_(.+)$")
        local targetKey = "GENERAL"
        local isMine = (sender == myName)

        local isAO = text:find("^%[O") or text:find("^%[A") or text:find("^%[О") or text:find("^%[А") or text:find("^|c%x%x%x%x%x%x%[O") or text:find("^|c%x%x%x%x%x%x%[A")

        local charTag
        if isAO then
            isMine = false
            charTag = nil
            if sender and sender ~= "" then
                text = text:gsub("^" .. sender .. ":%s*", "")
            end
        else
            charTag = isMine and myName or sender
        end

        -- ОТЛАДКА
        if chatType == "OFFICER" then
            print("|cff00ff00[DEBUG OFFICER]|r Пришло: " .. tostring(text) .. " | От: " .. tostring(sender) .. " | isAO: " .. tostring(isAO))
        end

        if isMine and NSQC4.Nowim._lastSentText == text and (GetTime() - (NSQC4.Nowim._lastSentTime or 0)) < 2 then
            return
        end

        if chatType == "CHANNEL" then
            local _, channelName = GetChannelName(text)
            if channelName then
                local filters = C.db.settings.filters or {}
                if filters[channelName] == false then return end
            end
        end

        local formattedText = text

        if chatType == "SYSTEM" or chatType == "LOOT" or chatType == "MONEY" or chatType == "SKILL" or not sender or sender == "" then
            local filters = C.db.settings.filters or {}
            if filters["SYSTEM"] == false then return end
            S.AddMessage("GENERAL", formattedText, false, nil, chatType)
        else
            if chatType == "OFFICER" then
                print("|cff00ff00[DEBUG OFFICER]|r Сохраняем в OFFICER.")
                S.AddMessage("OFFICER", formattedText, isMine, charTag, chatType)
                
                if C.db.settings.showOfficerInGuild then
                    print("|cff00ff00[DEBUG OFFICER]|r Дублируем в GUILD.")
                    S.AddMessage("GUILD", formattedText, isMine, charTag, chatType)
                end
                
                local hasOfficerTab = false
                for _, key in ipairs(C.db.openTabs) do
                    if key == "OFFICER" then hasOfficerTab = true break end
                end
                if not hasOfficerTab then
                    print("|cff00ff00[DEBUG OFFICER]|r Вкладки OFFICER нет, открываем.")
                    E.OpenDialog("OFFICER", true)
                else
                    print("|cff00ff00[DEBUG OFFICER]|r Вкладка OFFICER уже открыта.")
                end
                
                -- ПРИНУДИТЕЛЬНО ОБНОВЛЯЕМ ЧАТ, ЕСЛИ АКТИВНА ВКЛАДКА OFFICER
                if C.db.activeTab == "OFFICER" then
                    print("|cff00ff00[DEBUG OFFICER]|r Обновляем чат (активна вкладка OFFICER).")
                    local UI = NSQC4.Nowim.UI
                    if UI and UI.Chat and UI.Chat.Refresh then UI.Chat.Refresh() end
                else
                    -- Если вкладка не активна, обновляем, чтобы там были данные
                    local UI = NSQC4.Nowim.UI
                    if UI and UI.Chat and UI.Chat.Refresh then UI.Chat.Refresh() end
                end
                
                E.UpdateUI()
                return
            end

            if chatType == "WHISPER" then
                targetKey = sender
                E.OpenDialog(targetKey, true) 
            elseif chatType == "GUILD" then
                targetKey = "GUILD"
                S.AddMessage("GUILD", formattedText, isMine, charTag, chatType)
                S.AddMessage("GENERAL", formattedText, isMine, charTag, chatType)
            else
                targetKey = "GENERAL"
            end
            
            local filters = C.db.settings.filters or {}
            if filters[chatType] == false then return end

            if chatType ~= "GUILD" then
                S.AddMessage(targetKey, formattedText, isMine, charTag, chatType)
            end
            
            if C.db.activeTab == targetKey then
                local UI = NSQC4.Nowim.UI
                if UI and UI.Chat and UI.Chat.Refresh then UI.Chat.Refresh() end
            end
        end
        
        if C.db.activeTab == targetKey then
            E.UpdateUI()
        else
            C.db.unread[targetKey] = (C.db.unread[targetKey] or 0) + 1
            local UI = NSQC4.Nowim.UI
            if UI and UI.Chat and UI.Chat.Refresh then
                UI.Chat.Refresh()
            end
            E.UpdateUI()
        end
    end)
end)

-- ============================================================================
-- NSQC4 / nowim / Network
-- ============================================================================

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local S = NSQC4.Nowim.Storage

    local N = {}
    NSQC4.Nowim.Network = N

    local sendQueue = {}
    local isSending = false

    local function ProcessQueue()
        if isSending or #sendQueue == 0 then return end
        isSending = true

        local msgData = table.remove(sendQueue, 1)
        local target = msgData.target
        local text = U.TrimToBytes(msgData.text, C.MAX_BYTES)

        local isChannel = C.db.channels[target] ~= nil

        if target == UnitName("player") then
            S.AddMessage(target, text, true, UnitName("player"))
            N.UpdateUI(target)
        elseif isChannel then
            local chatType = C.db.channels[target].type

            NSQC4.Nowim._lastSentText = text
            NSQC4.Nowim._lastSentTime = GetTime()
            NSQC4.Nowim._lastSentTarget = target

            if chatType == "GUILD" or chatType == "RAID" then
                SendChatMessage(text, chatType)
            end

            S.AddMessage(target, text, true, UnitName("player"))
        else
            SendAddonMessage(C.ADDON_PREFIX, string.format("FROM:%s|TEXT:%s", UnitName("player"), text), "WHISPER", target)
            S.AddMessage(target, text, true, UnitName("player"))
        end

        local timer = CreateFrame("Frame")
        local elapsed = 0
        timer:SetScript("OnUpdate", function(self, dt)
            elapsed = elapsed + dt
            if elapsed >= C.SEND_DELAY then
                self:SetScript("OnUpdate", nil)
                isSending = false
                ProcessQueue()
            end
        end)
    end

    function N.QueueMessage(target, text)
        if not target or not text or text == "" then return end
        table.insert(sendQueue, { target = target, text = text })
        ProcessQueue()
    end

    if type(RegisterAddonMessagePrefix) == "function" then

NSQC4.RegisterModule("nowim", function()

    local C = NSQC4.Nowim.Config
    local U = NSQC4.Nowim.Utils
    local A = NSQC4.Nowim.Aliases

    local S = {}
    NSQC4.Nowim.Storage = S

    local function ResolveKey(nick)
        if not nick then return nil end
        local s = tostring(nick)
        if C.db.channels[s] then return s end
        if C.db.channels[nick] then return nick end
        return A.GetMainName(s)
    end

    function S.AddMessage(target, text, isMine, realSender, chatType)
        local key = ResolveKey(target)
        if not key or not text then return end
        if type(key) ~= "string" then return end

        C.db.dialogs[key] = C.db.dialogs[key] or { messages = {} }
        local dialog = C.db.dialogs[key]

        local charTag = realSender or (isMine and UnitName("player") or nil)

        local msgData = {
            text = text,
            isMine = isMine,
            time = U.GetTimeString(),
            char = charTag,
            chatType = chatType -- СОХРАНЯЕМ ТИП
        }

        table.insert(dialog.messages, msgData)

        if #dialog.messages > C.MAX_HISTORY then
            local oldMsg = table.remove(dialog.messages, 1)
            C.db.archives[key] = C.db.archives[key] or { messages = {} }
            table.insert(C.db.archives[key].messages, oldMsg)
        end
    end

    function S.GetMessages(key)
        local resolvedKey = ResolveKey(key)
        if not resolvedKey or not C.db.dialogs[resolvedKey] then return nil end
        return C.db.dialogs[resolvedKey].messages
    end

    function S.ClearHistory(key)
        local resolvedKey = ResolveKey(key)
        if resolvedKey and C.db.dialogs[resolvedKey] then
            C.db.dialogs[resolvedKey].messages = {}
        end
    end

    function S.GetContacts()
        local list = {}
        
        for key in pairs(C.db.dialogs) do
            local s = tostring(key)
            if not C.db.channels[s] and not C.db.channels[key] then
                table.insert(list, { key = s, name = s, isChannel = false })
            end
        end
        
        for id, data in pairs(C.db.channels) do
            local key = tostring(id)
            table.insert(list, { key = key, name = tostring(data.name or id), isChannel = true })
        end

        table.sort(list, function(a, b)
            local nameA = tostring(a.name or "")
            local nameB = tostring(b.name or "")
            if a.isChannel ~= b.isChannel then return a.isChannel == false end
            return nameA < nameB
        end)
        
        return list
    end

end)

        measure:SetFont(fontPath, fontSize)
        measure:SetWordWrap(true)
        measure:SetNonSpaceWrap(true)

        for _, msg in ipairs(messages) do
            local timeStr = C.COLOR_TIME .. "[" .. msg.time .. "]|r "

            local textColor = "|cffffffff"
            if msg.chatType == "SYSTEM" or msg.chatType == "LOOT" or msg.chatType == "MONEY" or msg.chatType == "SKILL" then
                textColor = "|cffffd100"
            elseif msg.isMine then
                textColor = C.COLOR_MY_MSG
            else
                textColor = C.COLOR_OTHER_MSG
            end

            local isAO = msg.text and (msg.text:find("^%[О") or msg.text:find("^%[А") or msg.text:find("^%[O") or msg.text:find("^%[A") or msg.text:find("^|c%x%x%x%x%x%x%[О") or msg.text:find("^|c%x%x%x%x%x%x%[А") or msg.text:find("^|c%x%x%x%x%x%x%[O") or msg.text:find("^|c%x%x%x%x%x%x%[A"))

            local senderPrefix = ""
            if not isAO and not msg.isMine and msg.char then
                senderPrefix = textColor .. msg.char .. ":|r "
            end

            local displayText = msg.text or ""

            if isAO then
                local factionLetter = ""
                if displayText:find("^%[О") or displayText:find("^|c%x%x%x%x%x%x%[О") then factionLetter = "О"
                elseif displayText:find("^%[А") or displayText:find("^|c%x%x%x%x%x%x%[А") then factionLetter = "А"
                elseif displayText:find("^%[O") or displayText:find("^|c%x%x%x%x%x%x%[O") then factionLetter = "О"
                elseif displayText:find("^%[A") or displayText:find("^|c%x%x%x%x%x%x%[A") then factionLetter = "А"
                end

                local channelLetter = (active == "OFFICER") and "O" or "G"
                local prefix = "[" .. channelLetter .. factionLetter .. "] "

                displayText = prefix .. displayText

                if NSQC4.AOClick_ReplaceAll then
                    displayText = NSQC4.AOClick_ReplaceAll(displayText)
                end

                displayText = textColor .. displayText .. "|r"
            else
                displayText = textColor .. displayText .. "|r"
            end

            local fullText = string.format("%s%s%s", timeStr, senderPrefix, displayText)

            local maxTextW = maxBubbleWidth - (BUBBLE_PADDING * 2)

            measure:SetWidth(0)
            measure:SetText(fullText)
            local measuredW = measure:GetStringWidth() or 100

            local finalTextW = math.max(40, measuredW)
            if finalTextW > maxTextW then finalTextW = maxTextW end

            measure:SetWidth(finalTextW)
            measure:SetText(fullText)
            local textHeight = measure:GetStringHeight() or (fontSize + 4)

            local bubble = CreateFrame("Frame", nil, content)

            else
                bg:SetVertexColor(0.2, 0.2, 0.2, 0.9)
                bubble:SetPoint("TOPLEFT", content, "TOPLEFT", PADDING_LEFT, -yOffset)
            end

            local textStr = bubble:CreateFontString(nil, "OVERLAY")
            textStr:SetFont(fontPath, fontSize)
            textStr:SetJustifyH("LEFT")
            textStr:SetWordWrap(true)
            textStr:SetNonSpaceWrap(true)
            textStr:SetPoint("TOPLEFT", bubble, "TOPLEFT", BUBBLE_PADDING, -BUBBLE_PADDING)
            textStr:SetWidth(finalTextW)
            textStr:SetText(fullText)

            yOffset = yOffset + bubble:GetHeight() + GAP
        end

        content:SetHeight(math.max(1, yOffset + PADDING_TOP))
        scroll:UpdateScrollChildRect()
        V.ScrollToBottom()
    end

        scrollBar:SetPoint("BOTTOMRIGHT", scroll, "BOTTOMRIGHT", 0, 4)

        if I._opening then return false end
        if C.db.settings.showInputIdle then return false end
        return not editBox:HasFocus()
    end

    local function GetInputHeight()
        if IsIdleHidden() then return 0 end

        local K = UI.Const or {}
        local minH = K.INPUT_HEIGHT or 30

        local text = editBox:GetText() or ""
        local lines = 1
        for _ in text:gmatch("\n") do lines = lines + 1 end
        local h = math.min(5, lines) * 16 + 10
        if h < minH then h = minH end
        return h
    end

    local lastH = nil

    local function ApplyContainerSize()
        local h = GetInputHeight()

        if h == lastH then
            return
        end
        lastH = h

        container:SetHeight(h)
        if h > 0 then
            container:Show()
            editBox:Show()
            local w = container:GetWidth()
            if not w or w < 20 then w = 200 end
            editBox:SetWidth(w - 10)
        else
            container:Hide()
        end

        if UI.Chat and UI.Chat.OnResize then UI.Chat.OnResize() end
    end

    -- Восстанавливаем перехват Enter на наш чат, не трогая lastChatFocus
    local function RestoreNowimBinding()
        if NSQC4.Nowim.ApplyEnterBinding then
            NSQC4.Nowim.ApplyEnterBinding()
        end

    end)

                    E.SendMessage(U.TrimToBytes(line:trim(), C.MAX_BYTES))
                end
            end
            self:SetText("")

            RestoreNowimBinding()
        end
    end)

    end)

        -- Фокус ушёл (например, после отправки) — переустанавливаем биндинг,

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local UI = NSQC4.Nowim.UI

    local FIELDS = {
        { key = "EDGE_PADDING",   label = "Отступ от края окна" },
        { key = "PANEL_HEIGHT",   label = "Отступ сверху панелей" },
        { key = "LEFT_WIDTH",     label = "Ширина левой панели" },
        { key = "RIGHT_WIDTH",    label = "Ширина правой панели" },
        { key = "TAB_WIDTH",      label = "Ширина вкладки" },
        { key = "TAB_HEIGHT",     label = "Высота вкладки" },
        { key = "BUBBLE_PADDING", label = "Отступ текста в облаке" },
        { key = "GAP",            label = "Зазор между облаками" },
        { key = "CHAT_PADDING_L", label = "Сдвиг чужих сообщений" },
        { key = "CHAT_PADDING_R", label = "Сдвиг своих сообщений" },
        { key = "CHAT_RIGHT_PAD", label = "Зазор справа (скроллбар)" },
        { key = "CHAT_PAD_TOP",    label = "Отступ чата сверху" },
        { key = "CHAT_PAD_BOTTOM", label = "Отступ чата снизу (над вводом)" },
        { key = "CHAT_PAD_PANEL",  label = "Зазор чата до панелей" },
        { key = "INPUT_HEIGHT",   label = "Высота поля ввода" },
    }

    C.db.settings.offsets = C.db.settings.offsets or {}
    local offsets = C.db.settings.offsets
    local base = {}
    local MIN = {
        INPUT_HEIGHT = 5, TAB_WIDTH = 5, TAB_HEIGHT = 5,
        LEFT_WIDTH = 5, RIGHT_WIDTH = 5, GAP = 0, BUBBLE_PADDING = 0,
        CHAT_RIGHT_PAD = 0, CHAT_PAD_TOP = -10, CHAT_PAD_BOTTOM = -10, CHAT_PAD_PANEL = -10,
    }

    local function RefreshBase()
        local K = UI.Const
        if not K then return end
        for _, f in ipairs(FIELDS) do
            base[f.key] = (K[f.key] or 0) - (offsets[f.key] or 0)
        end
    end

    local function ApplyOffset(key)
        local K = UI.Const
        if not K then return end
        local b = base[key] or 0
        local o = offsets[key] or 0
        local v = b + o
        local m = MIN[key]
        if m and v < m then v = m end
        K[key] = v
    end

    local function ApplyAll()
        for _, f in ipairs(FIELDS) do ApplyOffset(f.key) end
    end

    local function Rebuild()
        if UI.ReapplyLayout then UI.ReapplyLayout()
        else
            if UI.Tabs and UI.Tabs.Refresh then UI.Tabs.Refresh() end
            if UI.Chat and UI.Chat.OnResize then UI.Chat.OnResize() end
            if UI.Input and UI.Input.OnResize then UI.Input.OnResize() end
        end
    end

    local win = CreateFrame("Frame", "NSQC4_NowimConstSettings", UIParent)
    win:SetSize(380, 600)
    win:SetPoint("CENTER")
    win:SetFrameStrata("DIALOG")
    win:SetMovable(true)
    win:EnableMouse(true)
    win:SetClampedToScreen(true)
    win:SetBackdrop({
        bgFile = "Interface\\Buttons\\WHITE8x8", edgeFile = "Interface\\Buttons\\WHITE8x8",
        edgeSize = 1, insets = { left = 0, right = 0, top = 0, bottom = 0 },
    })
    win:SetBackdropColor(0, 0, 0, 0.95)
    win:SetBackdropBorderColor(1, 1, 1, 1)
    win:Hide()

    local title = win:CreateFontString(nil, "OVERLAY", "GameFontNormalLarge")
    title:SetPoint("TOP", 0, -8)
    title:SetText("Настройки констант")

    local close = CreateFrame("Button", nil, win, "UIPanelCloseButton")
    close:SetPoint("TOPRIGHT", -2, -2)
    close:SetScript("OnClick", function() win:Hide() end)

    local header = CreateFrame("Frame", nil, win)
    header:SetPoint("TOPLEFT", 30, 0)
    header:SetPoint("TOPRIGHT", -30, 0)
    header:SetHeight(28)
    header:EnableMouse(true)
    header:RegisterForDrag("LeftButton")
    header:SetScript("OnDragStart", function() win:StartMoving() end)
    header:SetScript("OnDragStop", function() win:StopMovingOrSizing() end)

    local scroll = CreateFrame("ScrollFrame", "NSQC4_NowimConstSettingsScroll", win, "UIPanelScrollFrameTemplate")
    scroll:SetPoint("TOPLEFT", 8, -34)
    scroll:SetPoint("BOTTOMRIGHT", -28, 40)

    local content = CreateFrame("Frame", nil, scroll)
    content:SetWidth(330)
    content:SetHeight(1)
    scroll:SetScrollChild(content)

    local ROW_H = 34
    local sliders = {}

    local function MakeSlider(parent, y, field)
        local row = CreateFrame("Frame", nil, parent)
        row:SetPoint("TOPLEFT", 0, -y)
        row:SetPoint("TOPRIGHT", 0, -y)
        row:SetHeight(ROW_H)

        local label = row:CreateFontString(nil, "OVERLAY", "GameFontNormal")
        label:SetPoint("TOPLEFT", 2, 0)
        label:SetText(field.label)

        local valText = row:CreateFontString(nil, "OVERLAY", "GameFontHighlightSmall")
        valText:SetPoint("TOPRIGHT", -2, 0)

        local function UpdateValText()
            local b = base[field.key] or 0
            local o = offsets[field.key] or 0
            valText:SetText(string.format("base=%d  offset=%+d  = %d", b, o, b + o))
        end
        UpdateValText()

        local sliderName = "NSQC4_NowimConstSlider_" .. field.key
        local slider = CreateFrame("Slider", sliderName, row, "OptionsSliderTemplate")
        slider:SetPoint("BOTTOMLEFT", 2, 0)
        slider:SetPoint("BOTTOMRIGHT", -2, 0)
        slider:SetHeight(16)
        slider:SetMinMaxValues(-50, 50)
        slider:SetValueStep(1)
        if slider.SetObeyStepOnDrag then slider:SetObeyStepOnDrag(true) end
        slider:SetValue(offsets[field.key] or 0)

        local low  = _G[sliderName .. "Low"]
        local high = _G[sliderName .. "High"]
        local text = _G[sliderName .. "Text"]
        if low  then low:Hide()  end
        if high then high:Hide() end
        if text then text:Hide() end

        slider:SetScript("OnValueChanged", function(self, value)
            value = math.floor(value + 0.5)
            offsets[field.key] = value
            ApplyOffset(field.key)
            UpdateValText()
            Rebuild()
        end)

        sliders[field.key] = { slider = slider, update = UpdateValText }
        return row
    end

    local y = 0
    for _, f in ipairs(FIELDS) do
        MakeSlider(content, y, f)
        y = y + ROW_H
    end

    local function MakeCheckbox(parent, y, labelText, getValue, setValue)
        local row = CreateFrame("Frame", nil, parent)
        row:SetPoint("TOPLEFT", 0, -y)
        row:SetPoint("TOPRIGHT", 0, -y)
        row:SetHeight(24)

        local cb = CreateFrame("CheckButton", nil, row, "UICheckButtonTemplate")
        cb:SetSize(24, 24)
        cb:SetPoint("LEFT", 2, 0)

        local label = row:CreateFontString(nil, "OVERLAY", "GameFontNormal")
        label:SetPoint("LEFT", cb, "RIGHT", 6, 0)
        label:SetText(labelText)

        local function Update() cb:SetChecked(getValue() and true or false) end
        Update()

        cb:SetScript("OnClick", function(self)
            setValue(self:GetChecked() and true or false)
            Update()
            Rebuild()
        end)
        return row
    end

    y = y + 8
    MakeCheckbox(content, y, "Отображать рамку окна",
        function() return C.db.settings.showBorder end,
        function(v) C.db.settings.showBorder = v end)
    y = y + 28

    MakeCheckbox(content, y, "Отображать строку ввода в покое",
        function() return C.db.settings.showInputIdle end,
        function(v) C.db.settings.showInputIdle = v end)
    y = y + 28

    content:SetHeight(math.max(1, y))

    reset:SetSize(90, 22)
    reset:SetPoint("BOTTOMLEFT", 8, 8)

        for _, f in ipairs(FIELDS) do offsets[f.key] = 0 end
        ApplyAll()
        for _, s in pairs(sliders) do
            s.slider:SetValue(0)
            s.update()
        end
        Rebuild()
    end)

    local btnClose = CreateFrame("Button", nil, win, "UIPanelButtonTemplate")
    btnClose:SetSize(90, 22)
    btnClose:SetPoint("BOTTOMRIGHT", -8, 8)
    btnClose:SetText("Закрыть")
    btnClose:SetScript("OnClick", function() win:Hide() end)

    local function InitWhenReady()
        if not UI.Const then return false end
        RefreshBase()
        ApplyAll()
        for _, s in pairs(sliders) do s.update() end
        return true
    end

    if not InitWhenReady() then
        local f = CreateFrame("Frame")
        f:RegisterEvent("PLAYER_LOGIN")
        f:SetScript("OnEvent", function(self)
            self:UnregisterAllEvents()
            if not InitWhenReady() then
                print("|cffff0000[Nowim]|r ConstSettings: UI.Const так и не появился.")
            end
        end)
    end

    -- ========================================================================

    -- ========================================================================

    UI.ConstSettings = CS

    local buttonsToShow = { 
        "NSQC4_NowimLeftToggle", 
        "NSQC4_NowimRightToggle" 
    }

    local function SetButtonsAlpha(alpha)
        for _, name in ipairs(buttonsToShow) do
            local btn = _G[name]
            if btn then 
                btn:SetAlpha(alpha) 
            end
        end
    end

    -- Функция отладки: показывает в чат, где кнопки и какая у них альфа
    local function DebugButtons()
        print("|cff00ff00[Nowim Debug]|r --- Проверка кнопок ---")
        
        -- Левая кнопка
        local leftBtn = _G["NSQC4_NowimLeftToggle"]
        if leftBtn then
            print("LeftToggle существует. Alpha:", leftBtn:GetAlpha(), "Shown:", tostring(leftBtn:IsShown()))
            print("LeftToggle родитель:", leftBtn:GetParent():GetName() or "Без имени")
        else
            print("|cffff0000LeftToggle НЕ НАЙДЕН!|r")
        end

        -- Правая кнопка
        local rightBtn = _G["NSQC4_NowimRightToggle"]
        if rightBtn then
            print("RightToggle существует. Alpha:", rightBtn:GetAlpha(), "Shown:", tostring(rightBtn:IsShown()))
            print("RightToggle родитель:", rightBtn:GetParent():GetName() or "Без имени")
        else
            print("|cffff0000RightToggle НЕ НАЙДЕН!|r")
        end

        -- Кнопка закрытия окна (крестик)
        local closeBtn = _G["NSQC4_NowimMainFrameCloseButton"]
        if closeBtn then
            print("CloseBtn существует. Alpha:", closeBtn:GetAlpha())
        else
            print("|cffff0000CloseBtn не имеет имени (это нормально, если создан через CreateFrame без имени)|r")
        end
        
        -- Контейнер вкладок
        local tabs = UI.Tabs
        if tabs and tabs.Buttons then
            print("Вкладок создано:", #tabs.Buttons)
        end
        
        print("|cff00ff00[Nowim Debug]|r --- Конец проверки ---")
    end

    function CS.Show()
        win:Show()
        win:Raise()
        NSQC4.Nowim.settingsOpen = true
        SetButtonsAlpha(1)
        DebugButtons() -- ВЫЗЫВАЕМ ОТЛАДКУ ПРИ ОТКРЫТИИ
    end

    function CS.Hide()
        win:Hide()
        NSQC4.Nowim.settingsOpen = false
        SetButtonsAlpha(0.01)
    end

    function CS.Toggle()
        if win:IsShown() then CS.Hide() else CS.Show() end
    end
end)

-- ============================================================================
-- NSQC4 / nowim / UI / Tabs
-- ============================================================================

NSQC4.RegisterModule("nowim", function()
    local C = NSQC4.Nowim.Config
    local E = NSQC4.Nowim.ChatEngine
    local UI = NSQC4.Nowim.UI

    local T = {}
    UI.Tabs = T
    local mainFrame = UI.MainFrame

    C.db.settings.filters = C.db.settings.filters or {
        GUILD = true, RAID = true, PARTY = true, WHISPER = true, SYSTEM = true, SAY = true, YELL = true
    }
    local filters = C.db.settings.filters

    local tabContainer = CreateFrame("Frame", nil, mainFrame)
    tabContainer:SetPoint("BOTTOMLEFT", mainFrame, "TOPLEFT", 30, 0)
    tabContainer:SetPoint("BOTTOMRIGHT", mainFrame, "TOPRIGHT", -30, 0)
    tabContainer:SetHeight(24)
    tabContainer:SetFrameLevel(mainFrame:GetFrameLevel() + 10)

    tabContainer:SetAlpha(0.01)
    tabContainer:EnableMouse(true)

    tabContainer:SetScript("OnUpdate", function(self)
        local hasUnread = false
        for _, key in ipairs(C.db.openTabs or {}) do
            if (C.db.unread[key] or 0) > 0 then hasUnread = true break end
        end
        if hasUnread then self:SetAlpha(1) return end

        local isOver = self:IsMouseOver()
        if not isOver then
            for _, btn in pairs(T.Buttons) do
                if btn:IsMouseOver() then isOver = true break end
            end
        end
        self:SetAlpha(isOver and 1 or 0.01)
    end)

    mainFrame.Tabs = tabContainer
    T.Buttons = {}
    local tabButtons = T.Buttons
    local TAB_WIDTH = 30
    local TAB_HEIGHT = 24

    local function GetTabLetter(key)
        if key == "GENERAL" then return "О" end
        if key == "GUILD" then return "Г" end
        if key == "OFFICER" then return "О" end
        if key == "RAID" then return "Р" end
        if type(key) ~= "string" then return "?" end
        return (key:match("^([%z\1-\127\194-\244][\128-\191]*)") or "?"):upper()
    end

    -- ========================================================================
    -- Меню настроек (ПКМ по вкладке)
    -- ========================================================================
    local menuFrame
    local function ShowTabMenu(btn, key)
        if not menuFrame then
            menuFrame = CreateFrame("Frame", "NSQC4NowimTabMenu", UIParent, "UIDropDownMenuTemplate")
        end

        UIDropDownMenu_Initialize(menuFrame, function(self, level)
            local info = UIDropDownMenu_CreateInfo()

            info.text = "|cffffd100Настройки вкладки:|r"
            info.isTitle = true
            info.notCheckable = true
            UIDropDownMenu_AddButton(info, level)

            -- ГАЛОЧКА ДЛЯ ГИЛЬДИИ
            if key == "GUILD" then
                info = UIDropDownMenu_CreateInfo()
                info.text = "|cff40ff40Показывать гильдейские|r"
                info.checked = C.db.settings.showGuildInGuild
                info.keepShownOnClick = true
                info.func = function()
                    C.db.settings.showGuildInGuild = not C.db.settings.showGuildInGuild
                end
                UIDropDownMenu_AddButton(info, level)

                info = UIDropDownMenu_CreateInfo()
                info.text = "|cff008000Показывать офицерские|r"
                info.checked = C.db.settings.showOfficerInGuild
                info.keepShownOnClick = true
                info.func = function()
                    C.db.settings.showOfficerInGuild = not C.db.settings.showOfficerInGuild
                end
                UIDropDownMenu_AddButton(info, level)
            end

            if key == "OFFICER" then
                info = UIDropDownMenu_CreateInfo()
                info.text = "|cff008000Показывать офицерские|r"
                info.checked = C.db.settings.showOfficerInOfficer
                info.keepShownOnClick = true
                info.func = function()
                    C.db.settings.showOfficerInOfficer = not C.db.settings.showOfficerInOfficer
                end
                UIDropDownMenu_AddButton(info, level)
            end

            if key == "GENERAL" then
                -- Фильтры для общего чата
                info = UIDropDownMenu_CreateInfo()
                info.text = "|cffffd100Фильтры общего чата:|r"
                info.isTitle = true
                info.notCheckable = true
                UIDropDownMenu_AddButton(info, level)

                local filterList = {
                    { key = "GUILD",   text = "Гильдия" },
                    { key = "RAID",    text = "Рейд" },
                    { key = "PARTY",   text = "Группа" },
                    { key = "WHISPER", text = "Шепот" },
                    { key = "SYSTEM",  text = "Системные (Лут/Ачивки)" },
                    { key = "SAY",     text = "Сказать (/say)" },
                    { key = "YELL",    text = "Крик (/yell)" },
                }

                for _, f in ipairs(filterList) do
                    info = UIDropDownMenu_CreateInfo()
                    info.text = f.text
                    info.checked = filters[f.key]
                    info.keepShownOnClick = true
                    info.func = function() filters[f.key] = not filters[f.key] end
                    UIDropDownMenu_AddButton(info, level)
                end

                info = UIDropDownMenu_CreateInfo()
                info.text = "|cffffd100Каналы:|r"
                info.isTitle = true
                info.notCheckable = true
                UIDropDownMenu_AddButton(info, level)

                for id, data in pairs(C.db.channels) do
                    if data.type == "CHANNEL" then
                        info = UIDropDownMenu_CreateInfo()
                        info.text = data.name
                        info.checked = filters[data.name] ~= false
                        info.keepShownOnClick = true
                        info.func = function()
                            filters[data.name] = not info.checked
                        end
                        UIDropDownMenu_AddButton(info, level)
                    end
                end
            end

            -- Общие настройки
            info = UIDropDownMenu_CreateInfo()
            info.text = ""
            info.notCheckable = true
            info.disabled = true
            UIDropDownMenu_AddButton(info, level)

            info = UIDropDownMenu_CreateInfo()
            info.text = "Автопрокрутка"
            info.checked = C.db.settings.autoScroll
            info.func = function() C.db.settings.autoScroll = not C.db.settings.autoScroll end
            UIDropDownMenu_AddButton(info, level)

            info = UIDropDownMenu_CreateInfo()
            info.text = "Перехватывать Enter в Nowim"
            info.checked = C.db.settings.interceptEnter
            info.keepShownOnClick = true
            info.func = function()
                C.db.settings.interceptEnter = not C.db.settings.interceptEnter
                -- Включаем — фокус сразу на Nowim, чтобы сразу работало
                if C.db.settings.interceptEnter then
                    C.db.settings.lastChatFocus = "nowim"
                end
                if NSQC4.Nowim.ApplyEnterBinding then NSQC4.Nowim.ApplyEnterBinding() end
                if NSQC4.Nowim.UpdateSystemChatVisibility then NSQC4.Nowim.UpdateSystemChatVisibility() end
            end
            UIDropDownMenu_AddButton(info, level)
        end)
        ToggleDropDownMenu(1, nil, menuFrame, btn, 0, 0)
    end

    local function CreateTabButton(key)
        local btn = CreateFrame("Button", nil, tabContainer)
        btn:SetSize(TAB_WIDTH, TAB_HEIGHT)
        btn:SetBackdrop({
            bgFile = "Interface\\Buttons\\WHITE8x8",
            edgeFile = "Interface\\Tooltips\\UI-Tooltip-Border",
            edgeSize = 8, insets = { left = 2, right = 2, top = 2, bottom = 0 }
        })
        btn:SetBackdropColor(0.15, 0.15, 0.15, 0.9)
        btn:SetBackdropBorderColor(0.3, 0.3, 0.3, 1)

        local text = btn:CreateFontString(nil, "OVERLAY", "GameFontNormalLarge")
        text:SetPoint("CENTER", 0, 2)
        text:SetText(GetTabLetter(key))
        text:SetTextColor(1, 1, 1)

        local close = CreateFrame("Button", nil, tabContainer)
        close:SetSize(14, 14)
        close:SetNormalTexture("Interface\\Buttons\\UI-GroupLoot-Pass-Up")
        close:SetFrameLevel(tabContainer:GetFrameLevel() + 20)
        close:EnableMouse(true)
        close:Hide()

        close:SetScript("OnClick", function(self, button)
            if button == "LeftButton" then
                E.CloseTab(key)
            end
        end)

        btn:RegisterForClicks("LeftButtonUp", "RightButtonUp")
        btn:SetScript("OnClick", function(self, button)
            if button == "RightButton" then
                ShowTabMenu(self, key)
            else
                E.OpenDialog(key)
            end
        end)

        btn:SetScript("OnEnter", function(self)
            close:Show()
            GameTooltip:SetOwner(self, "ANCHOR_BOTTOM")
            GameTooltip:SetText(key, 1, 1, 1)
            GameTooltip:AddLine("ЛКМ — открыть, ПКМ — настройки", 0.6, 0.6, 0.6)
            GameTooltip:Show()
        end)

        btn:SetScript("OnLeave", function()
            if not close:IsMouseOver() then
                close:Hide()
            end
            GameTooltip:Hide()
        end)

        close:SetScript("OnLeave", function(self)
            if not btn:IsMouseOver() then
                self:Hide()
            end
        end)

        btn.close = close
        tabButtons[key] = btn
        return btn
    end

    function T.Refresh()
        local openTabs = C.db.openTabs
        local active = C.db.activeTab

        for _, btn in pairs(tabButtons) do
            btn:Hide()
            if btn.close then btn.close:Hide() end
        end
        if not openTabs then return end

        local xPos = 0
        for _, key in ipairs(openTabs) do
            local btn = tabButtons[key] or CreateTabButton(key)
            btn:ClearAllPoints()
            btn:SetPoint("BOTTOMLEFT", tabContainer, "BOTTOMLEFT", xPos, 0)
            btn:Show()

            if btn.close then
                btn.close:ClearAllPoints()
                btn.close:SetPoint("TOPRIGHT", btn, "TOPRIGHT", 2, 2)
            end

            local unread = C.db.unread[key] or 0

            if key == active then
                btn:SetBackdropColor(0.1, 0.1, 0.1, 1)
                btn:SetBackdropBorderColor(0.5, 0.5, 0.5, 1)
                btn:SetHeight(TAB_HEIGHT + 2)
            else
                btn:SetBackdropColor(0.15, 0.15, 0.15, 0.9)
                btn:SetBackdropBorderColor(0.3, 0.3, 0.3, 1)
                btn:SetHeight(TAB_HEIGHT)
            end

            if unread > 0 then
                if not btn.RedDot then
                    btn.RedDot = btn:CreateTexture(nil, "OVERLAY")
                    btn.RedDot:SetSize(6, 6)
                    btn.RedDot:SetPoint("TOPLEFT", 2, -2)
                    btn.RedDot:SetTexture("Interface\\Buttons\\WHITE8x8")
                    btn.RedDot:SetVertexColor(1, 0, 0, 1)
                end
                btn.RedDot:Show()
            elseif btn.RedDot then
                btn.RedDot:Hide()
            end

            xPos = xPos + TAB_WIDTH + 2
        end
    end
end)

-- ============================================================================
-- NSQC4 / nowim / Utils
-- Вспомогательные функции: UTF-8, время, цвета.
-- ============================================================================

NSQC4.RegisterModule("nowim", function()

    local U = {}
    NSQC4.Nowim.Utils = U

    -- Правильная обрезка строки по байтам (UTF-8)
    function U.TrimToBytes(str, maxBytes)
        if not str then return "" end
        local bytes = 0
        local result = {}
        
        -- Итерируемся по символам UTF-8
        for char in str:gmatch("[%z\1-\127\194-\244][\128-\191]*") do
            local charBytes = #char
            if bytes + charBytes > maxBytes then break end
            table.insert(result, char)
            bytes = bytes + charBytes
        end
        
        return table.concat(result)
    end

    -- Получить текущее время в формате [ЧЧ:ММ]
    function U.GetTimeString()
        return date("%H:%M")
    end

end)
